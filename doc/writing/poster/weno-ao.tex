\head{Polynomial Approximation on Stencils}
$$
S_j^r(i) = \{I_{i+j-\lfloor{\frac{r}{2}}\rfloor}, \, \ldots \, , I_{i} \, ,\, \ldots \, , I_{i+j+\lfloor{\frac{r-1}{2}}\rfloor}\} 
$$
%where $-\floor{\frac{r-1}{2}} \leq j \leq \floor{\frac{r}{2}}$. 
Obtain the $r$th order stencil polynomial $P^r_j \in \mathcal{P}^{r-1}$ by imposing the interpolation conditions 
$$ \frac{1}{\Delta x_k}\int_{I_k}P^r_j(x)\,dx = \bar{u}_{k+1/2},\quad  \forall I_k \in S_j^r.$$

\head{Smoothness Indicators} \OliveGreen{(Jiang and Shu, 1996)}
\begin{align*}
  \sigma^r_j &= \sum_{\ell = 1}^{r-1} \int_{I_i}(\Delta x_i)^{2\ell-1}\bigg (\frac{d^{\ell}P^r_j(x)}{dx^{\ell}}\bigg )^{\hspace{-1.mm}2}dx.

\end{align*}

\head{Theorem: smooth case}\newline
\subhead{WENO-JS weights}\\ 
\vspace{-.05in}
($+$ sign: $\epsilon_h = \epsilon$, $-$ sign: $\epsilon_h = K\Delta x_i^2, K > 0$.)\\ 
WENO-AO$(r,s)$ achieves accuracy $\cO(h^r)$ if $r-s \leq s\pm 1$.\\
WENO-AO$_s(r_\ell, \ldots,r_1,s)$ achieves optimal convergence\\ \qquad $\cO(r_\ell)$ if $r_\ell-s \leq s\pm 1$.\\
WENO-AO$(r_\ell, \ldots,r_0)$ achieves optimal convergence if\\ \qquad $r_k-r_{k-1} \leq r_0\pm 1, \forall 1\leq k \leq \ell$.\\
\subhead{WENO-Z weights}\\
All of the reconstructions can achieve the optimal order by carefully choosing $\eta$.

\head{Theorem: discontinuous case}\\
\Red{Assume $\sigma \nrightarrow 0$ as the grid is refined.}\\
Take $K > 0$, $\epsilon_h = Kh^2$, conditions on $\eta$.\\
WENO-AO$(r,s)$ drops to order $s$ if $\eta \geq s/2$.\\
WENO-AO$_s(r_\ell, \ldots,r_1,s)$ drops to base order $s$ when $u$\\ \qquad has a discontinuity on $r_{\ell-1}$.\\
WENO-AO$(r_\ell, \ldots,r_0)$ drops to the accuracy of the\\ \qquad largest stencil where $u$ is smooth.

\heading{Summary and Conclusions}
The new reconstruction:
\begin{nlist}
\item Achieves the optimal order of accuracy for a wider choice of levels;
\item Drops to the highest possible order of accuracy near the discontinuity, rather than the base level;
\item All reconstructions can fail on carefully constructed meshes ($\sigma \nrightarrow 0$).
\end{nlist}
