% Title: Mixed Methods for Two-Phase Darcy-Stokes Mixtures of Partially Melted Materials with Regions
% of Zero Porosity
% 
% Speaker: Todd Arbogast (joint with Marc A. Hesse and Abraham L. Taicher)
% 
% Abstract: The Earth's mantle, or an ice sheet, involves a deformable solid matrix phase within which
% a second phase, a fluid, may form due to melting processes.  The mechanics of this system is modeled
% as a dual-continuum, with at each point of space the solid matrix being governed by a Stokes system
% and the fluid melt, if it exists, being governed by a Darcy law.  This system is mathematically
% degenerate when the porosity (volume fraction of fluid) vanishes.  We develop the variational
% framework needed for accurate approximation of this Darcy-Stokes system, even when there are regions
% of positive measure where only one phase exists.  We then develop an accurate mixed finite element
% method for solving the system and show some numerical results.
%

\begin{myslideNoFooter}
\preheading{Mixture Theory for Mantle Dynamics}
\centerline{\OliveGreen{(McKenzie 1984)}}

\bigskip
Based on mixture theory: Both fluid ($f$) and solid matrix ($s$) phases exist at each point of the
domain $\Omega$.

\bigskip
For a quantity $\Psi$, %having both a part in the fluid and in the matrix,
define the \Maroon{mixture variable}
\begin{equation*}
\overline{\Psi} = \phi \Psi_f + (1-\phi) \Psi_s = \Psi_s + \phi \Psi_r,\ \ \Psi_r = \Psi_f - \Psi_s. 
\end{equation*}

\begin{nlist}
\item \parbox[t]{0.59\textwidth}{\raggedright
\Blue{Darcy for fluid melt.}  Fluid melts between rock crystals, forming a porous medium.
\medskip
\item \Blue{Stokes for solid rock.} Highly viscous creep of the solid mantle.
}%(The relative permeability is $K(\phi)=\phi^{2}K_0$ for simplicity.)}%
\raisebox{-48pt}{\parbox{0.34\textwidth}{%
\hfil\picWidth{1.5in}{\pics/pics/Structure.jpg}\!\!\!
\rotatebox{90}{\includegraphics[width=1.6in,trim=1.75in 2.4in 0.75in 1.5in,clip=true]{\pics/pics/olivineRock.jpg}}\hfil}}

\medskip
\item \Blue{Thermodynamics and phase behavior.} Governs the transient behavior of $\phi$.  We assume
  that \Maroon{$\phi$ is given} and consider the mechanics only.
\end{nlist}

\subhead{Notation.}
\parbox{.5\textwidth}{
\begin{dlist}
\item[$\v_f, \v_s$] Phase velocity 
\item[$p_f, p_s$] Phase pressure 
\item[$\rho_f, \rho_s$] Phase density 
\item[$\mu_f, \mu_s$] Phase viscosity 
\item\strut
\item[$\rho_r=\rho_f-\rho_s$\,] Relative density 
\end{dlist}}\parbox{.5\textwidth}{
\begin{dlist}
\item[$\u$] Darcy velocity
\item[$\bar p$] Mixture pressure %$\phi p_f + (1-\phi)p_s$
\item[$\bar\rho$] Mixture density
\item[$\mathbf{g}$] Gravity vector 
\item\strut
\item[$\v_r=\v_f-\v_s$] Relative velocity 
\end{dlist}}

\end{myslideNoFooter}
%==================================================================
\begin{myslide}
\heading{Mixture Theory for Mantle Dynamics: Darcy System}

\begin{nlist}
\item \Blue{Darcy's Law.}  
\vspace*{-2pt}\begin{equation*}
\u = \phi\Maroon{\v_r} = \phi(\v_f-\v_s)
 = -\frac{k_0 \phi^{2+2\Theta}}{\mu_f} (\grad\Maroon{p_f} - \rho_f \mathbf{g})
\end{equation*}

The relative permeability $K(\phi)=\phi^{2+2\Theta}K_0$ for simplicity

\bigskip
\item \Blue{Conservation of mass}\vspace*{-2pt}
\begin{equation*}
\frac{\d}{\d t}\big(\phi\rho_f + (1-\phi)\rho_s\big) + \div\big(\phi\rho_f\v_f + (1-\phi)\rho_s\v_s\big) = 0
\end{equation*}
or
\vspace*{-6pt}\begin{equation*}
\frac{\d}{\d t}\bar\rho + \div\overline{\rho\v} = 0
\end{equation*}

For constant and equal densities for non-buoyancy terms\newline (or a Boussinesq approximation)
\begin{equation*}
\div\bar\v = \div(\phi\Maroon{\v_r} + \Maroon{\v_s}) = 0
\end{equation*}
\end{nlist}

\end{myslide}
%==================================================================
\begin{myslide}
\heading{Mixture Theory for Mantle Dynamics: Stokes System}

\begin{nlistCont}

\item \Blue{Viscous Stokes equation.} Conservation of momentum for
\begin{blist}
\item a highly viscous deformable matrix
\[
\bsigma_s = -p_s\,\mathbf{I}
     + \mu_s\big(2\cD\v_s - \tfrac23\div\v_s\,\mathbf{I}\big)
\]
where $\cD\v_s=\frac12(\grad\v_s + \grad\v_s^T)$
\item and fluid $\bsigma_f=-p_f\,\mathbf{I}$
\item as a mixture
\[
\div\bar\bsigma = -\bar\rho\mathbf{g}
\]
or
\[
-\grad\Maroon{\bar p} + \div\big((1-\phi)\mu_s\big(2\cD\Maroon{\v_s}
                           - \tfrac23\div\Maroon{\v_s}\,\mathbf{I}\big)\big) = -\bar\rho\mathbf{g}
\]
\end{blist}

\item \Blue{Compaction}\vspace*{-16pt}
\begin{equation*}
 \Maroon{p_s} - \Maroon{p_f} = -\frac{\mu_s}{\phi}\div\Maroon{\v_s},
\end{equation*}

\item \Blue{Thermodynamics and phase behavior.} Governs the transient behavior of $\phi$.  We assume
  that \Maroon{$\phi$ is given} and consider the mechanics only.

\end{nlistCont}

\end{myslide}
%==================================================================
\begin{myslide}
\heading{Primary Unknowns}

We have four equations and five primary unknowns:
\begin{blist}
\item $\v_r$ and $\v_s$
\item $p_f$, $p_s$, and $\bar p$
\end{blist}
We choose to eliminate $p_s$ from the system (simplifies Stokes).

%\bigskip We do {\em not\/} eliminate $\v_r$, so the Darcy system remains in \Maroon{mixed form,}
%i.e., a system of two first order equations.

\subhead{Pressure potentials.} Let
\begin{blist}
\item $q_f=p_f - \rho_fgz$
\item $q_s=p_s - \Maroon{\rho_f}gz$ $\implies$ $p_f-p_s=q_f-q_s$
\item $q = \bar q = \phi q_f - (1-\phi)q_s$
\end{blist}

Note that
\[
p_f-p_s = q_f - q_s = \frac{1}{1-\phi}(q_f - q)
\]

\end{myslide}
%==================================================================
\begin{myslide}
\heading{Reformulation}

\Blue{Darcy's law} for the fluid
\begin{alignat*}3
&\u + \frac{k_0\phi^{2+2\Theta}}{\mu_f}\grad q_f &&=0&&\quad\OliveGreen{\text{(Darcy law)}}\\
&\mu_s\div \u + \frac{\phi}{1-\phi}(q_f - q) &&=0&&\quad\OliveGreen{\begin{aligned}&\text{(conservation/}\\&\ \text{compaction)}\end{aligned}}\\
\intertext{\Blue{Stokes system} for a highly viscous, compressible mixture (matrix \& fluid)}
&\grad q - \div\hat\bsigma(\v_s) &&= -(1-\phi)\,\rho_r\,\mathbf{g}\equiv\mathbf{G}&&\quad\OliveGreen{\text{(Stokes)}}\\
&\mu_s\div \v_s - \frac{\phi}{1-\phi}(q_f - q) &&= 0&&\quad\OliveGreen{\text{(compaction)}}
\end{alignat*}
where
\begin{alignat*}2
\hat\bsigma(\v_s) &= 2\mu_s(1-\phi)\big(\cD\v_s - \tfrac{1}{3}\div\v_s\mathbf{I}\big)
&&\quad\OliveGreen{\text{(deviatoric stress of mixture)}}\\
\cD\v_s &= \frac12(\grad\v_s + \grad\v_s^T)&&\quad\OliveGreen{\text{(symmetric gradient)}}
\end{alignat*}

\bigskip
\Blue{Simple boundary conditions} (for the talk)
\begin{equation*}
\u\cdot\nu = 0\quad\text{and}\quad\v_s = \mathbf{0} \quad\text{on }\d\Omega
\end{equation*}

\end{myslide}
%==================================================================
\begin{myslide}
\preheading{Remarks on the Model}

\subhead{Advantage.} A single two-phase model holds when one phase disappears.

Model avoids the free boundary between one and two phase regions.

Important in a \Maroon{time-dependent problem} when the free boundary \Maroon{evolves.}

\subhead{Price.} A degenerate system when the fluid melt disappears ($\phi=0$).
\begin{nlist}
\item Stokes system is well-behaved
\item Darcy system degenerates when $\phi$ vanishes. We will show
\begin{equation*}
\| \phi^{-1-\Theta}\u \| + \|\phi^{-1/2}\div\u\|+ \|\phi^{1/2}q_f\|  \leq C
\end{equation*}
for some constant $C$, where $\|\cdot\|$ is the $L^2(\Omega)$-norm.

\bigskip
Where $\phi=0$,
\begin{blist}
\item $p_f$ (or $q_f$) may be {\em unbounded}.
\item The conservation equation is lost ($0=0$).
\item If we impose a small nonzero porosity $\phi_0$ everywhere for numerical approximation,
the condition number grows as $\phi_0\to0$.
\end{blist}
\end{nlist}

\end{myslide}

\begin{myslide}
\heading{A Standard Weak Formulation for Positive Porosity}

We use the function spaces
\begin{align*}
H_0(\Div;\Omega) &= \big\{\v\in(L^2(\Omega))^d : \div\v\in L^2(\Omega),\ \v\cdot\nu = 0\ \text{on} \ \d\Omega\big\}\\
(H_0^1(\Omega))^d &= \big\{\v\in(H^1(\Omega))^d : \v = \mathbf{0} \ \text{on} \ \d\Omega\big\}\\
L^2(\Omega)/\Re &= \Big\{w\in L^2(\Omega) : \int_{\Omega}w(x)\,dx = 0\Big\}
\end{align*}

\subhead{Standard weak formulation.}
Assume $\phi\ge\phi_*>0$, but consider $\phi_*\to0$.

\bigskip
Find $\u\in H_0(\Div)$, $q_f\in L^2$, $\v_s\in (H_0^1)^d$, and $q\in L^2/\Re$ such that
\begin{alignat*}3
&\Big ( \frac{\mu_f}{k_0}\phi^{-2-2\Theta} \u, \bpsi_r \Big ) - (q_f, \div \bpsi_r)
          &&= 0&&\quad\forall\,\bpsi_r \in H_0(\Div)\\
&(\div \u,w_f) + \Big ( \frac{\phi}{\mu_s(1-\phi)} (q_f-q),w_f \Big ) &&= 0&&\quad\forall w_f \in L^2\\
&\!-\!(q,\div \bpsi_s) + (\hat\bsigma(\v_s),\grad\bpsi_s)
          &&= (\mathbf{G},\bpsi_s)&&\quad\forall\,\bpsi_s\in (H_0^1)^d\\
&(\div \v_s,w) - \Big ( \frac{\phi}{\mu_s(1-\phi)} (q_f - q),w \Big )
         &&= 0&&\quad\forall w \in L^2/\Re
\end{alignat*}
%We used integration by parts 3 times.

We can take the test functions
\vspace*{-6pt}\begin{align*}
\bpsi_r=\u\in H_0(\Div)\quad&\text{and}\quad w_f=q_f\Blue{\,+\,\phi^{-1}\div\u}\in L^2\\
\bpsi_s=\v_s\in(H_0^1)^d\quad&\text{and}\quad w=q\in L^2/\Re
\end{align*}

\end{myslide}
%==================================================================
\begin{myslide}
\preheading{Energy Estimates}

\subhead{Theorem.} If $\phi\ge\phi_*>0$ and a solution exists, then
\begin{equation*}
\|\phi^{-1-\Theta}\u\|  + \|\phi^{-1/2}\div\u\| + \|\Maroon{\phi^{1/2}q_f}\| + \|\v_s\|_1 + \|q\|
\le C\|\phi^{1/2}q_s\|
\end{equation*}

\subhead{Problems.} If $\phi=0$ (where there is no fluid!):
\begin{blist}
\item We have no control on the size of the fluid potential $q_f$, so $q_f\not\in L^2$
\item We lose the conservation equation $\div(\phi\v_r) + \phi q_f = \phi q_s$.
\item Some of the integrals are ill defined as $\phi\to0$ (division by $\phi$).
\end{blist}

\subhead{Key idea.} \Maroon{Scale the variables} so these energy bounds make sense.

\bigskip
Define a {\em scaled potential and velocity}\vspace*{-8pt}
\[
\myfbox{\quad\tilde q_f = \phi^{1/2}q_f\quad\text{and}\quad \tilde\v_r = \phi^{-1-\Theta}\u= \phi^{-\Theta}\v_r\quad}
\]

\end{myslide}
%==================================================================
\begin{myslideNoFooter}
\preheading{The Scaled Equations}

\Blue{Darcy's law} for the fluid
\begin{alignat*}3
&\tilde\v_r + \frac{k_0\Maroon{\phi^{1+\Theta}}}{\mu_f}\Maroon{\grad(\phi^{-1/2}\tilde q_f)} &&= 0
   &&\quad\OliveGreen{\text{(Darcy law)}}\\
&\mu_s\Maroon{\phi^{-1/2}\div(\phi^{1+\Theta}\tilde\v_r)}
         + \frac{1}{1-\phi}\big(\tilde q_f - \phi^{1/2}q\big) &&= 0
   &&\quad\OliveGreen{\begin{aligned}&\text{(conservation/}\\&\ \text{compaction)}\end{aligned}}\\
\intertext{\Blue{Stokes system} for a highly viscous, compressible matrix plus fluid}
&\grad q - \div\hat\bsigma(\v_s) &&= \mathbf{G}
    &&\quad\OliveGreen{\text{(Stokes)}}\\
&\mu_s\div \v_s - \frac{\phi^{1/2}}{1-\phi}\big(\tilde q_f - \phi^{1/2}q\big) &&= 0
    &&\quad\OliveGreen{\text{(compaction)}}
\end{alignat*}

\bigskip
\Blue{Simple boundary conditions}
\begin{equation*}
\phi^{1+\Theta}\tilde\v_r\cdot\nu = 0\quad\text{and}\quad\v_s = \mathbf{0}\quad\text{on }\d\Omega
\end{equation*}

%\subhead{Assumption.} The divergence and gradient terms are well defined if
%\vspace*{-4pt}\begin{equation*}
%\myfbox{\phi\in L^\infty(\Omega)\text{ and }\phi^{\Theta-1/2}\grad\phi\in(L^\infty(\Omega))^d}
%\end{equation*}

\subhead{Remarks.} Where $\phi=0$:
\begin{blist}
\item The scaled fluid pressure $\tilde q_f$ remains defined and bounded
\item We do not lose the second equation
%\item Ignoring gradients, terms divided by $\phi$ are zero when $\phi=0$
\end{blist}

\subhead{Questions.} 
\begin{blist}
\item Does $\phi^{-1/2}\div(\phi^{1+\Theta}\tilde\v_r)$ make mathematical sense?
\item What Hilbert space does $\tilde\v_r$ lie within?
\end{blist}

\end{myslideNoFooter}
%==================================================================
\begin{myslide}
\heading{The Space $H_\phi(\Div)$}

Formally,\vspace*{-12pt}
\[
\phi^{-1/2}\div(\phi^{1+\Theta}\v)
 = (1+\Theta)\phi^{\Theta-1/2}\grad\phi\cdot\v + \phi^{1/2+\Theta}\div\v
\]

\subhead{Assumption.}
$\myfbox{\phi\in L^\infty(\Omega)\text{ and }\phi^{\Theta-1/2}\grad\phi\in(L^\infty(\Omega))^d}$

\subhead{Define:}\vspace*{-20pt}
\[
H_\phi(\Div;\Omega)
= \big\{\v\in(L^2(\Omega))^d\,:\,\phi^{-1/2}\div(\phi^{1+\Theta}\v)\in L^2(\Omega)\big\}
\]
That is, $\phi^{1/2+\Theta}$ times the weak divergence of $\v$ lies in $L^2$.

\subhead{Lemma.}
$H_\phi(\Div;\Omega)$ is a Hilbert space with the inner-product\vspace*{-6pt}
\begin{equation*}
(\u,\v)_{H_\phi(\Div)} = (\u,\v) + \big(\phi^{-1/2}\div(\phi^{1+\Theta}\u),\phi^{-1/2}\div(\phi^{1+\Theta}\v)\big).
\end{equation*}\vspace*{-20pt}\newline
Moreover, $H(\Div;\Omega) = H_1(\Div;\Omega) \subset H_\phi(\Div;\Omega)$.

\end{myslide}
%==================================================================
\begin{myslide}
\heading{A Normal Trace Operator}

We wish to apply the homogeneous Neumann boundary condition.

\subhead{Define:} $\gamma_\phi:H_\phi(\Div;\Omega)\to H^{-1/2}(\d\Omega)$ by the integration by
parts formula
\begin{equation*}
\<\gamma_\phi(\v),w\>
= \big(\phi^{-1/2}\div(\phi^{1+\Theta}\v),w\big) + \big(\v,\phi^{1+\Theta}\grad(\phi^{-1/2}w)\big),
\end{equation*}
wherein $w\in H^{1/2}(\d\Omega)$ has been extended to $w\in H^1(\Omega)$. 

\bigskip
We should interpret $\gamma_\phi(\v)=\phi^{1/2+\Theta}\v\cdot\nu$. 

\subhead{Lemma.}
The trace operator
\vspace*{-6pt}\[
\gamma_\phi:H_\phi(\Div;\Omega)\to H^{-1/2}(\d\Omega)
\]
is well defined and
\begin{equation*}
\|\phi^{1/2+\Theta}\v\cdot\nu\|_{-1/2,\d\Omega} \le C\|\v\|_{H_\phi(\Div)}\quad\forall\v\in H_\phi(\Div;\Omega)
\end{equation*}

\bigskip
\subhead{Define:}\vspace*{-12pt}
\begin{equation*}
H_{\phi,0}(\Div;\Omega) = \big\{\v\in H_\phi(\Div;\Omega)\ :
\ \phi^{1/2+\Theta}\v\cdot\nu=0\text{ on }\d\Omega\big\}
\end{equation*}

\end{myslide}
%==================================================================
\begin{myslide}
\preheading{A Scaled Weak Formulation}

\vspace*{-8pt}\begin{align*}
H_{\phi,0}(\Div;\Omega) &=\big\{\v\in(L^2(\Omega))^d :\\
&\qquad\phi^{-1/2}\div(\phi^{1+\Theta}\v)\in L^2(\Omega),\ \phi^{1/2+\Theta}\v\cdot\nu=0\text{ on }\d\Omega\big\}
\end{align*}

\subhead{Scaled weak formulation.}

Find $\tilde\v_r\in H_{\phi,0}(\Div)$, $\tilde q_f\in L^2$, $\v_s \in(H_0^1)^d$, and $q\in L^2/\Re$ such that
\begin{alignat*}3
&\Big ( \frac{\mu_f}{k_0} \tilde{\v}_r, \bpsi_r \Big )
            - \big(\tilde q_f, \Maroon{\phi^{-1/2}\div (\phi^{1+\Theta}\bpsi_r)}\big)
 &= 0\ &&&\forall\,\bpsi_r \in H_{\phi,0}(\Div)\\
&\big(\Maroon{\phi^{-1/2}\div(\phi^{1+\Theta}\tilde{\v}_r)},w_f\big)
               + \Big(\frac1{\mu_s(1-\phi)}(\tilde q_f-\phi^{1/2}q),w_f\Big) &= 0\ &&&\forall w_f \in L^2\\
&\!-\!(q,\div \bpsi_s) + (\hat\bsigma(\v_s),\grad\bpsi_s) &\llap{$= (\mathbf{G},\bpsi_s)$}\ &
  &&\forall\,\bpsi_s \in(H_0^1)^d\\
&(\div \v_s,w) - \Big(\frac{\phi^{1/2}}{\mu_s(1-\phi)}(\tilde q_f - \phi^{1/2}q),w\Big)
  &= 0\ &&&\forall w\in L^2/\Re
\end{alignat*}

\end{myslide}
%==================================================================
\begin{myslide}
\heading{Analysis of the Scaled Model}

\subhead{Theorem.} There exists a unique solution and
\begin{align*}
&\| \tilde\v_r\| + \|\phi^{-1/2}\div(\phi^{1+\Theta}\tilde\v_r)\|+ \|\tilde q_f\|
 + \|\v_s\|_1 +  \|q\| \leq C|\rho_r|.
\end{align*}
\[

\begin{myslide}
\heading{A Scaled Mixed Finite Element Method}

\subhead{Scaled mixed finite element method.}

Find $\tilde\v_{r,h}\in\V_\RT$, $\tilde q_{f,h}\in\W_{\RT}$, $\v_{s}\in\V_\St$, and $q_{h}\in\W_\St$ such that
\begin{alignat*}2
&\Big(\frac{\mu_f}{k_0} \tilde{\v}_{r,h}, \bpsi_r \Big )
            - \big(\tilde q_{f,h}, \phi^{-1/2}\div (\phi^{1+\Theta}\bpsi_r)\big)
 &= 0\ &\forall\,\bpsi_r \in\V_\RT\\
&\big(\phi^{-1/2}\div(\phi^{1+\Theta}\tilde{\v}_{r,h}),w_f\big)
               + \Big(\frac1{\mu_s(1-\phi)}(\tilde q_{f,h}-\phi^{1/2}q_{h}),w_f\Big) &= 0\ &\forall w_f \in \W_\RT\\
&\!-\!(q_{h},\div \bpsi_s) + (\hat\bsigma(\v_{s,h}),\grad\bpsi_s) &\llap{= $(\mathbf{G},\bpsi_s)$}\ &\forall\,\bpsi_s \in\V_\St\\
&(\div \v_{s,h},w) - \Big(\frac{\phi^{1/2}}{\mu_s(1-\phi)}(\tilde q_{f,h} - \phi^{1/2}q_{h}),w\Big)
  &= 0\ &\forall w\in\W_{\St}
\end{alignat*}

\subhead{Lemma.} There exists a unique solution to the discrete equations.

\subhead{Concern:} How do we handle $\phi^{-1/2}\div(\phi^{1+\Theta}\v)$ in a code?

\bigskip
It scales as $\phi^{1/2+\Theta}$, so set it to zero when $\phi=0$ (at a quadrature point).

\end{myslide}

of freedom:
\vspace*{-6pt}\begin{alignat*}5
\u &= \phi^{1+\Theta}\tilde\v_r &&\quad\implies\quad
   &&\u_h\cdot\nu|_e &&= \Maroon{\frac1{|e|}\int_e\phi^{1+\Theta}\,ds}\,\,\tilde\v_{r,h}\cdot\nu|_e&&\quad\forall e\in\cE_h\\
q_f &= \phi^{-1/2}\tilde q_f&&\quad\implies\quad
   &&q_{f,h}|_E &&= \Maroon{\phi_E^{-1/2}}\tilde q_{f,h}|_E&&\quad\forall E\in\cT_h
\end{alignat*}

We arbitrarily set $q_{f,h}|_E=0$ if $\ds\phi_E=\frac1{|E|}\int\phi\,dx=0$.

\subhead{$\RT$-projections.}
\begin{blist}
\item $\cP_\RT=\hat{\cdot}:L^2(\Omega)\to\W_{\RT}$ is $L^2(\Omega)$-projection onto piecewise constants
\item $\pi_\RT:H(\Div;\Omega)\cap L^{2+\epsilon}(\Omega)\to\V_\RT$ (any $\epsilon>0$) is the
standard Raviart-Thomas or Fortin operator % that preserves

%\vspace*{-8pt}\begin{bblist}

%\item element average divergence

\end{blist}\vspace*{-8pt}

\subhead{$\St$-projections.}
\begin{blist}
\item $\cP_\St:L^2(\Omega)\to\W_\St$ is $L^2(\Omega)$-projection onto $\W_\St$
\item $\pi_\St:H^1(\Omega)\to\V_\St$ is $H^1(\Omega)$-projection onto $\V_\St$
\end{blist}

\subhead{Theorem.} If $\cT_h$ is quasi-uniform,
\begin{align*}
&\|\tilde{\v}_{r}-\tilde{\v}_{r,h}\|
 + \big\|\Maroon{\cP_{\RT}}\big[\phi^{-1/2}\div(\phi^{1+\Theta}(\tilde{\v}_{r}-\tilde{\v}_{r,h}))\big]\big\|
 + \|\v_{s}-\v_{s,h}\|_1\\
&\qquad
 + \|\tilde q_{f}-\tilde q_{f,h}\|
 + \|q-q_h\|\\
&\quad\le C\big\{
     \|\tilde{\v}_{r} - \pi_\RT\tilde{\v}_{r}\|
 + \big\|\cP_{\RT}\big[\phi^{-1/2}\div (\phi^{1+\Theta}(\tilde{\v}_{r}
                                                - \pi_\RT\tilde{\v}_{r}))\big]\big\|\\
&\qquad\quad
 + \|q-\cP_{\St}q\| + \|\v_s-\pi_\St\v_{s}\|_1
 + \|\tilde q_{f}-\cP_{\RT}\tilde q_{f}\|\big\}
\end{align*}

\begin{myslide}
\heading{A Modification for Local Mass Conservation}

The mixed finite element method is not locally mass conservative.

\subhead{Locally average $\phi$.} Let $\hat{\phi}=\cP_{\W_\RT}\phi\in\W_\RT$, so
\begin{equation*}
\text{For }\mathbf{x}\in E,\quad\hat{\phi}(\mathbf{x}) = \phi_E = \frac1{|E|}\int_E\phi\,dx
\end{equation*}
When $\hat\phi|_E=\phi_E=0$, $\phi$ is identically zero on $E$.

\subhead{Modification.} Replace
\begin{align*}
\Maroon{\phi^{-1/2}}\grad\!\cdot\!(\phi^{1+\Theta}\v)\big|_E\quad\text{by}\quad\left\{
\begin{alignedat}2
&\Maroon{\phi_E^{-1/2}}\div(\phi^{1+\Theta}\v) &&\quad\!\text{if }\phi_E\ne0,\\
&0 &&\quad\!\text{if }\phi_E=0
\end{alignedat}\right.
\end{align*}
and modify two terms involving the pressure potentials.

\end{myslide}

\begin{myslide}
\preheading{A Locally conservative method}

\subhead{Locally conservative scaled mixed finite element method.} Find
$\tilde\v_{r,h}\in\V_{\RT}$, $\tilde q_{f,h}\in\W_{\RT}$,
$\v_{s,h}\in\V_\St$, and $q_{h}\in\W_\St$ such that
\begin{alignat*}2
&\Big(\frac{\mu_f}{k_0} \tilde{\v}_{r,h}, \bpsi_r \Big )
            - \big(\tilde q_{f,h}, \Maroon{\hat\phi^{-1/2}}\div (\phi^{1+\Theta}\bpsi_r)\big)
 &= 0\ &\forall\,\bpsi_r \in\V_{\RT}\\
&\big(\Maroon{\hat\phi^{-1/2}}\div(\phi^{1+\Theta}\tilde{\v}_{r,h}),\OliveGreen{w_f}\big)
        + \Big(\frac{\Maroon{\phi\hat\phi^{-1}}}{\mu_s(1-\phi)}(\tilde q_{f,h}-\hat\phi^{1/2}q_{h}),\OliveGreen{w_f}\Big)
&= 0\ &\forall \OliveGreen{w_f \in \W_\RT}\\
&\!-\!(q_{h},\div \bpsi_s) + (\hat\bsigma(\v_{s,h}),\grad\bpsi_s) &\llap{$= (\mathbf{G},\bpsi_s)$}
  \ &\forall\,\bpsi_s \in\V_\St\\
&(\div \v_{s,h},\OliveGreen{w}) - \Big(\frac{\Maroon{\phi\hat\phi^{-1/2}}}{\mu_s(1-\phi)}(\tilde q_{f,h} - \hat\phi^{1/2}q_{h}),\OliveGreen{w}\Big)
  &= 0\ &\forall \OliveGreen{w\in\W_{\St}}
\end{alignat*}
Where $\hat\phi|_E=\phi_E=0$, set $\hat\phi^{-1/2}=0$ and interpret $\phi\hat\phi^{-1}=1$.

\subhead{Local conservation.} Take
\[
w_f = \hat\phi^{1/2} \in\W_\RT\qquad\text{and}\qquad w = 1 \in\W_\St
\vspace*{-12pt}
\]
$\implies$
\[
0 = \int_E\big[\div(\phi^{1+\Theta}\tilde{\v}_{r,h}) + \div\v_{s,h}\big]dx = \int_E\big[\div\u_{h} + \div\v_{s,h}\big]dx
\]

\heading{A Closed Form Solution}

Let $\Omega=(-1,1)$ and\vspace*{-6pt}
\[
\phi(x) = \begin{cases}
0,&x<0\\
x^2,&x>0
\end{cases}
\quad\text{and}\quad f(x) = x^{1+\beta},\quad 0<x<1
\]

Then we have the Euler equation\vspace*{-4pt}
\[
\myfbox{-x^2\,q_f'' - 4x\,q_f' + q_f = x^\beta,\quad 0<x<1}
\]
The Euler exponents are
\begin{equation*}
r_1 = \frac{-3 + \sqrt{13}}{2} \approx 0.3 > 0
\quad\text{and}\quad
r_2 = \frac{-3 - \sqrt{13}}{2} \approx -3.3 < 0
\end{equation*}
Since $u=-\phi q_f'\in L^2(0,1)$, $x^{r_2}$ is not part of the solution.

\bigskip
If $\beta\ne r_1,r_2$, we have the closed form solution\vspace*{-4pt}
\begin{gather*}
q_f(x) = \begin{cases}0,&-1<x\le0\\
\dfrac{\beta
  x^{r_1}-r_1x^\beta}{r_1(\beta-r_1)(\beta-r_2)},&\00<x<1\end{cases}
\quad\text{and}\quad\tilde{q}_f(x) = x\,q_f(x)\\
u(x) = \begin{cases}0,&-1<x\le0\\
\dfrac{-\beta(x^{r_1+1}-x^{\beta+1})}{(\beta-r_1)(\beta-r_2)},&\00< x<1\end{cases}
\end{gather*}

\end{myslideNoFooter}

\preheading{1-D Compacting Column Discontinuous Porosity---1}

\[
\phi(z) = \begin{cases}  \phi_-&\text{if } z \leq 0\\
                                            \phi_+&\text{if } z > 0\end{cases}
\]\vspace*{-24pt}

\subhead{Solution.}
\begin{equation*}
u_\pm = \phi_\pm^{2+2\Theta}(1 -\phi_\pm)\big[1 + a_\pm\cosh(R_\pm z) + b_\pm\sinh(R_\pm z) \big]
\end{equation*}

\bigskip\PineGreen{$\bullet$} If both $\phi_\pm>0$, let $\cF_\pm=\phi_\pm^{2+2\Theta}(1 -\phi_\pm)$ and solve
\begin{align*}
\!\!&\left[\begin{matrix}
\cosh(R_+L) & 0 & \sinh(R_+L) &\ \,0\\
0 & \cosh(R_-L) & 0 &\ \,-\sinh(R_-L)\\
\cF_+& -\cF_- & 0 &\ \,0\\
0 & 0 & R_-(1-\phi_+) &\ \,-R_+(1-\phi_-)
\end{matrix}\right]\!\left[\begin{matrix}
a_+\\a_-\\b_+\\b_-
\end{matrix}\right]
=
\left[\begin{matrix}
-1\\-1\\\cF_- - \cF_+ \\0
\end{matrix}\right]
\end{align*}

\bigskip\PineGreen{$\bullet$} If $\phi_-=0$, $a_-=b_-=0$ ($u_-=v_{m-}=0$)
\begin{equation*}
a_+ = -1
\qquad b_+ = \frac{\cosh(R_+L) - 1}{\sinh(R_+L)}
\qquad q_{m-} = x - b_+\frac{1-\phi_+}{R_+}
\end{equation*}

\preheading{1-D Compacting Column Quadratic Porosity Approx.---1} 

\[
\phi(z) = \begin{cases} 0&\text{if } z \leq 0,\\
                                       \phi_+z^2&\text{if } z > 0,\end{cases}

\subhead{Solution.}  Set $\Theta=0$ and \Maroon{$\phi_+\ll1$.}
Approximately reduce to an Euler equation.
\begin{align*}
r_1 = \frac{3+\sqrt{9+4/\phi_+}}{2} > 3
\qquad
r_2 = \frac{3-\sqrt{9+4/\phi_+}}{2} < 0
\end{align*}

\bigskip\PineGreen{$\bullet$} For $L>z>0$,
\begin{align*}
u &= - v_s = \frac{\phi_+^2}{1-4\phi_+}\big(L^{4-r_1}\,z^{r_1} - z^4\big)\\[4pt]
q_f &= \frac{1}{1-4\phi_+}\Big(z - \frac{L^{4-r_1}\,z^{r_1-3}}{r_1-3}\Big)\\
q_s & = z
\end{align*}

\bigskip\PineGreen{$\bullet$} For $-L<z<0$, $u=v_s=0$ and $q_s=z$

\begin{myslide}
\preheading{Summary and Conclusions\,---\,1}

\begin{nlist}
\item We considered a \Blue{two phase mixture} of matrix and fluid melt
\vspace*{-6pt}\begin{blist}
\item The \Maroon{equations degenerate} as the porosity $\phi$ vanishes
\item Question: are the equations well-posed?
\end{blist}

\item We saw the \Blue{energy estimates}
\vspace*{-6pt}\begin{align*}
&\|\Maroon{\phi^{-1-\Theta}\u}\| + \|\phi^{-1/2}\div\u\|+ \|\Maroon{\phi^{1/2}q_f}\|
       + \|\v_s\|_1 + \|q\| \leq C
\end{align*}
\begin{blist}
\item The potential $q_f$ is uncontrolled in the energy estimates
\item One equation is lost when $\phi=0$
\end{blist}

\item We \Blue{scaled the Darcy system} 
\begin{blist}
\item $\tilde q_f = \phi^{1/2}q_f$ and $\tilde\v_r=\phi^{1-\Theta}\u$
\item Equations made sense provided $\phi\in L^\infty(\Omega)$
and $\phi^{\Theta-1/2}\grad\phi\in(L^\infty(\Omega))^d$
\end{blist}

\item We defined \Blue{$H_\phi(\Div)$} and a normal trace operator, which gave
\vspace*{-6pt}\begin{blist}
\item A scaled weak formulation
\item \Maroon{Existence and uniqueness}  of a solution
\item \Maroon{Continuous dependence} on the data
\end{blist}

\end{nlist}

\end{myslide}
%==================================================================
\begin{myslideNoFooter}
\preheading{Summary and Conclusions\,---\,2}

\begin{nlistCont}

\item We defined a \Blue{mixed finite element method}
\vspace*{-6pt}\begin{blist}
\item Proved \Maroon{optimal convergence} rates of $\cO(h)$
\item Modified the method for local mass conservation
\item Implemented using mass lumping to eliminate $\tilde\v_r$
\begin{bblist}
\item Stokes-like system with two pressures
%\item Schur complement for two pressures
\end{bblist}
\end{blist}

\item \Blue{Numerical results} for a simple degenerate Darcy problem
\begin{blist}
\item Closed form Euler equation tests showed
\vspace*{-6pt}\begin{bblist}
\item optimal convergence up to the \Maroon{regularity of the solution}
\item \Maroon{Superconvergence}
\end{bblist}
\item Two-dimensional tests showed
\vspace*{-6pt}\begin{bblist}
\item Superconvergence
\item Importance of $\phi^{-1/2}\grad\phi\in L^\infty$ (actually, \Maroon{$\phi^{-1/2}\grad\phi\in L^2$})
\item Mesh need not resolve the one to two phase boundary
\end{bblist}
\end{blist}

\item \Blue{Numerical mantle dynamics tests} showed promise and 
\vspace*{-6pt}\begin{blist}
\item Captured solution even for \Maroon{discontinuous porosity}
\item \Maroon{Low condition numbers} for the linear system
\end{blist}

\end{nlistCont}

\vspace*{-16pt}
\[
\myfbox{\parbox{8.5in}{\begin{center}
When $\phi$ is reasonable, we have an easy to implement\\
and convergent mixed finite element method
%\\for this degenerate problem of mantle dynamics
\end{center}}}
\]
