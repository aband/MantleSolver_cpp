\subheading{Mixture Theory} %\Gray{(McKenzie 1984)}}
\vspace*{-0.3cm}
Both liquid ($l$) and solid ($s$) phases
exist at each point of the domain $\Omega$.

Define the \Maroon{mixture variable} (phase averaged) as:
\begin{equation*}
\begin{aligned}
\overline{\Psi} &= \phi \Psi_l + (1-\phi) \Psi_s \\ 
&= \Psi_s + \phi \Psi_r,\\
\Psi_r &= \Psi_l - \Psi_s. 
\end{aligned}
\end{equation*}}
}

\bigskip
\subhead{Some assumptions.}
\begin{blist}
\item Density difference between phases are neglected expect
in the \Oorange{body force term}.
\item Constant and uniform material thermodynamics properties among phases.
\item Thermodynamical equilibrium is achieved at each time step.
\end{blist}

\begin{myslide}
\heading{McKenzie Equations}
		  \begin{center}
					 \OliveGreen{\small(McKenzie 1984)}
		  \end{center}
\subhead{A two-phase formulation of a viscous mantle.}
\begin{equation*}
\begin{alignedat}{3}
		  &\phi_l (\mathbf{v}_l - \mathbf{v}_s)
		  + \frac{k_{\phi_l}}{\mu_l}(\nabla p_l - \rho_f\mathbf{g})
	    &&= 0, \qquad &&\Maroon{\text{(Darcy)}}\\
%		  &\nabla \cdot (\phi_l\mathbf{v}_l + \phi_s\mathbf{v}_s) &&=0,\\
		  &\nabla\bar p - \nabla \cdot \pmb{\tau}(\mathbf{v}_s) + \overline{\rho}\mathbf{g}&&= 0, \qquad &&\Maroon{\text{(Stokes)}}\\
%		  &\mu_s\nabla \cdot \mathbf{v}_s - \phi_l(p_l - p_s) &&=0, 
&\frac{\partial \rho_l \phi}{\partial t} + \nabla\cdot (\rho_l \phi \mathbf{v}_l) &&= \Oorange{\Gamma}  \qquad && \\
&\frac{\partial \rho_s (1-\phi)}{\partial t} + \nabla\cdot (\rho_s (1-\phi) \mathbf{v}_s) &&= \Oorange{-\Gamma}  \qquad && 
\end{alignedat}
\end{equation*}
\Oblue{Deviatoric stress of mixture}
\begin{equation*}\label{eq:deviatoricstress}
		  \pmb{\tau}(\mathbf{v}_s)
		  = 2\mu_s\phi_s\big( \tfrac{1}{2}(\nabla \mathbf{v}_s + \nabla \mathbf{v}_s^T) - \tfrac{1}{3}\nabla \cdot \mathbf{v}_s \mathbf{I}\big)
\end{equation*}

\parbox{0.5\textwidth}{\raggedright
\subhead{Unknowns.} \\
\Oorange{Four} equations and \Oorange{five} unknowns.
\begin{blist}
\item Velocities $\mathbf{v}_l$ and $\mathbf{v}_s$.
\item Pressure $p_l$ and $p_s$.
\item Porosity $\phi$ / Melting source $\Gamma$.
\end{blist}
}\parbox{0.5\textwidth}{\raggedright
\subhead{Three parts.}
\begin{blist}
\item Static mechanic equations.
\item Transport of chemical species and energy.
\item Phase package or prescribed melting function.
\end{blist}
}

\begin{myslide}
\heading{A Degenerate Darcy-Stokes System}
\subhead{Pressure Potentials.}
\begin{equation*}
	q_l = p_l - \rho_l g z\quad\text{and}\quad q_s = p_s - \rho_l g z,
\end{equation*}
\begin{equation*}
		  p_l - p_s = q_l - q_s = \frac{1}{1-\phi}(q_l - q), \quad q=\bar{q}.
\end{equation*}
Then
		  \begin{equation*}
\begin{alignedat}{3}
		  &\phi\mathbf{v}_r + \frac{k_0\phi^{2+2\Theta}}{\mu_l}\nabla q_l &&= 0, \qquad \vspace*{0.5cm} && \Maroon{(\text{Darcy})}\\
		  &\mu_s\nabla \cdot (\phi\mathbf{v}_r) + \frac{\phi}{1-\phi}(q_l-q) &&=0,
&&\Maroon{(\text{Conservation}}\\[-20pt]
& && &&\Maroon{\text{/Compaction})}\\[8pt]
  		  &\nabla q - \nabla \cdot \tau(\mathbf{v}_s) &&= -(1-\phi)\rho_r\mathbf{g}, \quad && \Maroon{(\text{Stokes})}\\
		  &\mu_s\nabla \cdot \mathbf{v}_s - \frac{\phi}{1-\phi}(q_l-q) &&=0. &&\Maroon{(\text{Compaction})}
\end{alignedat}
		  \end{equation*}

\end{myslide}

\begin{myslide}
\heading{Remarks.}

\subhead{Advantages.} A unified two-phase framework handles both melting and solid regions without explicit 
interfaces, allowing smooth porosity evolution.

\subhead{Problems.} When no melt, $\phi = 0$, 
\begin{nlist}
\item Darcy system degenerates when $\phi$ vanishes, lose bound on $q_l$.
\item Lose the conservation equation.
\item The condition number grows as $\phi \rightarrow 0$.
\end{nlist}

\subhead{How others address this problem?} 
\begin{nlist}
\item Introducing a cut-off for certain material properties. 
\item Introducing a threshold switch between Stokes and Darcy/Stokes equations among cells. \OliveGreen{\small(Dannberg, Gassmöller, Grove and Heister, 2019)} 
\end{nlist}
Both involves a selected threshold value.
Mass is not \Oorange{locally conserved}.

		  \subhead{What we do?} \OliveGreen{\small(Arbogast, Hesse and Taicher, 2017)}
\begin{nlist}
\item Scale the variables. Bound in scaled energy space when $\phi \rightarrow 0$.
\item Modified the weak form to achieve local mass conservation.
\end{nlist}

\end{myslide}

\begin{myslide}

\heading{A Scaled Space $H_\phi(\text{div})$}
\subhead{Scaled Variables}\newline
%Define by Arbogast and Taicher [1]:
\begin{equation*}\label{eq:scaledVariables}
	\tilde{\mathbf{v}}_r = \phi^{-\Theta}\mathbf{v}_r
	\quad\text{and}\quad
	\tilde{q}_l = \phi^{1/2}q_l,
\end{equation*}
\subhead{Assumption.} \begin{equation*}\label{eq:phiboundness}
\phi \in L^{\infty}(\Omega) \quad \text{and} \quad
		  \phi^{\Theta - 1/2} \nabla \phi \in (L^{\infty}(\Omega))^2.
\end{equation*}

\subhead{A Scaled Space $H_{\phi}(\text{div})$}
\begin{equation*}\label{eq:scaledHspace}
		  H_{\phi}(\text{div}; \Omega) = \Big\{ \mathbf{v} 
		  \in (L^2(\Omega))^2: \phi^{-1/2}\nabla \cdot (\phi^{1+\Theta} \mathbf{v})
		  \in L^2(\Omega) \Big\}.
\end{equation*}
		  equipped with the \Maroon{weighted inner-product\vspace*{-6pt}}
\begin{equation*}
(\mathbf{u},\mathbf{v})_{H_\phi(\Div)} = (\mathbf{u},\mathbf{v}) + \big(\phi^{-1/2}\div(\phi^{1+\Theta}\mathbf{u}),\phi^{-1/2}\div(\phi^{1+\Theta}\mathbf{v})\big).
\end{equation*}\vspace*{-20pt}\newline
Moreover, $H(\Div;\Omega) = H_1(\Div;\Omega) \subset H_\phi(\Div;\Omega)$.

\bigskip
The \Oblue{normal trace} on $\partial \Omega$ is well-defined by
		  \begin{equation*}\label{eq:scaledtraceoperator}
		  \langle \phi^{1/2+\Theta} \mathbf{v}\cdot \pmb{\nu}, \psi \rangle = \big(\mathbf{v}, \phi^{1+\Theta}\nabla(\phi^{-1/2}\psi)\big) + 
		  \big(\phi^{-1/2}\nabla \cdot (\phi^{1+\Theta}\mathbf{v}),\psi \big), \, \forall \psi \in H^1(\Omega)
\end{equation*}

\end{myslide}

\begin{myslide}
\heading{The Scaled Dimensionless Boundary Value Problem}
Find two velocity-pressure potential pairs \Oorange{$(\check{\mathbf{v}}_r, \check{q}_l)$} and 
\Oorange{$(\check{\mathbf{v}}_s, \check{q})$} such that 
\begin{equation*}
\begin{alignedat}{2}
%\label{eq:dimensionlessdarcy_flow_sc_bv}
		  &\check{\mathbf{v}}_r
		  + \phi^{1+\Theta}\nabla(\phi^{-1/2}\check{q}_l)
        &&= 0,\\
%\label{eq:dimensionlessconsMass_flow_sc_bv}
		  &\phi^{-1/2}\nabla \cdot(\phi^{1+\Theta}\check{\mathbf{v}}_r)
        + \frac{1}{1-\phi}(\check{q}_l - \phi^{1/2}\check{q}) &&=0,\\
%\label{eq:dimensionlessstokes_flow_sc_bv}
		  &\nabla \check{q} - \nabla \cdot \check{\tau}(\check{\mathbf{v}}_s) &&= -(1-\phi)\mathbf{n},\\
%\label{eq:dimensionlesscompaction_flow_sc_bv}
		  &\nabla \cdot \check{\mathbf{v}}_s
        - \frac{\phi^{1/2}}{1-\phi}(\check{q}_l - \phi^{1/2}\check{q}) &&=0,
\end{alignedat}
\end{equation*}
and
\begin{equation*}\label{eq:bv}
		  \begin{alignedat}{2}
					 \phi^{1+\Theta}\check{\mathbf{v}}_r\cdot \pmb{\nu} &= g_{r} \quad &&\text{on} \hspace{0.2cm} \Gamma_{u,D}, \\
					 \check{\mathbf{v}}_s &= \mathbf{g}_{s} \quad &&\text{on} \hspace{0.2cm} \Gamma_{u,S}, \\
					 \check{q}_l &= \phi^{1/2}q_{l,0} \quad &&\text{on} \hspace{0.2cm} \Gamma_{i,D}, \\
					 \pmb{\tau}\cdot \pmb{\nu} - q &= t_0  \quad &&\text{on} \hspace{0.2cm} \Gamma_{i,S}.
		  \end{alignedat}
\end{equation*}

\subhead{Dimensionless Variables.}
\begin{blist}
\item $\check{\mathbf{v}}_r = \frac{\tilde{\mathbf{v}}_r}{u_0}$,\quad $\check{\mathbf{v}}_s = \frac{\mathbf{v}_s}{u_0}$. 
\item $\check{q} = \frac{q}{p_0}$,\quad $\check{q_l} = \frac{\tilde{q}_l}{p_0}$.
\end{blist}

\end{myslide}

\begin{myslide}
\heading{The Scaled Weak Formulation}
%\subhead{Function Spaces.}
		  \vspace*{-0.4cm}
\begin{equation*}\label{eq:spaces}
		  \begin{aligned}
					 \mathbb{V}^\phi_D &= \Big\{\mathbf{v}\in H_{\phi}(\text{div};\Omega): \phi^{1/2+\Theta} \mathbf{v}\cdot \pmb{\nu} = 0 \hspace{0.15cm} \text{on}\hspace{0.15cm} \Gamma_{u,D}  \Big\},\\
					 \mathbb{U}^\phi_D &= \Big\{\mathbf{u}\in H_{\phi}(\text{div};\Omega): \phi^{1/2+\Theta} \mathbf{u}\cdot \pmb{\nu} = \phi^{-1/2}g_r \hspace{0.15cm} \text{on}\hspace{0.15cm} \Gamma_{u,D}  \Big\} = \tilde{\mathbf{g}}_r + \mathbb{V}^\phi_D,\\
					 \mathbb{W}_D      &= L^2(\Omega),\\
					 \mathbb{V}_S &= \Big\{\mathbf{v}\in (H^1(\Omega))^2 : \mathbf{v} = \mathbf{0} \hspace{0.15cm} \text{on}\hspace{0.15cm}  \Gamma_{u,S} \Big\},\\
					 \mathbb{U}_S &= \Big\{\mathbf{u}\in (H^1(\Omega))^2 : \mathbf{u} = \mathbf{g}_s \hspace{0.15cm} \text{on}\hspace{0.15cm}  \Gamma_{u,S} \Big\} 
					 = \tilde{\mathbf{g}}_s + \mathbb{V}_S,\\
		  \mathbb{W}_S &= L^2(\Omega),
		  \end{aligned}
\end{equation*}

\subhead{A Well-Posed Weak Form.}

Find $\check{\mathbf{v}}_r \in \mathbb{U}^{\phi}_D$,
$\check{q}_l \in \mathbb{W}_D$, $\check{\mathbf{v}}_s \in \mathbb{U}_S$, and $\check{q}\in \mathbb{W}_S$ such that
\begin{equation*}
\begin{alignedat}{2}
%		  \label{eq:weak_coupled_1}
		  &(\check{\pmb{\tau}}(\bold{v}_{s}),\nabla \pmb{\psi}_s) - (\check{q}, \nabla \cdot \pmb{\psi}_s) = -((1-\phi)\bold{n}, \pmb{\psi}_s) &&\forall\pmb{\psi}_s \in \mathbb{V}_S,\\
%		  \label{eq:weak_coupled_2}
					 &(\nabla \cdot \check{\bold{v}}_{s}, \omega)  + \Big(\frac{\phi^{1/2}}{1-\phi} (\check{q}_l - \phi^{1/2}\check{q}), \omega\Big) = 0 &&\forall \omega \in \mathbb{W}_S,\\
%					 \label{eq:weak_coupled_3}
					 &(\bold{\check{v}}_{r},\pmb{\psi}_r)  - (\check{q}_{l},\phi^{-1/2} \nabla \cdot (\phi^{1+\Theta}\pmb{\psi}_r)) = 0, &&\forall \pmb{\psi}_r \in \mathbb{V}_D^\phi,\\
					 \nonumber
					 &(\phi^{-1/2} \nabla \cdot (\phi^{1+\Theta}\check{\bold{v}}_r),\omega_l) + \Big(\frac{1}{1-\phi}(\check{q}_{l} - \phi^{1/2}\check{q}),\omega_l \Big) = 0 \,&&\forall \omega_f \in 
		  \mathbb{W}_D.
\end{alignedat}
\end{equation*}

\subhead{Avoid Dividing by Zero when $\phi \rightarrow 0$}
\newline
		  On each element $E$, use the divergence theorem,
\begin{equation*}
\begin{aligned}
		  \int_E\phi_E^{-1/2}\div(\phi^{1+\Theta}\bpsi_{r,h})\,\omega_{l,h}\,dx
		  &= \int_{\d E}\phi_{E}^{-1/2}\phi^{1+\Theta}\bpsi_{r,h}\cdot\nu\,\omega_{l,h}\,ds\\
\nonumber
		  &\quad-\int_E\phi_{E}^{-1/2}\phi^{1+\Theta}\bpsi_{r,h}\cdot\grad\omega_{l,h}\,dx.
\end{aligned}
\end{equation*}
		  The second integral vanishs since \Maroon{$\omega_{l,h}$} is constant on $Q$. We avoid problem of dividing by zero of the term $\phi^{-1/2}$ 
		  when $\phi \rightarrow 0$.

\heading{Transport of Chemical Species}

\subhead{An advection-diffusion model}
\begin{equation*} \label{eq:specietransport}
		  \frac{\partial \overline{\rho c_{i}}}{\partial t} + \nabla \cdot \big( \overline{\rho c_{i} \mathbf{v}} - \phi_l \rho_l \mathcal{D}_{i,l} \nabla c_{i,l} \big) = 0,
\end{equation*}

\vspace{20pt}

\Oorange{Phase averaged} quantities:
\begin{equation*}
\overline{\rho c_i} = \sum_{j\in \{s,l\}} \phi_j \rho_j c_{i,j}, \quad 
\overline{\rho c_i \mathbf{v}} = \sum_{j\in \{s,l\}} \phi_j \rho_j c_{i,j} \mathbf{v}_j.
\end{equation*}

\subhead{Separate phase averaged concentration}
\begin{equation*}
\label{eq:effectivevel}
		  \overline{c_{i}\bold{v}} = \phi_s \bold{v}_s c_{i,s} + \phi_l \bold{v}_l c_{i,l} = 
		  \bigg( \frac{\phi \bold{v}_l + D(1-\phi) \bold{v}_s}{\phi + (1-\phi) D} \bigg) \overline{c}_i.
\end{equation*}
The partition coefficient \Oorange{$D = c_{i,s}/c_{i,l}$}. 

\subhead{Effective Velocity}
\begin{equation*}
\mathbf{v}_{\text{eff}} = \frac{\phi \bold{v}_l + D(1-\phi) \bold{v}_s}{\phi + (1-\phi) D} 
\end{equation*}

\subhead{Specific internal energy formulation}:
\begin{equation*}\label{eq:energybalanceinternalenergy}
\frac{\partial \overline{\rho u}}{\partial t} + \nabla \cdot \overline{\rho u \mathbf{v}} = 
		  \overline{\rho \mathcal{Q}} + \nabla \cdot \overline{K} \nabla T + \nabla \cdot \overline{\pmb{\sigma} \cdot \mathbf{v}} +
		  \overline{\rho \mathbf{v}} \cdot \mathbf{g},

\begin{equation*}\label{eq:normalizedenthalpy}
		  \frac{\partial \overline{H}}{\partial t} + \nabla \cdot (\overline{\mathbf{v}} T) =

\end{equation*}

\begin{equation*}\label{eq:specificenthalpy}

\begin{myslide}
		  \heading{Two New Alternatives of Smoothness Indicator}
\begin{itemize}
		  \item \textit{Simple smoothness indicator}\OliveGreen{\small(Huang, Arbogast and Tian, 2023)}
%~\cite{WENOAOSimplifiedSmoothness}
\begin{equation*}\label{eq:simpsmoothnessIC}
\sigma_{\text{simp}} = \frac{1}{N_\mathcal{S} -1 } \sum_k (\overline{u}_k - \overline{u}_0)^2,
\end{equation*}
where $\overline{u}_0$ is the cell averaged solution in the target cell.
This simple smoothness indicator is designed to reduce the computational cost and to deal with
distorted stencils on unstructured meshes. However, this simple smoothness indicator is not as robust
as the classic Jiang-Shu smoothness indicator.

\vspace{0.5cm}

\item \textit{Polynomial smoothness indicator}\OliveGreen{\small(Arbogast, Huang, Tian and Gray, 2025)}
		  %~\cite{sigmaP}
		  \begin{equation*}\label{eq:polysigma}
					 \sigma_{P} = \sum_{\alpha} \zeta_{\alpha, \alpha} c_{\alpha}^2
		  \end{equation*}
		  which is obtained by dropping all the cross terms appeared in polynomial multiplication. %appeared in the formulation~\eqref{eq:cofformulation}.
		  This alternative serves as a good approximation to the classic Jiang-Shu smoothness indicator while the 
		  computing efficiency is greatly improved.
\end{itemize}

\heading{Time Stepping}
\subhead{A Multi-stage Explicit SSP Runge-Kutta method.}
\begin{equation*}\label{eq:multistageSSP}
\begin{aligned}
		  u^{0} &= u^n ,\\
		  u^{(i)}   &= \sum_{j=0}^{i-1} \alpha_{i,j}u^{(j)} + \Delta t \beta_{i,j} \tilde{\mathbf{F}}(u^{(j)}), \quad 1\leq i \leq m\\
		  u^{n+1} &= u^m,
\end{aligned}
\end{equation*}
where $\sum_{j}^{i-1} \alpha_{i,j} = 1$ for consistency. 

\vspace{0.5cm}

\subhead{A Second-order Two-stage Scheme.}
\begin{equation*}\label{eq:2ndrk}
\begin{aligned}
		  u^{0}   &= u^n,\\
		  u^{(1)} &= u^{(0)} + \Delta t \tilde{\mathbf{F}}(u^{(0)}),\\
		  u^{n+1} &= u^{(0)} + \frac{1}{2} \Delta t \tilde{\mathbf{F}}(u^{(0)}) + \frac{1}{2} \Delta t \tilde{\mathbf{F}}(u^{(1)}),
\end{aligned}
\end{equation*}

where $\text{line}_{1-4}$ is defined as $H(C) = \check{T}_e + \check{L} C/C_e$.
We can determine the phase fractions and the current temperature as
\begin{equation*}
		  \phi_l = \frac{H - \check{T}_e}{\check{L}}, \quad
		  \phi_2 = C - \phi_l C_e, \quad
		  \phi_1 = 1- \phi_l - \phi_2 \quad
		  \text{and} \quad \check{T} = \check{T}_e.
\end{equation*}

%%\begin{equation*}
%\frac{\partial\check{T}}{\partial H} = 0 \quad \text{and} \quad \frac{\partial\check{T}}{\partial C} = 0.
%\end{equation*}

\end{myslide}

\begin{myslide}
\heading{Phase Split Calculation Cont'd}
Region IV. \Oblue{Super-eutectic two phase region: All solid phase opx has
been exhausted.} 
The conditions to be in the region IV are
\begin{equation*}
		  C \leq C_e, \quad
		  \text{line}_{1-4} < H \leq \text{line}_{2-4},
\end{equation*}
where $\text{line}_{2-4}$ is defined as $H(C) = H_m - c_l/C_e$,
and we can determine phase fractions as 
\begin{equation*}
    \phi_2 = 0, \quad \phi_l = \frac{C}{c_l(\check{T})} \quad \text{and} \quad \phi_1 = 1- \phi_l.
\end{equation*}
%The corresponding temperature can be determined by solving the following equation
%\begin{equation}
%F(\check{T}) = H - \check{T} - 
%		  \frac{C}{c_l(\check{T})} \check{L} = 0. 
%\end{equation}
Let $c_l = C_e (\check{T}_m - \check{T})$,   
we can obtain a quadratic equation about temperature as
\begin{equation*}
		  \check{T}^2 - (H + \check{T}_m)\check{T} + 
		  \check{T}_m H - C L/C_e = 0,
\end{equation*}
which can be solved with standard quadratic root finding formula.
We omit the root that exceeding melting temperature, so
\begin{equation*}
		  \begin{aligned}
		  \check{T} &= \frac{H + \check{T}_m - 
		  \sqrt{ (H + \check{T}_m)^2 - 
		  4(\check{T}_m H - C \check{L}/C_e)} }{2} \\
					  &= 
		  \frac{H + \check{T}_m - 
		  \sqrt{ (H - \check{T}_m)^2 + 
		  4C \check{L}/C_e} }{2}. 
		  \end{aligned}
\end{equation*}

\end{myslide}

\begin{myslide}
\heading{Continuous Velocities}

The bulk mass composition represents a relatively smooth advected quantity.
A harmonic average would add artificial damping and suppress mixing, while an arithmetic average
is more suitable without over-penalizing discontinuities.

\begin{equation*}
c^e = (c^+ + c^-)/2
\end{equation*}

\subhead{Effective Velocity}

\begin{equation*}
\bold{v}_{\text{eff}} = \frac{c^e_l \overline{\phi} \bold{v}_l + 
c^e_s(1-\overline{\phi}) \bold{v}_s}{c^e_l\overline{\phi} + c^e_s (1-\overline{\phi})}
\end{equation*}

\subhead{Phase Averaged Velocity}

\begin{equation*}
\check{\overline{\mathbf{v}}} = \overline{\phi}\mathbf{v}_l  + (1-\overline{\phi})\mathbf{v}_s
\end{equation*}

\end{myslide}
