\section{Direct Serendipity Space}
Serendipity finite elements have the same accuracy of approximation with 
classical tensor product elements while using fewer degrees of freedom.
However, the classical serendipity spaces suffer from poor approximation
on quadrilaterals. 
To address this issue, we follow the approach of~\cite{directSerendipity}
by defining local shape functions directly on quadrilateral elements, thereby recovering the desired accuracy.
In the next sections, $H^1$ and $H(\text{div})$ conforming direct mixed finite element spaces
are constructed with the direct serendipity function spaces.

To begin with, we introduce some auxiliary distance functions. 
We define the linear polynomial $\lambda_i(\mathbf{x})$ giving the
distance of a point at coordinate $\mathbf{x}\in \mathbb{R}^2$ to edge $e_i$ as
\begin{equation}\label{eq:distancelambda}
		  \lambda_i(\mathbf{x}) = -(\mathbf{x}-\mathbf{x}^*_i)\cdot \pmb{\nu}_i, \quad i = 0,1,2,3,
\end{equation}
where $\mathbf{x}^*_i \in e_i$  is any point on edge $e_i$.
We pick $\mathbf{x}^*_i = \mathbf{x}_{v,i}$ for simplicity, where $\mathbf{x}_{v,i}$ denotes
the coordinates of corner $v_i$.
The
distance function $\lambda_i$ returns a positive value as long as the point is located in the
interior of the element $E$. 

We define two additional distance functions giving 
the distance of a point at coordinate $\mathbf{x} \in \mathbb{R}^2$ to the diagonal line $d_{i}$ 
joining $v_i$ with $v_{i+2}$, $i\in{0,1}$.
Let $\pmb{\nu}_{d,i}$ be the unit normal vector to the diagonal line 
$d_{i}$. The direction of the $\mathbf{\nu}_{d,i}$ is defined by 
rotating counterclockwise of the unit vector pointing from $v_i$ to $v_{i+2}$.
The resulting diagonal distance function is defined as
\begin{equation}\label{eq:distancelambdad}
		  \lambda_{d,i}(\mathbf{x}) = -(\mathbf{x} - \mathbf{x}^*_{d,i}) \cdot \pmb{\nu}_{d,i}, 
		  \quad i=0,1,
\end{equation}
where $\mathbf{x}^*_{d,i}$ is any point on the diagonal line $d_{i}$. Here,
we use the corner point $v_i$ for simplicity, i.e.,
$\mathbf{x}^*_{d,i} = \mathbf{x}_{v,i}$.

We now present a set of nodal basis functions for the first and second
order direct serendipity spaces, $\mathcal{DS}_1$ and $\mathcal{DS}_2$.
%defined by ~\cite{directSerendipity}.

\subsection{Second-order Space $\mathcal{DS}_2$}
The dimension of the second-order polynomials in two dimension is calculated by~\eqref{eq:dimpolynomial} as
\begin{equation}
		  \text{dim}\mathbb{P}_2(\mathbb{R}^2) = \begin{pmatrix}
2+2 \\
2
		  \end{pmatrix} = \frac{(4)!}{2!2!} = 6,
\end{equation}
while a quadrilateral element requires a minimum of $4$ vertex dofs and $4$ edge 
dofs. Therefore, we supplement polynomial space $\mathbb{P}_2(E)$ 
with two linearly independent
functions, $\phi_{s,1}(\mathbf{x})$ and $\phi_{s,2}(\mathbf{x})$, 
spanning the supplemental function space $\mathbb{S}_r^{\mathcal{DS}}(E) = 
\text{span}\{\phi_{s,1},\phi_{s,2}\}, \hspace{0.2cm} r\geq 2$. We present the 
second-order direct serendipity space as
\begin{equation}\label{eq:definitionds2}
		  \mathcal{DS}_2(E) = \mathbb{P}_2(E) \oplus \mathbb{S}_2^{\mathcal{DS}}(E).
\end{equation}
The supplemental functions are defined as
\begin{equation}\label{eq:defsuppfunc}
		  \phi_{s,i}(\mathbf{x}) = \lambda_{i+1}(\mathbf{x}) \lambda_{i+2}(\mathbf{x}) R_{i,i+2}(\mathbf{x}), \quad i\in\{0,1\}.
\end{equation}
We use a simple definition for rational
function $R_{i,i+2}$ as
\begin{equation}\label{eq:defR}
		  R_{i,i+2}(\mathbf{x}) = \frac{\lambda_{i}(\mathbf{x}) - \lambda_{i+2}(\mathbf{x})}{\lambda_i(\mathbf{x}) + \lambda_{i+2}(\mathbf{x})},
\end{equation}
which satisfies the conditions 
\begin{equation}\label{eq:Rproperty}
		  R_{i,i+2}|_{e_i} = -1, \quad R_{i,i+2}|_{e_{i+2}} = 1.
\end{equation}
We continue to define
\begin{equation}\label{eq:def}
		  %R_i(\mathbf{x}) = \frac{1}{2} \Big(1 - (-1)^{\floor*{i/2}} R_{i, i+2}(\mathbf{x})\Big) 
		  R_i(\mathbf{x}) = \frac{1}{2} \Big(1 -  R_{i, i+2}(\mathbf{x})\Big) 
		  = \frac{\lambda_{i+2}(\mathbf{x})}{\lambda_i(\mathbf{x}) + \lambda_{i+2}(\mathbf{x})},\quad i \in \{0,1,2,3\},
\end{equation}
where $i \equiv i \Mod{4}$ as usual, and $R_{i,i+2} = -R_{i+2,i}$ according to 
the equation~\eqref{eq:defR}. 
Here, $R_i|_{e_i} = 1$, and
$R_i|_{e_{i+2}} = 0$ by definition. In Figure~\ref{fig:R0}, we
draw $R_0$ on a general quadrilateral element, which evaluates to $1$ 
on edge $e_0$, and evaluates to $0$ on the opposite edge $e_2$.
\begin{figure}[ht]

\includegraphics[width=5.5cm,height=4cm]{chapters/pics/R0_2.png}
		  \caption{Rational function $R_0$.}
		  \label{fig:R0}
\end{figure}

The edge nodal basis functions for $\mathcal{DS}_2$ are constructed as
\begin{equation}\label{eq:edgebasisds2}
		  \varphi^{(2)}_{e,i}(\mathbf{x}) = \frac{\lambda_{i+1}(\mathbf{x})\lambda_{i+3}(\mathbf{x})R_i(\mathbf{x})}
		  {\lambda_{i+1}(\mathbf{x}_{e,i})\lambda_{i+3}(\mathbf{x}_{e,i})R_i(\mathbf{x}_{e,i})},
\end{equation}
where $\mathbf{x}_{e,i}$ are the coordinates of the middle point of the 
edge $e_i$,
again $i \equiv i \Mod{4}$.
We use superscript $(2)$ to denote the order of the function space.
We can restrict the edge basis function to edges as
\begin{equation}\label{eq:edgerestriction}
		  \varphi^{(2)}_{e,i}|_{e,j} = \left \{
					 \begin{aligned}
								&\frac{\lambda_{i+1}(\mathbf{x})\lambda_{i+3}(\mathbf{x})}
								{\lambda_{i+1}(\mathbf{x}_{e,i})\lambda_{i+3}(\mathbf{x}_{e,i})}, \hspace{0.2cm} i=j, \\
								& 0, \hspace{3.4cm} i\neq j.
					 \end{aligned}
		  \right .
\end{equation}
In Figure~\ref{fig:phie0}, we present
edge basis function $\varphi^{(2)}_{e,0}$ on a general quadrilateral element, 
where 
$\varphi^{(2)}_{e_0}$ evaluates $0$ on edges $\{e_i, \quad i\neq 0\}$, and 
evaluates to a quadratic function on edge $e_0$.

\includegraphics[width=5.5cm,height=4cm]{chapters/pics/phie0_2.png}
		  \caption{Edge nodal basis function $\varphi^{(2)}_{e,0}$ in $\mathcal{DS}_2$.}
		  \label{fig:phie0}
\end{figure}

The vertex nodal basis functions for $\mathcal{DS}_2$ are constructed as
\begin{equation}\label{eq:vertexbasisds2}
		  \varphi^{(2)}_{v,i}(\mathbf{x}) = \frac{\lambda_{i+2}(\mathbf{x}) \lambda_{i+3}(\mathbf{x}) - 
		  \sum_{k=i}^{i+1}\lambda_{i+2}(\mathbf{x}_{e,i})\lambda_{i+3}(\mathbf{x}_{e,i})\varphi^2_{e,i}(\mathbf{x})}
		  {\lambda_{i+2}(\mathbf{x}_{v,i}) \lambda_{i+3}(\mathbf{x}_{v,i}) - 
		  \sum_{k=i}^{i+1}\lambda_{i+2}(\mathbf{x}_{e,i})\lambda_{i+3}(\mathbf{x}_{e,i})\varphi^{(2)}_{e,i}(\mathbf{x}_{v,i})},
\end{equation}
where $\mathbf{x}_{v,i}$ is the coordinates of the vertex node $v_i$, and $i \equiv i \Mod{4}$ as well. At the nodal points
\begin{equation}
		  \varphi^{(2)}_{v,i}(\mathbf{x}) = \left \{ 
		  \begin{alignedat}{2}
					 &1, \quad && \mathbf{x} = \mathbf{x}_{v,i}, \\
					 &0, \quad && \mathbf{x} = \mathbf{x}_{v,j} \text{ or } \mathbf{x}_{e,k}, \quad j\neq i,\, \forall k.
        \end{alignedat}
        \right.
\end{equation}
In Figure~\ref{fig:phiv0}, we 
show $\varphi^{(2)}_{v,0}$ on a general quadrilateral.

\end{figure}

We conclude that unisolvence follows directly from the constructed nodal basis functions.

\subsection{First-order Space $\mathcal{DS}_1$}
The dimension of the first-order polynomials in two dimension is also 
calculated by equation~\eqref{eq:dimpolynomial} as
\begin{equation}
		  \text{dim}\mathbb{P}_1(\mathbb{R}^2) = \begin{pmatrix}
1+2 \\
2
		  \end{pmatrix} = \frac{(3)!}{2!1!} = 3,
\end{equation}
while a quadrilateral element requires a minimum of $4$ vertex dofs. 
Therefore, we supplement polynomial space $\mathbb{P}_1(E)$ with 
one additional function.
The supplemental function for $\mathcal{DS}_1$ differs from 
that for $\mathcal{DS}_2$. We present the first order direct serendipity space
as
\begin{equation}\label{eq:definitionds1}
		  \mathcal{DS}_1(E) = \mathbb{P}_1(E) \oplus \text{span}\{R\},
\end{equation}
where $R(\mathbf{x}_{v,0}) = R(\mathbf{x}_{v,2}) = 
-R(\mathbf{x}_{v,1}) = -R(\mathbf{x}_{v,3}) = 1$. The construction 
of the rational function $R$ is not unique. We present one way to construct
$R(\mathbf{x})$ using edge nodal basis function~\eqref{eq:edgebasisds2} 
in $\mathcal{DS}_2$ as
\begin{equation}\label{eq:rationalfunctionds1}
		  R(\mathbf{x}) = r(\mathbf{x}) - \sum_{i=0}^3r(\mathbf{x}_{e,i}) \varphi^{(2)}_{e,i}(\mathbf{x}),
\end{equation}
where the auxiliary function $r(\mathbf{x})$ is defined as
\begin{equation}\label{eq:auxiliaryrationalfunction}
r(\mathbf{x}) = \frac{\lambda_2(\mathbf{x})\lambda_3(\mathbf{x})}{\lambda_2(\mathbf{x}_{v,0})\lambda_3(\mathbf{x}_{v,0})} - 
 \frac{\lambda_0(\mathbf{x})\lambda_3(\mathbf{x})}{\lambda_0(\mathbf{x}_{v,1})\lambda_3(\mathbf{x}_{v,1})} + 
 \frac{\lambda_0(\mathbf{x})\lambda_1(\mathbf{x})}{\lambda_0(\mathbf{x}_{v,2})\lambda_1(\mathbf{x}_{v,2})} - 
 \frac{\lambda_1(\mathbf{x})\lambda_2(\mathbf{x})}{\lambda_1(\mathbf{x}_{v,3})\lambda_2(\mathbf{x}_{v,3})} .
\end{equation}
In Figure~\ref{fig:rationalds1}, we plot the rational function $R(\mathbf{x})$ on a general quadrilateral.
\begin{figure}[ht]

\includegraphics[width=5.5cm,height=4cm]{chapters/pics/R_1.png}
\includegraphics[width=5.5cm,height=4cm]{chapters/pics/R_2.png}
		  \caption{Supplemental rational function $R$ in $\mathcal{DS}_1$.}
		  \label{fig:rationalds1}
\end{figure}

We then define vertex nodal basis function accordingly as
\begin{equation}\label{eq:vertexbasisds1}
		  \varphi^{(1)}_{v,i}(\mathbf{x}) = \frac{\lambda_{d,m}(\mathbf{x}) - \frac{1}{2}\lambda_{d,m}(\mathbf{x}_{v,i+2})(1+ (-1)^i R(\mathbf{x}))}
{\lambda_{d,m}(\mathbf{x}_{v,i}) - \lambda_{d,m}(\mathbf{x}_{v,i+2})},
\end{equation}
where $m\equiv i+1 \Mod{2}$, and $i\equiv i \Mod{4}$.
In Figure~\ref{fig:vertexbasisds1}, we show the vertex nodal basis function $\varphi^{(1)}_{v,0}$ in $\mathcal{DS}_1$ on a 
general quadrilateral, which evaluates $1$ at vertex $v_0$ and $0$ at other vertices.

\subsection{An Interpolation Operator}
For completeness, 
we describe an interpolation operator mapping
onto $\mathcal{DS}_r(K_i)$, $r=1,2,$ following~\cite{directSerendipity}. 
For each node $a_i$, the associated $K_i$ has different meanings
depending on the type of the nodal basis functions. If the basis function is associated
with an edge node, we set $K_i = e_i$, where $e_i$ is the corresponding edge. 
For the basis function associated with a
vertex node, we set $K_i$ to be any edge $e$ containing node $a_i$,
with the restriction that if $a_i \in \partial \Omega$, then $e \in \partial \Omega$.

An interpolation
operator $\mathcal{I}^r_h:W_p^l(\Omega) \rightarrow \mathcal{DS}_r$ is defined as
\begin{equation}\label{eq:interpolation}
					 \mathcal{I}^r_h v(\mathbf{x}) = \sum_{i=1}^{N_r} \varphi_i(\mathbf{x})\int_{K_i} \psi_i(\mathbf{y}) v(\mathbf{y})\, d\mathbf{y} \in \mathcal{DS}_r,
\end{equation}
where $1\leq p \leq \infty$ and $l > 1/p$ (but $l\geq 1$ if $p=1$),
and $\psi(\mathbf{x})$ is the dual nodal basis such that
\begin{equation}
		  \int_{K_i} \psi_{i}(\mathbf{x}) \varphi_j(\mathbf{x})\, d\mathbf{x} = \delta_{i,j}, \quad i,j \in \{0,1,2,\dots, N_r \},
\end{equation}
where $N_r$ is the number of nodal basis functions.
We present the following two lemmas regarding the constructed interpolation
operator.

\begin{lem}\label{lem:interpolationop}
		  (Boundedness of the interpolation operator.)

		  Let $\mathcal{T}_h$ be uniformly shape regular according to Definition~$\ref{eq:regularityineq}$
		  with shape regularity parameter $\sigma_*$.
		  %For each element $E \in \mathcal{T}_h$, 
		  %There exists a bounded interpolation operator~\eqref{eq:interpolation}.
		  Let $v \in W^l_p(\Omega)$, with $1\leq p \leq \infty$ and $l >q/p$. %Suppose that the basis functions of $\mathcal{DS}_r(E)$ are constructed using
		  %$\lambda_{24}$ and $\lambda_{13}$ such that the zero set $\mathcal{Z}_{i,i+2}$ intersects $e_{i+3}$ and $e_{i+1}$.
		  Moreover, assume that $R_{i,i+2}$ is a uniformly differentiable functions of the vertices of $E$ up to order m.
		  Then for $r= 1,2$, $E \in \mathcal{T}_h$, 
		  $1\leq q \leq \infty$, and any non negative integer m,
		  \begin{equation}\label{eq:interpolationopbound}
		 \norm{\mathcal{I}_h^r v}_{W_q^m(E)} \leq C(\sigma_*, m, q) \sum_{k=0}^l
					 h_E^{k-m+\frac{2}{q}-\frac{2}{p}} \lvert v\rvert_{W^k_p(E^*)},
		  \end{equation}
		  where $E^* = \cup_{F \in \mathcal{T}_h, F \cap E \neq \emptyset} F$ and
		  $\lvert\hspace{0.1cm} \cdot\hspace{0.1cm} \rvert_{W_p^k}$ is the seminorm of k-th order derivatives.
\end{lem}
\begin{proof}
		  See~\cite{directSerendipity}.
\end{proof}

Combining Lemma~\ref{lem:interpolationop} and the Bramble-Hilbert lemma~\ref{lem:BHlemma}, we 
obtain local and global error estimation.
\begin{lem}\label{lem:approximation}
		  (Approximability of the interpolation operator.)

		  Under the assumption of Lemma~$\ref{lem:interpolationop}$, there exists a 
		  constant $C = C(r, \sigma_*) > 0$ such that for all functions 
		  $v \in W_{l,p}(E^*)$, with $1\leq p \leq \infty$ and $l>1/p$ or 
		  ($l\geq 1$ if $p=1$),
		  \begin{equation}\label{eq:brambleHilbertDS}
					 \norm{v-\mathcal{I}_h^r v}_{W^m_p(E)} \leq C h_E^{l-m}
					 \lvert v \rvert_{W^l_p(E^*)}, \quad 0 \leq m \leq \min(l,r+1).
		  \end{equation}
		  Moreover, there exists a constant $C = C(r,\sigma_*) > 0$, independent
		  of $h = \max_{E \in \mathcal{T}} h_E$, such that for all functions 
		  $v \in W_p^l(\Omega)$, $p<\infty$
		  \begin{equation}
					 \bigg(\sum_{E\in\mathcal{T}_h} \norm{v - \mathcal{I}_h^r v}^p_{W_p^m(E)}\bigg)^{1/p} \leq C h^{l-m} \lvert v \rvert_{W_p^l(\Omega)}, \quad 0 \leq m \leq \min(l, r+1).
		  \end{equation}
\end{lem}
\begin{proof}
		  See~\cite{directSerendipity}.
\end{proof}

\section{Mixed Finite Elements}
In this section, we describe our mixed finite element approximation for the scaled coupled system~\eqref{eq:dimensionlessdarcy_flow_sc} 
to~\eqref{eq:dimensionlesscompaction_flow_sc}. 
We begin by presenting the mixed variational formulations of the Stokes and Darcy model problems separately. 
Subsequently, we introduce $H^1$- and $H(\text{div})$-conforming mixed finite element spaces.
We describe the construction of mixed finite elements from the aforementioned direct serendipity space.
Recall Ciarlet's definition~\cite{doi:10.1137/1.9780898719208} 
of a finite element.
\begin{defn}\label{def:ciarletFEM}
		  A triple $(E, X(E), \{\phi_1,\ldots,\phi_n\})$ defines a \textit{finite element}, where
		  \begin{itemize}
					 \item The element $E \in \mathcal{T}_h$;
					 \item The space of shape functions is $X(E)$ with
								basis $\{\varphi_1,\ldots,\varphi_n\}$,where
								the dimension of
								the shape function space is $\text{dim}(X(E)) = n$;
					 \item A set of continuous and linear functionals is $\{\phi_1,\ldots,\phi_n \}$, defined on a subset $\mathcal{X}(E)$
								of the energy space containing the finite element space $X(E)$ as
								\begin{equation}
										  \phi_j : \mathcal{X}(E) \rightarrow \mathbb{R},
								\end{equation}
								such that the unisolvence condition is achieved as
								\begin{equation}
										  \langle \phi_j, \varphi_i \rangle = \delta_{ij}, \quad i,j \in \{1,2,3,\dots, n \}.
								\end{equation}
		  \end{itemize}
\end{defn}
In later subsections, we give an explicit construction of the nodal basis functions and degrees of freedom (values of the $\phi_j,\hspace{0.2cm}j=1,2,\ldots,n$) that is 
associated to them, thereby completing the definition of the finite element.

\subsection{A Darcy Model Problem}
Consider a model problem for Darcy flow equipped with homogeneous boundary condition as
\begin{equation}
\begin{aligned}
\bold{u} + \nabla p & = \bold{f}, \hspace{0.2cm} \text{in} \hspace{0.2cm} \Omega,\\
\nabla \cdot \bold{u} & = 0, \hspace{0.25cm} \text{in} \hspace{0.2cm} \Omega, \\
		  \mathbf{u}\cdot\pmb{\nu} &= 0, \hspace{0.2cm} \text{on} \hspace{0.2cm} \partial \Omega.
\end{aligned}
\end{equation}
We seek the displacement and pressure solution pair $(\mathbf{u},p)$ in the 
identical test and trial spaces 
\begin{equation}\label{eq:HdivspaceVariation}
		  \begin{aligned}
					 \mathbb{V}_D &= \{\mathbf{v} \in (H(\text{div};\Omega))^2 : \mathbf{v}\cdot\pmb{\nu} = \mathbf{0} \hspace{0.3cm} \text{on} \hspace{0.2cm} \partial \Omega\}, \\
					 \mathbb{U}_D &= \{\mathbf{u} \in (H(\text{div};\Omega))^2 : \mathbf{u}\cdot\pmb{\nu} = \mathbf{0} 
					 \hspace{0.3cm} \text{on} \hspace{0.2cm} \partial \Omega\} = \mathbb{V}_D, \\
					 \mathbb{W}_D &= \bigg\{p \in L^2(\Omega) : \int_\Omega p\, d\mathbf{x} = 0\bigg\} = L^2(\Omega)/\mathbb{R}.
		  \end{aligned}
\end{equation}
The standard mixed weak form reads as follows: 

Find $\mathbf{u} \in \mathbb{U}_D$  
and $p \in \mathbb{W}_D$ such that
\begin{equation}\label{eq:hdivweakmodel}
		  \begin{aligned}
					 (\mathbf{u}, \mathbf{v}) - (p, \nabla \cdot \mathbf{v}) &= (\mathbf{f},\mathbf{v}), \hspace{0.2cm} \forall \mathbf{v} \in \mathbb{V}_D,\\
					 (\nabla \cdot \mathbf{u}, \omega) &= 0, \hspace{1cm} \forall \omega \in \mathbb{W}_D.
		  \end{aligned}
\end{equation}
We introduce two finite-dimensional subspaces $\mathbb{V}_{D,h} \subset \mathbb{V}_{D}$ and $\mathbb{W}_{D,h} \subset \mathbb{W}_{D}$, and consider
the discretized problem: 

Find $(\mathbf{u}_h,p_h) \in \mathbb{V}_{D,h} \times \mathbb{W}_{D,h}$ such that
\begin{equation}\label{eq:hdivwekamodel_dis}
		  \begin{aligned}
					 (\mathbf{u}_h, \mathbf{v}_h) - (p_h, \nabla \cdot \mathbf{v}_h) &= (\mathbf{f},\mathbf{v}_h), \hspace{0.2cm} \forall \mathbf{v}_h \in \mathbb{V}_{D,h},\\
					 (\nabla \cdot \mathbf{u}_h, \omega_h) &= 0, \hspace{1.15cm} \forall \omega_h \in \mathbb{W}_{D,h}.
		  \end{aligned}
\end{equation}
In the next subsection, we present a construction of the discretized space pair $\mathbb{V}_{D,h} \times \mathbb{W}_{D,h}$ for this Darcy model problem.

\subsection{A reduced $H(\text{div})$ conforming space $\mathbb{V}_1^{0}$}
We have some choice to discretize the space $\mathbb{V}_D$.
For example, we can choose the classical Ravier-Thomas element 
~\cite{10.1007/BFb0064470}. The serendipity space is
the natural precursor of the Brezzi-Douglas-Marini element~\cite{BDM} related 
by a de Rham complex.
We generate a reduced first-order $H(\text{div})$ approximation from 
the direct serendipity space, where $\nabla \cdot \mathbf{u}$
is approximated one less order than approximation of $\mathbf{u}$.
We can also recover the Arbogast-Correa element~\cite{doi:10.1137/15M1013705} if we 
use the mapped serendipity space, as defined in that paper.

We write the de Rham exact sequence for the related energy spaces as
\begin{equation}\label{eq:exactsequence}
\begin{alignedat}{5}
&\mathbb{R} \xhookrightarrow[]{\text{i}}\, &&H^1 \xrightarrow[]{\text{curl}} H&&(\text{div}) \xrightarrow[]{\text{div}} &&L^2 && \xrightarrow[]{0} 0\\
&\quad &&\cup                            &&\cup                                     &&\cup &&\quad\\
&\quad &&\mathcal{DS}_2  \xrightarrow[]{\text{curl}} &&\mathbf{V}_1^{0} \xrightarrow[]{\text{div}} &&\mathbb{P}_{0} && \quad.
\end{alignedat}
\end{equation}
We define the reduced first-order $H(\text{div})$ conforming space as the first-order polynomial
space supplemented with the curl of the
second-order direct serendipity supplemental space of $\mathcal{DS}_2$, i.e.,
\begin{equation}\label{eq:defhdiv}
		  \mathbb{V}_1^0(E) = (\mathbb{P}_1(E))^2 \oplus \text{curl}\,\mathbb{S}_2^{\mathcal{DS}}(E).
\end{equation}
Here, $\mathbb{V}_1^0(E)$ can be given a global basis with the normal
components of the shape functions merged continuously on the shared edge of
two elements.

The dimension of this reduced first-order $H(\text{div})$ conforming space is
\begin{equation}\label{eq:hdivdim}
		  \text{dim}(\mathbb{V}_1^0(E)) = 2\times 3 + 2 = 8.
\end{equation}
On each edge of a quadrilateral element $E$, two independent basis functions are defined: 
one with vanishing average normal flux across the edge, and the other with non-vanishing average flux.

We state the degrees of freedom as normal fluxes on each edge as
\begin{equation}
%		  \int_{e_i} \mathbf{u}\cdot \pmb{\nu} \varphi_i d\mathbf{x}, \quad \varphi_i \in \mathbb{V}_1^0(E).
		  DOF_{e_i}(\lambda_1) = \langle \mathbf{u}_h \cdot \pmb{\nu}_i, \lambda_1 \rangle_{e_i}, \quad \lambda_1 \in \mathbb{P}_1,
\end{equation}
where $\lambda_1$ is a linear function restricted to edge $e_i$, and 
$\text{dim}(\mathbb{P}_1) = 2$.
We then present a construction of two independent shape functions on each edge:
$\pmb{\varphi}_{l,i}$ and $\pmb{\varphi}_{c,i}$, such that
\begin{equation}\label{eq:constbasishdiv_show}
\pmb{\varphi}_{c,i}\cdot \pmb{\nu}_j \rvert_{e_j} = \left \{
		 \begin{aligned}
					&0, \hspace{0.8cm} j \neq i, \\
					&1, \hspace{0.8cm} j=i.
		 \end{aligned}
		 \right.
\end{equation}
\begin{equation}\label{eq:linearbasishdiv_show}
		  \pmb{\varphi}_{l,i}\cdot \pmb{\nu}_j \rvert_{e_j} = \left \{
					\begin{aligned}
							  &0, \hspace{0.8cm} j \neq i, \\
							  &l_i(\mathbf{x}),\hspace{0.2cm} j=i,
					\end{aligned}
					\right.
\end{equation}
where $l_i$ is a non vanishing linear polynomial defined on edge $e_i$ 
orthogonal to one, i.e., $\int_{e_i}l_i\, ds = 0$.

To begin with, we take the curl of the auxiliary rational function $R_i$~\eqref{eq:defR} as
\begin{equation}\label{eq:curlauxiliary}
		 \text{curl}\, R_i(\mathbf{x}) = 
					 \frac{\pmb{\tau}_{i+2}(\mathbf{x}) \lambda_i(\mathbf{x}) - \pmb{\tau}_{i}(\mathbf{x}) \lambda_{i+2}(\mathbf{x})}{(\lambda_i(\mathbf{x}) + \lambda_{i+2}(\mathbf{x}))^2},
\end{equation}
where $i \equiv i \Mod{4}$. 
We present one way to construct the ``linear'' shape 
function $\pmb{\varphi}_{l,i}$  in two steps as
\begin{equation}\label{eq:linearbasis_raw}
		  \tilde{\pmb{\varphi}}_{l,i} = \text{curl}(\lambda_{i+1}\lambda_{i+3}R_i)
		  =(\pmb{\tau}_{i+1}\lambda_{i+3} + \lambda_{i+1}\pmb{\tau}_{i+3})R_i + 
		  \lambda_{i+1} \lambda_{i+3}\text{curl}R_i.
\end{equation}
We then normalize it such that $l_i$ varies from $-1$ to $1$ when restricted to edge $e_i$ as
\begin{equation}\label{eq:linearbasis}
		  \pmb{\varphi}_{l,i} = \frac{\tilde{\pmb{\varphi}}_{l,i}(\mathbf{x})
		  \lvert e_i \rvert}{4\lambda_{i+1}(\mathbf{x}_{e,i}) \lambda_{i+3}(\mathbf{x}_{e,i})R_i(\mathbf{x}_{e,i})},
\end{equation}
where the index $i \equiv i \Mod{4}$, and 
the length of edge $e_i$ is  $\lvert e_i \rvert = \lvert \mathbf{x}_{v,i} - \mathbf{x}_{v,i+3} \rvert$. In Figure~\ref{fig:varphil},
we draw $\pmb{\varphi}_{l,2}$ on the left element and $\pmb{\varphi}_{l,0}$ on
the right element sharing an edge,
and show the normal flux, but not the tangential flux
joining continuously as a linear function on the shared edge.

The construction of the constant basis functions is a bit more complicated.
We follow the process by~\cite{directSerendipity}.
We start by defining a linear nodal function $\phi^*_{v,i} \in \mathcal{DS}_{2}(E)$
such that $\phi^*_{v,i}(\mathbf{x}_{v,k}) = \delta_{ik}$. Its restriction to 
each edge of $E$ is a linear function as
\begin{equation}\label{eq:phi*}
		  \phi^*_{v,i}(\mathbf{x}) = \varphi^2_{v,i}(\mathbf{x}) + \frac{1}{2} \big[\varphi^2_{e,i} + \varphi^2_{e,i+1} \big],
\end{equation}
where $i \equiv i \Mod{4}$. We then define $\pmb{\varphi}^*_{v,i}(\mathbf{x}) = \text{curl}\, \phi^*_{v,i}$ so that
\begin{equation}
		  \pmb{\varphi}^*_{v,i}\cdot \pmb{\nu}_j \rvert_{e,j} = \left \{
					 \begin{aligned}
								&1/\lvert e_i \rvert, \hspace{0.8cm} j=i,\\
								&-1/\lvert e_i \rvert, \hspace{0.3cm} j=i+1,\\
								&0,                 \hspace{1.6cm}   j=i+2,i+3.
					 \end{aligned}
					 \right.
\end{equation}
The ``constant" basis function on the edge can now be defined as
\begin{equation}\label{eq:constbasis}
		  \pmb{\varphi}_{c,i}= \frac{(\mathbf{x}_{v,i} - \mathbf{x}_{v,i+2}) \cdot \pmb{\nu}_{i+1} \lvert e_{i+1} \rvert\pmb{\varphi}^*_{v,i} + \mathbf{x} - \mathbf{x}_{v,i+2}}
		  {(\mathbf{x}_{v,i} - \mathbf{x}_{v,i+2}) \cdot \pmb{\nu}_{i+1} \lvert e_{i+1} \rvert / \lvert e_i \rvert + (\mathbf{x}_{v,i} - \mathbf{x}_{v,i+2})\cdot \pmb{\nu}_i},
\end{equation}
where $i \equiv i \Mod{4}$. The ``constant" shape function $\pmb{\varphi}_{c,i}$ 
has a normal flux of constant value $1$ on edge $e_i$ and evaluates to $0$ 
normal flux on all the other edges. In Figure~\ref{fig:varphic},
we draw $\pmb{\varphi}_{c,2}$ on the left element and $\pmb{\varphi}_{c,0}$ on
the right element sharing an edge,
and show that the normal flux
joins continuously as a constant $1$ on the shared edge.

\subsection{Stability and Approximation Properties for $\mathbb{V}_1^0$}
The accuracy and stability of $\mathbb{V}_1^0$ has been well studied and 
tested in the work by~\cite{directSerendipity}.
We state the result directly for completeness.
They define a projection operatior $\pi$ that preserves element average 
divergence and average edge normal fluxes,
\begin{equation}\label{eq:RT}
		  \pi : H(\text{div}; \Omega) \cap (L^{2+\epsilon}(\Omega))^2
		  \rightarrow \mathbb{V}_1^{0},
\end{equation}
where $\epsilon >0$. This global projection operator $\pi$ is a
combination of local operators $\pi_E$ defined on each element $E$
according to the degrees of freedom.
The operator $\pi$ also satisfies the commuting diagram property 
\begin{equation}
		  \mathcal{P}_{W_0} \nabla \cdot \mathbf{v} = \nabla \cdot \pi \mathbf{v},
\end{equation}
where $\mathcal{P}_{W_0}$ is the $L^2$-orthogonal projection operator onto
$W_0 = \nabla \cdot \mathbb{V}_1^0$, which is a space of piecewise 
constant values.
\begin{lem}\label{lem:theoryhdiv}
		  Stability and approximation properties for $\mathbb{V}_1^0$.

		  Under the assumptions of Lemma~$\ref{lem:interpolationop}$ and 
		  Lemma~$\ref{lem:BHlemma}$, there exists a 
		  constant $C>0$, independent of $\mathcal{T}_h$ and $h>0$, such that
%		  \begin{equation}
%		\norm{p-p_h}_{m,\Omega} \leq C h^{l+1-m} \lvert p \rvert_{l+1,\Omega}, 
%					 \quad l=0,1,2,\dots, r,
%		  \end{equation}
%		  where $p_h \in \mathcal{DS}_2$. We have
		  \begin{equation}
					 \begin{alignedat}{2}
								\norm{\mathbf{u} - \pi_D \mathbf{u}}_{0,\Omega} &\leq C \norm{\mathbf{u}}_{k,\Omega}h^k, \hspace{0.8cm} &&k=1, 2, \\
								\norm{\nabla\cdot(\mathbf{u} - \pi_D \mathbf{u})}_{0,\Omega} &\leq C \norm{\nabla \cdot \mathbf{u}}_{k,\Omega}h^k, \hspace{0.5cm} &&k=0, 1,\\
								\norm{p - \mathcal{P}_{W_s} p }_{0,\Omega} &\leq C \norm{p}_{k,\Omega}h^k, \hspace{0.9cm} &&k=0, 1. 
		 \end{alignedat}
		  \end{equation}
		  Moreover, the discrete inf-sup condition holds
		  \begin{equation}
					 \sup_{\mathbf{v}_h \in \mathbb{V}_1^0} \frac{(\omega_h, \nabla \cdot \mathbf{v}_h)}{\norm{\mathbf{v}_h}_{H(\text{div})}} \geq \gamma \norm{\omega_h}_{0,\Omega}, \quad \forall \omega_h \in W_0,
		  \end{equation}
		  for some $\gamma > 0$ independent of $\mathcal{T}_h$ and $h>0$.
\end{lem}
\begin{proof}
		  See~\cite{directSerendipity}.
\end{proof}

\subsection{A Stokes Model Problem}
Consider a model problem of an incompressible isotropic equation equipped with homogeneous 
boundary condition as
\begin{equation}\label{eq:stokesmodel}
		  \begin{aligned}
					 -\Delta \mathbf{u} + \nabla p &= \mathbf{f} \hspace{0.3cm} \text{in} \hspace{0.2cm} \Omega, \\
					 \nabla \cdot \mathbf{u}  &= 0 \hspace{0.3cm} \text{in} \hspace{0.2cm} \Omega,\\
					 \mathbf{u} &= \mathbf{0} \hspace{0.25cm} \text{on} \hspace{0.2cm} \partial \Omega.
		  \end{aligned}
\end{equation}
%where $f \in L^2(\Omega)$, and 
%$\pmb{\tau}(\mathbf{u}) = 2(\mathcal{D} \mathbf{u} - 1/3 \nabla \cdot \mathbf{u}
%\mathbf{I})$. The symmetrical gradient 
%$\mathcal{D} \mathbf{u} = 1/2(\nabla \mathbf{u} + \nabla \mathbf{u}^T)$.
We choose energy spaces for the displacement-pressure pair $(\mathbf{u},p)$  in 
the identical test and trial spaces as
\begin{equation}
		  \begin{aligned}
					 \mathbb{V}_S &= \{\mathbf{v} \in (H^1(\Omega))^2 : \mathbf{v} = \mathbf{0} \hspace{0.3cm} \text{on} \hspace{0.2cm} \partial \Omega\} = (H^1_0(\Omega))^2, \\
					 \mathbb{U}_S &= \{\mathbf{u} \in (H^1(\Omega))^2 : \mathbf{u} = \mathbf{0} 
					 \hspace{0.3cm} \text{on} \hspace{0.2cm} \partial \Omega\} = \mathbb{V}_S, \\
					 \mathbb{W}_S &= \bigg\{p \in L^2(\Omega) : \int_{\Omega}p\, d\mathbf{x} = 0\bigg\} = L^2(\Omega)/\mathbb{R}.
		  \end{aligned}
\end{equation}
We consider the standard mixed variational form as

Find $\mathbf{u} \in \mathbb{U}_S$  
and $p \in \mathbb{W}_S$ such that
\begin{equation}\label{eq:stokesweakmodel}
		  \begin{aligned}
					 (\nabla \mathbf{u}, \nabla\mathbf{v}) - (p, \nabla \cdot \mathbf{v}) &= (\mathbf{f},\mathbf{v}), \hspace{0.2cm} \forall \mathbf{v} \in \mathbb{V}_S,\\
					 (\nabla \cdot \mathbf{u}, \omega) &= 0, \hspace{1cm} \forall \omega \in \mathbb{W}_S.
		  \end{aligned}
\end{equation}
We introduce two finite-dimensional subspaces $\mathbb{V}_{S,h} \subset \mathbb{V}$ and $\mathbb{W}_{S,h} \subset \mathbb{W}$, and consider
the discretized problem: 

Find $(\mathbf{u}_h,p_h) \in \mathbb{V}_{S,h} \times \mathbb{W}_{S,h}$ such that
\begin{equation}\label{eq:stokesweakmodel_dis}
		  \begin{aligned}
					 (\nabla \mathbf{u}_h, \nabla \mathbf{v}_h) - (p_h, \nabla \cdot \mathbf{v}_h) &= (\mathbf{f},\mathbf{v}_h), \hspace{0.2cm} \forall \mathbf{v}_h \in \mathbb{V}_{S,h},\\
					 (\nabla \cdot \mathbf{u}_h, \omega_h) &= 0, \hspace{1.15cm} \forall \omega_h \in \mathbb{W}_{S,h}.
		  \end{aligned}
\end{equation}
Now we construct a pair of discretized space $\mathbb{V}_{S,h} \times \mathbb{W}_{S,h}$ for this Stokes model problem in the next subsection.

\subsection{Modified Bernardi-Raugel Element}
Stable mixed finite element pairs for the Stokes problem have been widely and thoroughly studied on triangles and rectangles.
One notable construction on rectangles was discussed in~\cite{BRelement} and~\cite{Fortin1981OldAN},
where polynomial spaces are supplemented with bubble functions.
Consider the standard type of Bernardi-Raugel (BR) element using the space of tensor product
polynomials
\begin{equation}
(\mathbb{Q}_{1,1} (E))^2 \oplus \text{span}\{\mathbf{b}_0, \mathbf{b}_1, \mathbf{b}_2, \mathbf{b}_3 \} \subset (\mathbb{Q}_{2,2}(E))^2,
\end{equation}
where $\mathbf{b}_i$ is a quadratic bubble function defined for each edge $e_i$.
A similar but more general type of element was discussed by~\cite{doi:10.1137/S0036142903435247}.

We now modify the original BR element to extend it to a new finite element suited to quadrilaterals using the direct serendipity space as 
\begin{equation}\label{eq:br}
		  \mathbb{V}_1^{\mathcal{BR}}(E) = (\mathcal{DS}_1(E))^2 \oplus \text{span}\{\mathbf{b}_0, \mathbf{b}_1, \mathbf{b}_2, \mathbf{b}_3\} \subset (\mathcal{DS}_2(E))^2,
\end{equation}
where $\text{dim} \mathbb{V}_1^{\mathcal{BR}}(E) = 12.$
One natural choice for the supplemental bubble function $\mathbf{b}_i$ is the previously defined 
edge nodal basis function $\varphi^{(2)}_{e,i} \in \mathcal{DS}_2$ times the unit normal, i.e., 
\begin{equation}\label{eq:bubble}
		  \mathbf{b}_i(\mathbf{x}) = \varphi^{(2)}_{e,i}(\mathbf{x})\pmb{\nu}_i, 
\end{equation}
which is quadratic when restricted to edge $e_i$ according to~\eqref{eq:edgerestriction}. 
The quadratic bubble is guaranteed to be conforming since it evaluates to $0$ on two vertices and $1$ at 
the middle point of the edge due to normalization, which is uniquely 
determined by these three points. The integral of this quadratic bubble can be
accurately computed by a three-point Gaussian quadrature rule.

We state the degrees of freedom of this type of element as follows.
\begin{defn}\label{lem:unisolvence}
		  For an element $E\in \mathcal{T}_h$, $\mathbf{u}_h$ is uniquely defined
		  by the following 12 degrees of freedom:

		  \text{(i)} for corner point $v_i$ and Cartesian direction unit normal vectors $\{\mathbf{i},\mathbf{j}\}$,
		  \begin{equation*}
					 DOF^{\text{(i)}}_{(v_i,\mathbf{i})}(\mathbf{u}_h) = \mathbf{u}_h(v_i)\cdot \mathbf{i}, 
					 \quad DOF^{\text{(i)}}_{(v_i,\mathbf{j})}(\mathbf{u}_h) = \mathbf{u}_h(v_i)\cdot \mathbf{j};
		  \end{equation*}

		 \text{(ii)} for edge $e_i$,
		  \begin{equation*}
					 DOF^{(ii)}_{e_i} = \langle \mathbf{u}_h \cdot \pmb{\nu}_i, 1 \rangle.
		  \end{equation*}
\end{defn}
We give an explicit construction of the shape functions. 
We choose the nodal basis function on vertex $v_i$ in two Cartesian directions as
		  \begin{equation}
					 N_{v_i,x} = \begin{bmatrix}
								\varphi^{(1)}_{v,i}(\mathbf{x})\\
								0
					 \end{bmatrix}, \quad  N_{v_i,y} = \begin{bmatrix}
								0\\
								\varphi^{(1)}_{v,i}(\mathbf{x})
					 \end{bmatrix},
		  \end{equation}
where $\varphi^{(1)}_{v,i}$ is the vertex nodal shape function in $\mathcal{DS}_1(E)$.
We choose the edge basis function on edge $e_i$ as~$\eqref{eq:bubble}$, i.e.,
\begin{equation}
		  N_{e} = \varphi^{(2)}_{e,i}\pmb{\nu}_i
\end{equation}
Note that bubble functions are quadratic when restricted to edge $e_i$, while
nodal basis functions are linear when restricted to edge $e_i$. Therefore 
the bubble
functions cannot be in the same space as the nodal basis functions.
We conclude
that this element is well-defined, i.e., the degrees of freedom uniquely 
determine the function.

We pair it with the space 
\begin{equation}
\mathbb{W}_0 = \{\omega \in L^2(\Omega); \hspace{0.3cm} \omega|_E \in P_0 \} /\mathbb{R},
\end{equation}
which has piecewise constant values for each element $E$.
We provide stability and approximation analysis for this space pair $\mathbb{V}_1^{\mathcal{BR}} \times \mathbb{W}_0$
in the next subsection.

\subsection{Stability and Approximation Properties for $\mathbb{V}_1^{\mathcal{BR}}\times\mathbb{W}_0$}
Before analyzing the stability and approximation properties for the new 
modified BR element, we state one useful lemma.
\begin{lem}\label{lem:intestimate}
Let $E$ be a nondengenate convex quadrilateral element
that is shape regular as in definition~$\ref{def:shaperegularity}$.
For each integer $m\geq 0$
and real number $p \in [1,\infty]$ there exist positive constants $C_1$
and $C_2$, possibly depending on $\sigma^*$ but otherwise independent of 
 the geometry of $E$, such that the following
upper bounds hold for all $v \in W_p^m(E)$:
\begin{equation}
\left \{  \begin{aligned}
&\lvert v \rvert_{W^0_p(E)} \leq C_1 h_E^{2/p}\lvert \tilde{v} \rvert_{W^0_p(\tilde{E})},\\
		  &\lvert v \rvert_{W^1_p(E)} \leq C_2(\sigma^*,p) h^{2/p-1} \lvert \tilde{v} \rvert_{W^1_p(\tilde{E})},
\end{aligned}
\right.
\end{equation}
where $\tilde{E}$ represents the local reference quadrilateral element $\tilde{E}$ illustrated in Figure~$\ref{fig:meshexample}$, and
$\tilde{v}$ is the function defined on the local reference element $\tilde{E}$.
\end{lem}
\begin{proof}
See~\cite{FinElemMethNavierStokes}. 
\end{proof}

We reuse the interpolation operator $\mathcal{I}_{h}^1$ defined in equation~\eqref{eq:interpolation}.
Define the projection operator $\pi \in (H^1_0(\Omega))^2 \rightarrow \mathbb{V}_1^{\mathcal{BR}} \cap (H^1_0(\Omega))^2$ 
as a combination of local projection operators $\pi_E$ as:

(i) For each vertex $v \in \partial E$
\begin{equation}\label{eq:nodaldof}
		  \pi_E \mathbf{v}(\mathbf{x}_v) = \mathcal{I}^1_{h}\mathbf{v}(\mathbf{x}_v);
\end{equation}

(ii) For each edge $e \subset \partial E$,
\begin{equation}\label{eq:edgedof}
\int_{e} (\pi_E \mathbf{v} - \mathbf{v})\cdot \pmb{\nu}\, ds = 0.
\end{equation}

\begin{lem}\label{lem:approximibility}
For all $\mathbf{v} \in (H^k(\Omega))^2 \cap (H^1_0(\Omega))^2,\quad k=1,2$,
the operator $\pi_E$ defined by~\eqref{eq:nodaldof}--\eqref{eq:edgedof} satisfies
\begin{equation}
(q_h, \nabla \cdot (\mathbf{v} - \pi_E \mathbf{v})) = 0, \quad \forall q_h \in \mathbb{W}_0
\end{equation}
Assuming shape regularity as defined in Definition~$\ref{def:shaperegularity}$, the 
global projection operator $\pi$ satisfies the 
error bound
\begin{equation}
\norm{\mathbf{v} - \pi \mathbf{v}}_{H^1(\Omega)} \leq Ch^m\norm{\mathbf{v}}_{H^{m+1}(\Omega)}, \quad \forall \mathbf{v} 
\in (H^{m+1}(\Omega))^2, \hspace{0.2cm} m=0,1.
\end{equation}
\end{lem}

\begin{proof}
According to the divergence theorem, we have
\begin{equation}
\int_{E} \nabla \cdot (\mathbf{v} - \pi_E \mathbf{v})\, d\mathbf{x} = \sum_{i=0}^3 \int_{e_i} (\mathbf{v} - \pi_E \mathbf{v}) \cdot \pmb{\nu}\, ds = 0.
\end{equation}
Given that $q_h$ is piecewise constant on the element $E$, it is  clear that we have
\begin{equation}
(q_h, \nabla \cdot (\mathbf{v} - \pi_E \mathbf{v})) = 0, \quad \forall q_h \in \mathbb{W}_0.
\end{equation}

From the definition of the degrees of freedom~\eqref{eq:nodaldof} and~\eqref{eq:edgedof}, we decompose the projection
operator $\pi_E$ as
\begin{equation}\label{eq:decomposition}
		  \pi_E \mathbf{v} = \mathcal{I}^1_{h} \mathbf{v} + 
		  \sum_{i=0}^3 \alpha_i \mathbf{b}_i,
\end{equation}
where the coefficient 
\begin{equation}
		  \alpha_i = \frac{ \int_{e_i} (\mathbf{v} - \mathcal{I}^1_{h} \mathbf{v})\cdot \pmb{\nu}_i\, ds}{\int_{e_i} \mathbf{b}_i \cdot 
		  \pmb{\nu}_i\, ds}.
\end{equation}
The approximation property of the interpolation operator $\mathcal{I}_h^1$ has been discussed in Lemma~\ref{lem:approximation},
\begin{equation}\label{eq:interpoapproxbr}
\norm{\mathbf{v} - \mathcal{I}^1_{h}\mathbf{v}}_{H^m(E)} \leq h^{n-m} \lvert \mathbf{v} \rvert_{H^n(E^*)}, \quad m= 0,1, \quad n = 1,2.
\end{equation}
where $E^* = \cup_{F\in \mathcal{T}_h, F\cap E \neq \emptyset}$ defined in Lemma~\ref{lem:interpolationop}.

Lemma~\ref{lem:intestimate} indicates an estimation of the bubble function as
\begin{equation}\label{eq:estimatebubble}
\lvert\mathbf{b}_i\rvert_{H^m(E)} \leq C(\sigma^*)h^{1-m}\lvert \tilde{\mathbf{b}}_i \rvert_{H^1(\tilde{E})} 
\leq C(\sigma^*)h^{1-m}, \quad m = 0,1.
\end{equation}
We continue to estimate $\lvert \alpha_i \rvert$. Note
\begin{equation}
		  \int_{e,i} \mathbf{b}_i\cdot \pmb{\nu}_i \, ds = 
		  \int_{e,i} \varphi^{(2)}_{e,i} \, ds = 
		  \lvert e_i \rvert \int_{\tilde{e}_i} \tilde{\varphi}^{(2)}_{e,i}\, d\tilde{s} = C_1 \lvert e_i \rvert.
\end{equation}
		  %According to Trace Theorem~\ref{thm:tracethm}, a continuous trace mapping can be constructed. 
		  %Combining with Lemma~\ref{lem:intestimate}, we estimate the term $\lvert \alpha_i \rvert$ as
%\begin{equation}\label{eq:estimatealpha}
%					 \lvert \alpha_i \rvert \leq C \int_{\tilde{e},i} \lvert\tilde{\mathbf{v}} - \tilde{\mathcal{I}}^1_{h} \tilde{\mathbf{v}}\rvert\, ds \leq C h^k h^{-1} \lvert \mathbf{v} \rvert_{H^k(E^*)} , \quad k = 1,2.
%\end{equation}
Then
		  \begin{equation}
					 \begin{aligned}
					 \lvert \alpha_i \rvert = \bigg\lvert \frac{ \int_{e_i} (\mathbf{v} - \mathcal{I}^1_{h} \mathbf{v})\cdot \pmb{\nu}_i\, ds}{\int_{e_i} \mathbf{b}_i \cdot 
								\pmb{\nu}_i\, ds} \bigg\rvert &\leq C \int_{\tilde{e},i} \lvert\tilde{\mathbf{v}} - \tilde{\mathcal{I}}^1_{h} \tilde{\mathbf{v}}\rvert\, ds \\
								&\leq C \norm{\tilde{\mathbf{v}} - 
								\tilde{\mathcal{I}}_h^1\mathbf{v}}_{L^2(\tilde{e_i})}.
					 \end{aligned}
		  \end{equation}
As the trace mapping is contimuous from $H^1(\tilde{E})$ into $L^2(\tilde{e})$, according to the Trace Theorem~\ref{thm:tracethm},
		  \begin{equation}
					 \begin{aligned}
								\lvert \alpha_i \rvert &\leq C \norm{\tilde{\mathbf{v}} - 
								\tilde{\mathcal{I}}_h^1\mathbf{v}}_{H^1(\tilde{E})}\\
								&\leq C \bigg( \norm{\tilde{\mathbf{v}} - 
								\tilde{\mathcal{I}}_h^1\mathbf{v}}_{L^2(E)} + h^{-1} 
								\Big\lvert \tilde{\mathbf{v}} - 
								\tilde{\mathcal{I}}_h^1\mathbf{v}\Big\rvert_{H^1(E)}\bigg).
					 \end{aligned}
		  \end{equation}
		  According to Lemma~$\ref{lem:approximation}$,
		  \begin{equation}
					 \lvert \alpha_i \rvert \leq C h^{k-1}\norm{\mathbf{v}}_{H^k(E^*)}, \quad k = 1,2. 
		  \end{equation}
The overall estimate is then
\begin{equation}\label{eq:estimationalphav}
		  \lvert\alpha_i \rvert \lvert \mathbf{b}_i \rvert_{H^m(E)} \leq C h^{k-m} \norm{ \mathbf{v}}_{H^k(E^*)}, \quad m=0,1, \quad k=1,2. 
\end{equation}
Combining with the interpolation error estimate result in equation~\eqref{eq:interpoapproxbr},
we get a local error estimate as
\begin{equation}
		  \norm{\mathbf{v} - \pi_E \mathbf{v}}_{H^1(E)} \leq C h \norm{\mathbf{v}}_{H^2(E)}.
\end{equation}
The global error estimate is obtained by combining local results together.
\end{proof}
In addition to the approximability lemma~\ref{lem:approximation}, we
also need a stability result.
\begin{lem}\label{lem:brstability}
		  %For any $\mathbf{v} \in H^1_0(\Omega)$, there exists $\mathbf{v}_h \in \mathbb{V}_1^{\mathcal{BR}}$ such that 
		  Assuming that $\mathbf{v} \in (H^1(\Omega))^2$,
		  the linear operator $\pi$ is bounded on $(H^1(\Omega))^2$ independently of h, i.e., 
		  \begin{equation}\label{eq:mypiboundness}
					 \norm{\pi \mathbf{v}}_{H^1(\Omega)} \leq C\norm{\mathbf{v}}_{H^1(\Omega)}.
		  \end{equation}
\end{lem}
\begin{proof}
		  %We express discrete solution $\mathbf{v}_h$ with the predefined
		  %projection operator $\mathbf{v}_h = \pi_h \mathbf{v}$, and denote
		  %the $\mathbf{w}_h = \mathcal{I}_h^1 \mathbf{v}$, then we have the following local
		  %inequality as
		  %\begin{equation}
		%			 \norm{\mathbf{v}_h} \leq \norm{\mathbf{w}_h}_{H^1(E)} + h^{-1} {\big\{}\norm{\mathbf{v}-\mathbf{w}_h}_{L^2(E)} + h \lvert \mathbf{v} - \mathbf{w}_h \rvert_{H^1(E)}  {\big\}} \leq C \norm{\mathbf{v}}_{H^1(E)}.

\begin{thm}
With the established approximability result stated in $Lemma~\ref{lem:approximibility}$, and 
		  the stability result stated in $Lemma~\ref{lem:brstability}$,
for a convex domain $\Omega$, there exists a constant $\gamma > 0$ such that
		  \begin{equation}
					 \inf_{q_h \in \mathbb{W}_0} \sup_{\mathbf{v}_h \in \mathbb{V}_1^{\mathcal{BR}}} \frac{(\nabla \cdot \mathbf{v}_h, q_h)}{\norm{\mathbf{v}_h}_{H^1(\Omega)}\norm{q_h}_{L^2(\Omega)}} \geq \gamma >0.
		  \end{equation}
\end{thm}
\begin{proof}
Note that
		  \begin{equation}
					 \sup_{\mathbf{v}_h \in \mathbb{V}_1^{\mathcal{BR}}} \frac{(\nabla \cdot \mathbf{v}_h, q_h)}{\norm{\mathbf{v}_h}_{H^1(\Omega)}\norm{q_h}_{L^2(\Omega)}} \geq \frac{(\nabla \cdot \pi\mathbf{v}, q_h)}{\norm{\pi \mathbf{v}}_{H^1(\Omega)} \norm{q_h}_{L^2(\Omega)}}.
		  \end{equation}
		  Recall that the inf-sup condition holds for the continuous space as
		  \begin{equation}
					 \inf_{q \in L^2(\Omega)} \sup_{\mathbf{v} \in (H^1(\Omega))^2} \frac{(\nabla \cdot \mathbf{v}, q)}{\norm{\mathbf{v}}_{H^1(\Omega)}\norm{q}_{L^2(\Omega)}} \geq \gamma >0.
		  \end{equation}
		  With the condition $(q_h, \nabla \cdot (\mathbf{v} - \pi_E \mathbf{v})) = 0$ proved in Lemma~\ref{lem:approximibility} 
		  we have
		  \begin{equation}
 \frac{(\nabla \cdot \pi\mathbf{v}, q_h)}{\norm{\pi \mathbf{v}}_{H^1(\Omega)} \norm{q_h}_{L^2(\Omega)}} = \frac{(\nabla \cdot \mathbf{v}, q_h)}{\norm{\pi \mathbf{v}}_{H^1(\Omega)} \norm{q_h}_{L^2(\Omega)}} \geq  \frac{\gamma}{C} > 0,
		  \end{equation}
where $C$ here is the constant in~\eqref{eq:mypiboundness}.
\end{proof}
We conclude that our space $\mathbb{V}_1^{\mathcal{BR}} \times \mathbb{W}_0$ 
forms a $\inf$-$\sup$ stable pair with first order overall accuracy.
