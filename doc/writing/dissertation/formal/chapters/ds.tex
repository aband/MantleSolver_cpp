\section{Direct Serendipity Space}
Serendipity element have the same accuracy of approximation with 
classical tensor product element while use smaller number of degrees of freedom.
However, the classical serendipity space suffer from poor approximation
on quadrilaterals. We define local shape functions directly
on quadrilaterals according to~\cite{directSerendipity}.
$H^1$ and $H(\text{div})$ conforming direct mixed finite elemenent space
are constructed with the direct serendipity function space.

To begin with, we introduce some distance auxilliary functions. 
We define the linear polynomial $\lambda_i(\mathbf{x})$ giving the
distance of $\mathbf{x}\in \mathbb{R}^2$ to edge $e_i$ as
\begin{equation}\label{eq:distancelambda}
		  \lambda_i(\mathbf{x}) = -(\mathbf{x}-\mathbf{x}^*_i)\cdot \pmb{\nu}_i, \quad i = 0,1,2,3,
\end{equation}
where $\mathbf{x}^*_i \in e_i$  is any point on the edge $e_i$.
We pick $\mathbf{x}^*_i = \mathbf{x}_{v,i}$ for simplicity. The value of
distance function $\lambda_i$ keeps positive as long as $\pmb{x}$ is in the
interior of the quadrilateral. We define two additional distance functions giving 
the distance of $\mathbf{x} \in \mathbb{R}^2$ to diagonal line $d_{i}$ 
joining $v_i$ with $v_{i+2}$, $i\in{0,1}$.
Let $\mathbf{\nu}_{d,i}$ be the unit normal vector to the diagonal line 
$d_{i}$. The direction of the $\mathbf{\nu}_{d,i}$ is defined by 
counterclockwise rotating of the vector pointing from $v_i$ to $v_{i+2}$.
The resulting diagonal distance function is defined as
\begin{equation}\label{eq:distancelambdad}
		  \lambda_{d,i}(\mathbf{x}) = -(\mathbf{x} - \mathbf{x}^*_{d,i}) \cdot \pmb{\nu}_{d,i}, 
		  \quad i=0,1,
\end{equation}
where $\mathbf{x}^*_{d,i}$ is any point on the diagonal line $d_{i}$. Here,
we use the corner point $v_i$ for simplicity,
$\mathbf{x}^*_{d,i} = \mathbf{x}_{v,i}$, where $\mathbf{x}_{v,i}$ denotes
the coordinates of corner $v_i$.

Now we present set of nodal basis functions in first and second
order direct serendipity space $\mathcal{DS}_1$ and $\mathcal{DS}_2$
defined by ~\cite{directSerendipity}.

\subsection{Direct Serendipity Space $\mathcal{DS}_2$}
We supplement polynomial space $\mathbb{P}_r(E), r\geq 2$ 
with two linearly independent
functions, $\phi_{s,1}(\mathbf{x})$ and $\phi_{s,2}(\mathbf{x})$, 
spanning the supplemental function space as $\mathbb{S}_r^{\mathcal{DS}}(E) = 
\text{span}\{\phi_{s,1},\phi_{s,2}\}, \quad r\geq 2$. We present the second order 
direct serendipity
space as
\begin{equation}\label{eq:definitionds2}
		  \mathcal{DS}_2(E) = \mathbb{P}_2(E) \oplus \mathbb{S}_2^{\mathcal{DS}}(E).
\end{equation}
The supplemental functions are defined as
\begin{equation}\label{eq:defsuppfunc}
		  \phi_{s,i}(\mathbf{x}) = \lambda_{i+1}(\mathbf{x}) \lambda_{i+2}(\mathbf{x}) R_{i,i+2}(\mathbf{x}), \quad i\in\{0,1\},
\end{equation}
where $i \equiv i\Mod{4}$ . We use a simple definition for rational
function $R_{i,i+2}$ as
\begin{equation}\label{eq:defR}
		  R_{i,i+2}(\mathbf{x}) = \frac{\lambda_{i}(\mathbf{x}) - \lambda_{i+2}(\mathbf{x})}{\lambda_i(\mathbf{x}) + \lambda_{i+2}(\mathbf{x})},
\end{equation}
which satisfies the conditions as
\begin{equation}\label{eq:Rproperty}
		  R_{i,i+2}|_{e_i} = -1, \quad R_{i,i+2}|_{e_{i+2}} = 1.
\end{equation}
We continue to define
\begin{equation}\label{eq:def}
		  %R_i(\mathbf{x}) = \frac{1}{2} \Big(1 - (-1)^{\floor*{i/2}} R_{i, i+2}(\mathbf{x})\Big) 
		  R_i(\mathbf{x}) = \frac{1}{2} \Big(1 -  R_{i, i+2}(\mathbf{x})\Big) 
		  = \frac{\lambda_{i+2}(\mathbf{x})}{\lambda_i(\mathbf{x}) + \lambda_{i+2}(\mathbf{x})},\quad i \in \{0,1,2,3\}
\end{equation}
where $i \equiv i \Mod{4}$ as usual, and $R_{i,i+2} = -R_{i+2,i}$ according to 
the equation~\eqref{eq:defR}. 
Here, $R_i|_{e_i} = 1$, and
$R_i|_{e_{i+2}} = 0$ by definition. In Figure~\ref{fig:R0}, we
draw $R_0$ on a general quadrilateral element, which evaluates $1$ 
on the edge $e_0$, and evaluates $0$ on the opposite edge $e_2$.

\includegraphics[width=5.5cm,height=4cm]{chapters/pics/R0_2.png}
		  \caption{Rational function $R_0$.}

\end{figure}

The edge nodal basis functions for $\mathcal{DS}_2$ are constructed as
\begin{equation}\label{eq:edgebasisds2}
		  \varphi^{(2)}_{e,i}(\mathbf{x}) = \frac{\lambda_{i+1}(\mathbf{x})\lambda_{i+3}(\mathbf{x})R_i(\mathbf{x})}
		  {\lambda_{i+1}(\mathbf{x}_{e,i})\lambda_{i+3}(\mathbf{x}_{e,i})R_i(\mathbf{x}_{e,i})},
\end{equation}
where $\mathbf{x}_{e,i}$ is the coordinates of the middle point of the 
edge $e_i$. 
Here, $i \equiv i \Mod{4}$.
We can restrict the edge basis function to edges as
\begin{equation}\label{eq:edgerestriction}
		  \varphi^{(2)}_{e,i}|_{e,j} = \left \{
					 \begin{aligned}
								&\frac{\lambda_{i+1}(\mathbf{x})\lambda_{i+3}(\mathbf{x})}
								{\lambda_{i+1}(\mathbf{x}_{e,i})\lambda_{i+3}(\mathbf{x}_{e,i})}, \hspace{0.2cm} i=j \\
								& 0, \hspace{3.4cm} i\neq j
					 \end{aligned}
		  \right .
\end{equation}

In Figure~\ref{fig:phie0}, we present
$\varphi^{(2)}_{e,0}$ on a general quadrilateral element, where 
$\varphi^{(2)}_{e_0}$ evalues $0$ on the edges $\{e_i, \quad i\neq 0\}$, and 
evaluates a quadratic bubble function on the edge $e_0$.

The vertex nodal basis functions for $\mathcal{DS}_2$ are constructed as
\begin{equation}\label{eq:vertexbasisds2}
		  \varphi^2_{v,i}(\mathbf{x}) = \frac{\lambda_{i+2}(\mathbf{x}) \lambda_{i+3}(\mathbf{x}) - 
		  \sum_{k=i}^{i+1}\lambda_{i+2}(\mathbf{x}_{e,i})\lambda_{i+3}(\mathbf{x}_{e,i})\varphi^2_{e,i}(\mathbf{x})}
		  {\lambda_{i+2}(\mathbf{x}_{v,i}) \lambda_{i+3}(\mathbf{x}_{v,i}) - 
		  \sum_{k=i}^{i+1}\lambda_{i+2}(\mathbf{x}_{e,i})\lambda_{i+3}(\mathbf{x}_{e,i})\varphi^2_{e,i}(\mathbf{x}_{v,i})},
\end{equation}
where $\mathbf{x}_{v,i}$ is the coordinates of the vertex node $v_i$, and $i \equiv i \Mod{4}$ as well. 
In Figure~\ref{fig:phiv0}, we

\subsection{Direct Serendipity Space $\mathcal{DS}_1$}
The supplemental space for $\mathcal{DS}_1$ differs from 
that for $\mathcal{DS}_2$. We present the first order serendipity
as
\begin{equation}\label{eq:definitionds1}
		  \mathcal{DS}_1(E) = \mathbb{P}_1(E) \oplus \text{span}\{R\},
\end{equation}
where $R(\mathbf{x}_{v,0}) = R(\mathbf{x}_{v,2}) = 
-R(\mathbf{x}_{v,1}) = -R(\mathbf{x}_{v,3}) = 1$. The construction 
of rational funcion R is not unique. We present one way to construct
$R(\mathbf{x})$ with edge nodal basis function~\eqref{eq:edgebasisds2} in $\mathcal{DS}_2$ as
\begin{equation}\label{eq:rationalfunctionds1}
R(\mathbf{x}) = r(\mathbf{x}) - \sum_{i=0}^3r(\mathbf{x}_{e,i}) \varphi^2_{e,i}(\mathbf{x}),
\end{equation}
where auxiliary function $r(\mathbf{x})$ is defined as
\begin{equation}\label{eq:auxiliaryrationalfunction}
r(\mathbf{x}) = \frac{\lambda_2(\mathbf{x})\lambda_3(\mathbf{x})}{\lambda_2(\mathbf{x}_{v,0})\lambda_3(\mathbf{x}_{v,0})} - 
 \frac{\lambda_0(\mathbf{x})\lambda_3(\mathbf{x})}{\lambda_0(\mathbf{x}_{v,1})\lambda_3(\mathbf{x}_{v,1})} + 
 \frac{\lambda_0(\mathbf{x})\lambda_1(\mathbf{x})}{\lambda_0(\mathbf{x}_{v,2})\lambda_1(\mathbf{x}_{v,2})} - 
 \frac{\lambda_1(\mathbf{x})\lambda_2(\mathbf{x})}{\lambda_1(\mathbf{x}_{v,3})\lambda_2(\mathbf{x}_{v,3})} .
\end{equation}
In Figure~\ref{fig:rationalds1}, we plot the rational function on a general quadrilateral.

{\lambda_{d,m}(\mathbf{x}_{v,i}) - \lambda_{d,m}(\mathbf{x}_{v,i+2})},
\end{equation}
where $m\equiv i+1 \Mod{2}$, and index $i\equiv i \Mod{4}$.

\subsection{An Interpolation Operator}
We refer to an interpolation operator mapping
onto $\mathcal{DS}_r(K_i)$, $r=1,2$ defined in~\cite{directSerendipity}. Here, $K_i$ has different meanings
with different types of basis functions. For each basis functions associated
with edge node, we set $K_i = e_i$. For each basis functions associated with
vertex node, we set $K_i$ being a set of edges sharing the vertex $v_i$.
Following the work by~\cite{directSerendipity}, we define a interpolation
operator $\mathcal{I}^r_h:W_p^l(E) \rightarrow \mathcal{DS}_r$ as
\begin{equation}\label{eq:interpolation}
					 \mathcal{I}^r_h v(\mathbf{x}) = \sum_{i=1}^{N_r} \varphi_i(\mathbf{x})\int_{K_i} \psi_i(\mathbf{y}) v(\mathbf{y}) d\mathbf{y} \in \mathcal{DS}_r,
\end{equation}
where $1\leq p \leq \infty$ and $l > 1/p$ (but $l\geq 1$ if $p=1$),
and $\psi(\mathbf{x})$ is the dual nodal basis such that
\begin{equation}
		  \int_{K_i} \psi_{i}(\mathbf{x}) \varphi_j(\mathbf{x}) d\mathbf{x} = \delta_{i,j}, \quad i,j \in \{0,1,2,\dots, N_r \},
\end{equation}
where $N_r$ is the number of nodal basis defined on $K_i$.
We present the following two lemmas regarding the constructed interpolation
operator for future reference.

\begin{lem}\label{lem:interpolationop}
		  (Boundedness of the interpolation operator.)

		  Let $\mathcal{T}_h$ be uniformly shape regular~\eqref{eq:regularityineq}
		  with shape regularity parameter $\sigma_*$.
		  For each element $E \in \mathcal{T}_h$, 
		  There exists a bounded interpolation operator~\eqref{eq:interpolation}.
		  For every $E \in \mathcal{T}_h$, suppose that the basis functions of $\mathcal{DS}_r(E)$ are constructed using
		  $\lambda_{24}$ and $\lambda_{13}$ such that the zero set $\mathcal{Z}_{i,i+2}$ intersects $e_{i+3}$ and $e_{i+1}$.
		  Moreover, assume that $R_{i,i+2}$ is uniformly differentiable functions of the vertices of $E$ up to order m.
		  Then for $r\geq 1$, $E \in \mathcal{T}_h$, 
		  $1\leq q \leq \infty$, and any nonnegative integer m,
		  \begin{equation}\label{eq:interpolationopbound}
		 \norm{\mathcal{I}_h^r v}_{W_q^m} \leq C(\sigma_*, m, q) \sum_{k=0}^l
					 h_E^{k-m+\frac{2}{q}-\frac{2}{p}} \lvert v\rvert_{W^k_p(E^*)},
		  \end{equation}
		  where $E^* = \cup_{F \in \mathcal{T}_h, F \cap E \neq \emptyset}$ and
		  $\lvert\hspace{0.1cm} \cdot\hspace{0.1cm} \rvert_{W_p^k}$ is the seminorm of k-th order derivatives.
\end{lem}
\begin{proof}
		  See~\cite{directSerendipity}.
\end{proof}

\begin{lem}\label{lem:approximation}
		  (Approximability of the interpolation operator.)

		  Under the assumptions of~\ref{lem:interpolationop}, there exists a 
		  constant $C = C(r, \sigma_*) > 0$ such that for all functions 
		  $v \in W_{l,p}(E^*)$, with $1\leq p \leq \infty$ and $l>1/p$ or 
		  ($l\geq 1$ if $p=1$),
		  \begin{equation}\label{eq:brambleHilbertDS}
					 \norm{v-\mathcal{I}_h^r v}_{W^m_p} \leq C h_E^{l-m}
					 \lvert v \rvert_{W^l_p(E^*)}, \quad 0 \leq m \leq \min(l,r+1).
		  \end{equation}
		  Morevoer, there exists a constant $C = C(r,\sigma_*) > 0$, independent
		  of $h = \max_{E \in \mathcal{T}}$, such that for all functions 
		  $v \in W_p^l(\Omega)$,
		  \begin{equation}
					 (\sum_{E\in\mathcal{T}_h} \norm{v - \mathcal{I}_h^r v}^p_{W_p^m})^{1/p} \leq C h^{l-m} \lvert v \rvert_{W_p^l(\Omega)}, 0 \leq m \leq \min(l, r+1).
		  \end{equation}
\end{lem}
\begin{proof}
		  See~\cite{directSerendipity}.
\end{proof}

\section{Mixed Finite Elements}
We then construct mixed finite elements from the introduced direct serendipity space.
Now recall Ciarlet's definition~\cite{doi:10.1137/1.9780898719208} 
of a finite element.
\begin{defn}\label{def:ciarletFEM}
		  A triple $(E, X(E), \varphi_j)$ defines a finite element.
		  \begin{itemize}
					 \item Element $E \in \mathcal{T}_h$;
					 \item Space of shape function $X(E)$, and a set of 
								shape function $\varphi_i \in X(E)$. The dimension of
								the shape function space $\text{dim}(X(E)) = n$;
					 \item A set of continuous and linear functionals $\phi_j$
								\begin{equation}
										  \phi_j : X(E) \supset \mathcal{X}(E) \rightarrow \mathbb{R},
								\end{equation}
								such that unisolvence condition is achieved as
								\begin{equation}
										  \langle \phi_j, \varphi_i \rangle = \delta_{ij}, \quad i,j \in \{1,2,3,\dots, n \}
								\end{equation}
		  \end{itemize}
\end{defn}
In the following sections, we state our shape function space constructed
using first and second order direct serendipity space and corresponding 
degrees of freedom.
We also provide explicit construction of shape functions.

We introduce a $H^1$ and a $H(\text{div})$ 
conforming mixed element space constructed with the pre-defined direct serendipity space
for Darcy and Stokes problems.

\subsection{A Darcy Model Problem}
Consider a model problem for Darcy equation as
\begin{equation}
\begin{aligned}
\bold{u} + \nabla p & = \bold{f}, \hspace{0.2cm} \text{in} \hspace{0.2cm} \Omega\\
\nabla \cdot \bold{u} & = 0, \hspace{0.25cm} \text{in} \hspace{0.2cm} \Omega \\
\mathbf{u} &= \mathbf{0}, \hspace{0.2cm} \text{on} \hspace{0.2cm} \partial \Omega
\end{aligned}
\end{equation}
We seek solution in the following energy space as
\begin{equation}
		  \begin{aligned}
					 \mathbb{V}_D &= \{\mathbf{v} \in (H(\text{div};\Omega))^2 : \mathbf{v}\cdot\pmb{\nu} = 0 \hspace{0.3cm} \text{on} \hspace{0.2cm} \partial \Omega\}, \\
					 \mathbb{U}_D &= \{\mathbf{u} \in (H(\text{div};\Omega))^2 : \mathbf{u}\cdot\pmb{\nu} = \mathbf{0} 
					 \hspace{0.3cm} \text{on} \hspace{0.2cm} \partial \Omega\} = \mathbb{V}_D, \\
					 \mathbb{W}_D &= \{p \in L^2(\Omega) : \int_\Omega p d\mathbf{x} = 0\} = L^2(\Omega)/\mathbb{R}.
		  \end{aligned}
\end{equation}
We write weak form with these energy space, and consider the standard variational form : find $\mathbf{u} \in \mathbb{U}_D$  
and $p \in L^2(\Omega)$ such that
\begin{equation}\label{eq:hdivweakmodel}
		  \begin{aligned}
					 (\mathbf{u}, \mathbf{v}) - (p, \nabla \cdot \mathbf{v}) &= (\mathbf{f},\mathbf{v}), \hspace{0.2cm} \forall \mathbf{v} \in \mathbb{V}_D,\\
					 (\nabla \cdot \mathbf{u}, \omega) &= 0, \hspace{1cm} \forall \omega \in \mathbb{W}_D.
		  \end{aligned}
\end{equation}
We introduce two finite-dimensional subspace $\mathbb{V}_{D,h} \subset \mathbb{V}$ and $\mathbb{W}_{D,h} \subset \mathbb{W}$, and consider
the discretized problem: find $(\mathbf{u}_h,p_h) \in \mathbb{V}_{D,h} \times \mathbb{W}_{D,h}$ such that
\begin{equation}\label{eq:hdivwekamodel_dis}
		  \begin{aligned}
					 (\mathbf{u}_h, \mathbf{v}_h) - (p_h, \nabla \cdot \mathbf{v}_h) &= (\mathbf{f},\mathbf{v}_h), \hspace{0.2cm} \forall \mathbf{v}_h \in \mathbb{V}_{D,h},\\
					 (\nabla \cdot \mathbf{u}_h, \omega_h) &= 0, \hspace{1.15cm} \forall \omega_h \in \mathbb{W}_{D,h}.
		  \end{aligned}
\end{equation}
Now we construct a pair of discretized space $\mathbb{V}_{D,h} \times \mathbb{W}_{D,h}$ for this Darcy model problem in the next section.

\subsection{A reduced $H(\text{div})$ conforming space $\mathbb{V}_1^{0}$}
We have some choice for $H(\text{div})$ confirming space. 
For example, we can choose the famous Ravier-Thomas element 
by~\cite{10.1007/BFb0064470}. The serendipity space is
the precursor of the Brezzi-Douglas-Marini element by~\cite{BDM}.
We recover Arbogast-Correa element by~\cite{doi:10.1137/15M1013705} if we 
use mapped direct serendipity space.
We generated a reduced $H(\text{div})$ approximation, that $\nabla \cdot \mathbf{u}$
is approximated one less order than approximation of $\mathbf{u}$.

We write the rotated 2D exact sequence for energy space as
\begin{alignat}{3}\label{eq:exactsequence}
&H^1 \xrightarrow[]{\text{curl}} H&&(\text{div}) \xrightarrow[]{\text{div}} &&L^2 \\
&\cup                            &&\cup                                     &&\cup\\
&\mathcal{DS}_2  \xrightarrow[]{\text{curl}} &&\mathbf{V}_r^{r-1} \xrightarrow[]{\text{div}} &&\mathbb{P}_{r-1}.
\end{alignat}
We define the first order reduced $H(\text{div})$ conforming space with $\mathcal{DS}_2$ as
\begin{equation}\label{eq:defhdiv}
		  \mathbb{V}_1^0 = \mathbb{P}_1^2 \oplus \text{curl}\mathbb{S}_2^{\mathcal{DS}}.
\end{equation}
The reduced space $\mathbb{V}_1^0$ can be given a global basis with the normal
components of the shape functions merged continuously on the shared edge of
two elements.
The dimension of the derived space is 
\begin{equation}\label{eq:hdivdim}
		  \text{dim}(\mathbb{V}_1^0) = 2\times 3 + 2 = 8.
\end{equation}
We need two independent basis functions on each edge of a quadrilateral $E$,
one of which has a vanishing average flux on the edge and the other one has a non-vanishing 
average flux on the edge.

We state the degree of freedom of element as normal flux on each edge as
\begin{equation}
		  \int_{e_i} \mathbf{u}\cdot \pmb{\nu} \varphi_i d\mathbf{x}, \quad \varphi_i \in \mathbb{V}_1^0.
\end{equation}
We need two independent shape functions on each edge.
We present a construction of two independent basis functions on each edge:
$\pmb{\varphi}_{l,i}$ and $\pmb{\varphi}_{c,i}$, such that
\begin{equation}\label{eq:constbasishdiv_show}
\pmb{\varphi}_{c,i}\cdot \pmb{\nu}_j \rvert_{e_j} = \left \{
		 \begin{aligned}
					&0, \hspace{0.8cm} j \neq i, \\
					&1, \hspace{0.8cm} j=i.
		 \end{aligned}
		 \right.
\end{equation}
\begin{equation}\label{eq:linearbasishdiv_show}
		  \pmb{\varphi}_{l,i}\cdot \pmb{\nu}_j \rvert_{e_j} = \left \{
					\begin{aligned}
							  &0, \hspace{0.8cm} j \neq i, \\
							  &l_i(\mathbf{x}),\hspace{0.2cm} j=i,
					\end{aligned}
					\right.
\end{equation}
where $l_i$ is a non vanishing linear polynomial defined on edge $e_i$ 
orthogonal to one, i.e., $\int_{e_i}l_i ds = 0$.

To begin with, we write curl of the auxiliary rational function $R_i$~\eqref{eq:defR} as
\begin{equation}\label{eq:curlauxiliary}
		 \text{curl}R_i(\mathbf{x}) = 
					 \frac{\pmb{\tau}_{i+2}(\mathbf{x}) \lambda_i(\mathbf{x}) - \pmb{\tau}_{i}(\mathbf{x}) \lambda_{i+2}(\mathbf{x})}{(\lambda_i(\mathbf{x}) + \lambda_{i+2}(\mathbf{x}))^2},
\end{equation}
where the index $i$ is understood with modulo of 4. 
We first present a construction of basis function $\pmb{\varphi}_{l,i}$ as
\begin{equation}\label{eq:linearbasis_raw}
		  \tilde{\pmb{\varphi}}_{l,i} = \text{curl}(\lambda_{i+1}\lambda_{i+3}R_i)
		  =(\pmb{\tau}_{i+1}\lambda_{i+3} + \lambda_{i+1}\pmb{\tau}_{i+3})R_i + 
		  \lambda_{i+1} \lambda_{i+3}\text{curl}R_i.
\end{equation}
We then normalize it such that $l_i$ varying from $-1$ to $1$ as
\begin{equation}\label{eq:linearbasis}
		  \pmb{\varphi}_{l,i} = \frac{\tilde{\pmb{\varphi}}_{l,i}(\mathbf{x})
		  \lvert e_i \rvert}{4\lambda_{i+1}(\mathbf{x}_{e,i}) \lambda_{i+3}(\mathbf{x}_{e,i})R_i(\mathbf{x}_{e,i})},
\end{equation}
where the index $i$ is understood in the sense of modulo 4, and 
edge length $\lvert e_i \rvert = \lvert \mathbf{x}_{v,i} - \mathbf{x}_{v,i+3} \rvert$. In Figure~\ref{fig:varphil},
we draw $\pmb{\varphi}_{l,2}$ on the left element and $\pmb{\varphi}_{l,0}$ on
the right element sharing a edge,
and show normal flux
joining continuously as a linear function on the shared edge.
\begin{figure}[ht]
\centering\small
\includegraphics[width=5.5cm,height=4cm]{chapters/pics/linear1.png}
\includegraphics[width=5.5cm,height=4cm]{chapters/pics/linear2.png}
		  \caption{Linear basis function $\pmb{\varphi}_l$.}
		  \label{fig:varphil}
\end{figure}

The construction of a the constant basis function is a bit more complicated.
We follow the process by~\cite{directSerendipity}.
We start by defining a linear nodal function $\phi^*_{v,i} \in \mathcal{DS}_{2}(E)$
such that $\phi^*_{v,i}(\mathbf{x}_{v,k}) = \delta_{ik}$. Its restriction to 
each edge of $E$ is a linear function as
\begin{equation}\label{eq:phi*}
		  \phi^*_{v,i}(\mathbf{x}) = \varphi^2_{v,i}(\mathbf{x}) + \big[\frac{1}{2}\varphi^2_{e,i} + \frac{1}{2}\varphi^2_{e,i+1} \big],
\end{equation}
where the index $i$ is understood in modulo of 4. Then define $\pmb{\varphi}^*_{v,i}(\mathbf{x}) = \text{curl} \phi^*_{v,i}$ so that
\begin{equation}
		  \pmb{\varphi}^*_{v,i}\cdot \pmb{\nu}_j \rvert_{e,j} = \left \{
					 \begin{aligned}
								&1/\lvert e_i \rvert, \hspace{0.8cm} j=i,\\
								&-1/\lvert e_i \rvert, \hspace{0.3cm} j=i+1,\\
								&0,                 \hspace{1.6cm}   j=i+2,i+3.
					 \end{aligned}
					 \right.
\end{equation}
We then define the "constant" basis function defined on the edge as
\begin{equation}\label{eq:constbasis}
		  \pmb{\varphi}_{c,i}= \frac{(\mathbf{x}_{v,i} - \mathbf{x}_{v,i+2}) \cdot \pmb{\nu}_{i+1} \lvert e_{i+1} \rvert\pmb{\varphi}^*_{v,i} + \mathbf{x} - \mathbf{x}_{v,i+2}}
		  {(\mathbf{x}_{v,i} - \mathbf{x}_{v,i+2}) \cdot \pmb{\nu}_{i+1} \lvert e_{i+1} \rvert / \lvert e_i \rvert + (\mathbf{x}_{v,i} - \mathbf{x}_{v,i+2})\cdot \pmb{\nu}_i},
\end{equation}
where the index $i$ is again understood in the sense of modulo of 4. The defined basis function $\pmb{\varphi}_{c,i}$ has flux of constant value $1$ on $e_i$ and $0$ 
on all the other edges. In Figure~\ref{fig:varphic},

\subsection{Stability and Approximation Properties for $\mathbb{V}_1^0$}
The accuracy and stability of such element has been well studied and 
tested in the work by~\cite{directSerendipity}.
We state the result directly for completeness.
Consider a projection operatior $\pi_D$ preserves element average 
divergence and average edge normal fluxes.
\begin{equation}\label{eq:RT}
		  \pi_D : H(\text{div}; \Omega) \cap (L^{2+\epsilon}(\Omega))^2
		  \rightarrow \mathbb{V}_1^{0},
\end{equation}
where $\epsilon >0$. The global projection operator $\pi$ is understood
as combination of local operator $\pi_E$ according to degree of freedom.
The operator $\pi$ also satisfies the commuting diagram property as
\begin{equation}
		  \mathcal{P}_{W_s} \nabla \cdot \mathbf{v} = \nabla \cdot \pi \mathbf{v},
\end{equation}
where $\mathcal{P}_{W_s}$ is the $L^2$-orthogonal projection operator onto
$W_s = \nabla \cdot \mathbb{V}_1^0$, which is a constant value in our case.
\begin{lem}\label{lem:theoryhdiv}
		  Stability and approximation properties.

		  Under the assumption of~\ref{lem:interpolationop}, there exists a 
		  constant $C>0$, independent of $\mathcal{T}_h$ and $h>0$, such that
		  \begin{equation}
		\norm{p-p_h}_{m,\Omega} \leq C h^{l+1-m} \lvert p \rvert_{l+1,\Omega}, 
					 \quad l=0,1,2,\dots, r,
		  \end{equation}
		  where $p_h \in \mathcal{DS}_2$. We have
		  \begin{equation}
		 \begin{aligned}
					\norm{\mathbf{u} - \pi_D \mathbf{u}}_{0,\Omega} &\leq C \norm{\mathbf{u}}_{k,\Omega}h^k, \hspace{0.8cm} k=1,\dots, 2 \\
					\norm{\nabla\cdot(\mathbf{u} - \pi_D \mathbf{u})}_{0,\Omega} &\leq C \norm{\nabla \mathbf{u}}_{k,\Omega}h^k, \hspace{0.5cm} k=0,\dots, 1\\
					\norm{p - \mathcal{P}_{W_s} p }_{0,\Omega} &\leq C \norm{p}_{k,\Omega}h^k, \hspace{0.9cm} k=0,\dots, 1 
		 \end{aligned}
		  \end{equation}
		  Moreover, the discrete inf-sup condition holds
		  \begin{equation}
					 \sup_{\mathbf{v}_h \in \mathbb{V}_1^0} \frac{(\omega_h, \nabla \cdot \mathbf{v}_h)}{\norm{\mathbf{v}_h}_{H(\text{div})}} \geq \gamma \norm{\omega_h}_{0,\Omega}, \quad \forall \omega_h \in W_s,
		  \end{equation}
		  holds for some $\gamma > 0$ independent of $\mathcal{T}_h$ and $h>0$.
\end{lem}
\begin{proof}
		  See~\cite{directSerendipity}.
\end{proof}

\subsection{A Stokes Model Problem}
Consider a model problem for Stokes equation as
Find the displacement $\mathbf{u}$ and pressure $p$ such that
\begin{equation}\label{eq:stokesmodel}
		  \begin{aligned}
					 -\Delta \mathbf{u} + \nabla p &= \mathbf{f} \hspace{0.3cm} \text{in} \hspace{0.2cm} \Omega, \\
					 \nabla \cdot \mathbf{u}  &= 0 \hspace{0.3cm} \text{in} \hspace{0.2cm} \Omega,\\
					 \mathbf{u} &= \mathbf{0} \hspace{0.25cm} \text{on} \hspace{0.2cm} \partial \Omega.
		  \end{aligned}
\end{equation}
%where $f \in L^2(\Omega)$, and 
%$\pmb{\tau}(\mathbf{u}) = 2(\mathcal{D} \mathbf{u} - 1/3 \nabla \cdot \mathbf{u}
%\mathbf{I})$. The symmetrical gradient 
%$\mathcal{D} \mathbf{u} = 1/2(\nabla \mathbf{u} + \nabla \mathbf{u}^T)$.
We pick energy space for displacement and pressure as
\begin{equation}
		  \begin{aligned}
					 \mathbb{V}_S &= \{\mathbf{v} \in (H^1(\Omega))^2 : \mathbf{v} = \mathbf{0} \hspace{0.3cm} \text{on} \hspace{0.2cm} \partial \Omega\} = (H^1_0(\Omega))^2, \\
					 \mathbb{U}_S &= \{\mathbf{u} \in (H^1(\Omega))^2 : \mathbf{u} = \mathbf{0} 
					 \hspace{0.3cm} \text{on} \hspace{0.2cm} \partial \Omega\} = \mathbb{V}_S, \\
					 \mathbb{W}_S &= \{p \in L^2(\Omega) : \int_{\Omega}p d\mathbf{x} = 0\} = L^2(\Omega)/\mathbb{R},
		  \end{aligned}
\end{equation}
We write weak form with these energy space, and consider the standard variational form : find $\mathbf{u} \in \mathbb{U}_S$  
and $p \in \mathbb{W}_S$ such that
\begin{equation}\label{eq:stokesweakmodel}
		  \begin{aligned}
					 (\nabla \mathbf{u}, \nabla\mathbf{v}) - (p, \nabla \cdot \mathbf{v}) &= (\mathbf{f},\mathbf{v}), \hspace{0.2cm} \forall \mathbf{v} \in \mathbb{V}_S,\\
					 (\nabla \cdot \mathbf{u}, \omega) &= 0, \hspace{1cm} \forall \omega \in \mathbb{W}_S.
		  \end{aligned}
\end{equation}
We introduce two finite-dimensional subspace $\mathbb{V}_{S,h} \subset \mathbb{V}$ and $\mathbb{W}_{S,h} \subset \mathbb{W}$, and consider
the discretized problem: find $(\mathbf{u}_h,p_h) \in \mathbb{V}_{S,h} \times \mathbb{W}_{S,h}$ such that
\begin{equation}\label{eq:stokesweakmodel_dis}
		  \begin{aligned}
					 (\nabla \mathbf{u}_h, \nabla \mathbf{v}_h) - (p_h, \nabla \cdot \mathbf{v}_h) &= (\mathbf{f},\mathbf{v}_h), \hspace{0.2cm} \forall \mathbf{v}_h \in \mathbb{V}_{S,h},\\
					 (\nabla \cdot \mathbf{u}_h, \omega_h) &= 0, \hspace{1.15cm} \forall \omega_h \in \mathbb{W}_{S,h}.
		  \end{aligned}
\end{equation}
Now we construct a pair of discretized space $\mathbb{V}_{S,h} \times \mathbb{W}_{S,h}$ for this Stokes model problem in the next section.

\subsection{Modified Bernardi-Raugel Element}
Stable mixed finite element pair for Stokes problem has been widely researched.
We are referring to the element discussed by~\cite{BRelement} and by~\cite{Fortin1981OldAN}.
A similar type of element was discussed by~\cite{doi:10.1137/S0036142903435247}.

Inspired by the work in direct serendipity space,
we consider the following modified space for each element $E \in \mathcal{T}_h$ as
\begin{equation}\label{eq:br}
		  V_1^{\mathcal{BR}}(E) = (\mathcal{DS}_1(E))^2 \oplus \text{span}\{\mathbf{b}_0, \mathbf{b}_1, \mathbf{b}_2, \mathbf{b}_3\},
\end{equation}
where $\mathbf{b}_i$ is a quadratic bubble function defined on each edge $e_i$.
One natural pick is the previously defined function $\varphi^{(2)}_{e,i} \in \mathcal{DS}_2$ as
\begin{equation}\label{eq:bubble}
		  \mathbf{b}_i = \varphi^{(2)}_{e,i}(\mathbf{x})\mathbf{n}_i, 
\end{equation}
which is quadratic when restricted to edge $e_i$ according to equation~\eqref{eq:edgerestriction}. 
The quadratic bubble is guaranteed to be conforming since it evaluates zeros on two vertex and $1$ at 
the middle point of the edge due to normalization. The integral of this quadratic bubble can be
accurately computed by a three points gaussian quadrature rule.
\begin{defn}\label{lem:unisolvence}
		  For an element $E\in \mathcal{T}_h$, $\mathbf{u}_h$ is uniquely defined
		  by the following degrees of freedom:

		  \text{(i)} for corner point $v_i$ and Cartesian direction unitnormal vectors $\{\mathbf{i},\mathbf{j}\}$,
		  \begin{equation*}
					 DOF^{\text{(i)}}_{(v_i,\mathbf{i})}(\mathbf{u}_h) = \mathbf{u}_h(v_i)\cdot \mathbf{i}, 
					 \quad DOF^{\text{(i)}}_{(v_i,\mathbf{j})}(\mathbf{u}_h) = \mathbf{u}_h(v_i)\cdot \mathbf{j};
		  \end{equation*}

		 \text{(ii)} for edge $e_i$,
		  \begin{equation*}
					 DOF^{(ii)}_{e_i} = \langle \mathbf{u}_h \cdot \pmb{\nu}_i, 1 \rangle,
		  \end{equation*}
\end{defn}
We give an explicit construction of basis functions using driect serendipity space. 
We pick nodal basis on vertex $v_i$ in two Cartesian direction as
		  \begin{equation}
					 N_{v_i,x} = \begin{bmatrix}
								\psi_{v,i}(\mathbf{x})\\
								0
					 \end{bmatrix}, \quad  N_{v_i,y} = \begin{bmatrix}
								0\\
								\psi_{v,i}(\mathbf{x})
					 \end{bmatrix}
		  \end{equation}
We pick edge basis on edge $e_i$ as
\begin{equation}
		  N_{e} = \varphi^{(2)}_{e,i}\pmb{\nu}_i
\end{equation}
Note that bubble functions are quadratic when restricted to edge $e_i$, while
nodal basis functions are linear when restricted to edge $e_i$, therefore bubble
functions cannot be in the same space as nodal basis function.
We conclude
that this element is well-defined, i.e., the degrees of freedom uniquely 
determine the function.

We define the corresponding discrete pressure sapce as
\begin{equation}
\mathbb{W}_0 = \{\omega \in L^2{\Omega}; \omega|_E \in P_0 \} \cap L^2(\Omega)/\mathbb{R},
\end{equation}
which holds piece wise constant values for each element $E$.
We provide stability and approximation analysis for this space pair $\mathbb{V}_1^{\mathcal{BR}} \times \mathbb{W}_0$
in the next section.

\subsection{Stability and Approximation Properties for $\mathbb{V}_1^{\mathcal{BR}}\times\mathbb{W}_0$}
Before analyze the stability and approximation properties for the 
modified, we state some useful lemma.
\begin{lem}\label{lem:intestimate}
Let $K$ be a nondengenate convex quadrilateral.
Assuming we have shape regularity~\ref{def:shaperegularity}.
For each integer $m\geq 0$
and real number $p \in [1,\infty]$ there exist positive constants $C_1$,
and $C_2$ independent of the geometry of $E$ such that the following
upper bound hold for all $v \in W_p^m(E)$:
\begin{equation}
\left \{  \begin{aligned}
&\lvert v \rvert_{W^0_p(K)} \leq C_1 h_E^{2/p}\lvert \tilde{v} \rvert_{W^0_p(\tilde{K})},\\
&\lvert v \rvert_{W^1_p(K)} \leq C_2(\sigma^*,p) \lvert \tilde{v} \rvert_{W^1_p(\tilde{K})},
\end{aligned}
\right.
\end{equation}
where $\tilde{K}$ represents quadrilateral before scaling illustrated in Figure~\ref{fig:meshexample}, and
$\tilde{v}$ is function defined on $\tilde{K}$ before scaling.
\end{lem}
\begin{proof}
See~\cite{FinElemMethNavierStokes}. 
\end{proof}

With the interpolation operator $\mathcal{I}_{\mathcal{DS}_1}$ defined in~\ref{eq:interpolation},
we define the projection operator $\pi_h \in (H^1_0(\Omega))^2 \rightarrow \mathbb{V}_1^{BR} \cap (H^1_0(\Omega))^2$ by:

(i) For each corner point $v \in \partial E$
\begin{equation}\label{eq:nodaldof}
		  \pi_h \mathbf{v}(\mathbf{x}_v) = \mathcal{I}_{\mathcal{DS}_1}\mathbf{v}(\mathbf{x}_v).
\end{equation}

(ii) On each edge $e \subset \partial E$,
\begin{equation}\label{eq:edgedof}
\int_{e} (\pi_h \mathbf{v} - \mathbf{v})\cdot \pmb{\nu} ds = 0.
\end{equation}

\begin{lem}\label{lem:approximibility}
For all $\mathbf{v} \in (H^k(\Omega))^2,\quad k=1,2$. 
The operator $\pi_h$ defined by~\eqref{eq:nodaldof} and ~\eqref{eq:edgedof} satisfies:
\begin{equation}
(p_h, \nabla \cdot (\mathbf{v} - \pi_h \mathbf{v})) = 0, \quad \forall p_h \in \mathbb{W}_0
\end{equation}
Assuming shape regularity defined in~\ref{def:shaperegularity}, $\pi_h$ should have the 
error bound as
\begin{equation}
\norm{\mathbf{v} - \pi_h \mathbf{v}}_{1,\Omega} \leq Ch^m\norm{\mathbf{v}}_{m+1,\Omega}, \quad m=0,1.
\end{equation}
\end{lem}

\begin{proof}
From the definition of degree of freedom~\eqref{eq:nodaldof} and~\eqref{eq:edgedof}, we decompose the projection
operator $\pi_h$ as
\begin{equation}
		  \pi_h \mathbf{v} = \mathcal{I}_{\mathcal{DS}_1} \mathbf{v} + 
		  \sum_{i=0}^3 \alpha_i \mathbf{b}_i,
\end{equation}
where the coefficient 
\begin{equation}
\alpha_i = \Big( \int_{e_i} (\mathbf{v} - \mathcal{I}_{\mathcal{DS}_1} \mathbf{v})\cdot \pmb{\nu}_i d\sigma \Big) / \int_{e_i} \mathbf{b}_i \cdot 
\pmb{\nu}_i d\sigma.
\end{equation}
We use the approximation property of interpolation operator in direct derendipity space in Lemma~\ref{lem:approximation},
\begin{equation}
\norm{\mathbf{v} - \mathcal{I}_{\mathcal{DS}_1}\mathbf{v}}_{m,\Omega} \leq h_E^{2-m} \lvert \mathbf{v} \rvert_{2,E^*}, \quad m= 0,1,
\end{equation}
where $E^* = \Cup_{F\in \mathcal{T}_h, F\cap E \neq \empty}$ defined in Lemma~\ref{lem:interpolationop}.

Lemma~\ref{lem:intestimate} indicates :
\begin{equation}
\lvert\mathbf{b}_i\rvert_{m,E} \leq C(\sigma^*)h_E^{1-m}\lvert \tilde{\mathbf{b}}_i \rvert_{1,\tilde{E}} 
\leq C(\sigma^*)h_E^{1-m}, \quad m = 0,1.
\end{equation}
We continue to estimate $\lvert \alpha_i \rvert$ as
\begin{equation}
\int_{e,i} \varphi^{(2)}_{e,i} ds = \lvert e_i \rvert \int_{\tilde{e}_i} \tilde{\varphi}^{(2)}_{e,i} d\tilde{s}.
\end{equation}
Then
\begin{equation}
\lvert \alpha_i \rvert \leq C \int_{\tilde{e},i} \lvert\tilde{\mathbf{v}} - \tilde{\mathcal{I}}_{\mathcal{DS}_1} \tilde{\mathbf{v}}\rvert ds
\end{equation}

Trace theorem, poincare inequality $\dots$

The final estimate is then
\begin{equation}
		  \lvert\alpha_i \rvert \lvert \mathbf{b}_i \rvert_{m,E} \leq C h^{k-m} \vert \mathbf{v} \rvert_{k,E^*}.
\end{equation}
\end{proof}

Our space $\mathbb{V}_1^{\mathcal{BR}} \times \mathbb{W}_0$ form a 
$\int$-$\sup$ stable pair.
\begin{thm}
For a convex domain $\Omega$, there exists a constant $\gamma > 0$ such that
		  \begin{equation}
					 \inf_{\omega \in \mathbb{W}_0} \sup_{\mathbf{v} \in \mathbb{V}_1^{\mathcal{BR}}} \frac{(\nabla \cdot \mathbf{v}, \omega)}{\norm{\mathbf{v}}_1\norm{\omega}_0} \geq \gamma >0.
		  \end{equation}
\end{thm}
\begin{proof}
		  \begin{equation}
					 \sup_{\mathbf{v} \in \mathbb{V}_1^{\mathcal{BR}}} \frac{(\nabla \cdot \mathbf{v}, \omega)}{\norm{\mathbf{v}}_1\norm{\omega}_0} \geq \frac{(\nabla \cdot \pi_h\mathbf{\varphi}, \omega)}{\norm{\pi_h\mathbf{\varphi}}_1 \norm{\omega}_0} \geq \frac{\norm{\omega}_0}{C \norm{\mathbf{\varphi}_1}} \geq \gamma > 0,
		  \end{equation}
\end{proof}
where $\pi$ is a bounded projection operator.
