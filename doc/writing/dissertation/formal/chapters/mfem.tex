\subsection{$H(\text{div})$ Conforming Element}
We have some choice for $H(\text{div})$ confirming space. 
For example, we can choose the famous Ravier-Thomas element 
by~\cite{10.1007/BFb0064470}. The serendipity space is
the precursor of the Brezzi-Douglas-Marini element by~\cite{BDM}.
We recover Arbogast-Correa element by~\cite{doi:10.1137/15M1013705} if we 
use mapped direct serendipity space.
We generated a reduced $H(\text{div})$ approximation, that $\nabla \cdot \mathbf{u}$
is approximated one less order than approximation of $\mathbf{u}$.

We write the rotated 2D exact sequence for energy space as
\begin{alignat}{3}\label{eq:exactsequence}
&H^1 \xrightarrow[]{\text{curl}} H&&(\text{div}) \xrightarrow[]{\text{div}} &&L^2 \\
&\cup                            &&\cup                                     &&\cup\\
&\mathcal{DS}_2  \xrightarrow[]{\text{curl}} &&\mathbf{V}_r^{r-1} \xrightarrow[]{\text{div}} &&\mathbb{P}_{r-1}.
\end{alignat}
We define the first order reduced $H(\text{div})$ conforming space with $\mathcal{DS}_2$ as
\begin{equation}\label{eq:defhdiv}
		  \mathbb{V}_1^0 = \mathbb{P}_1^2 \oplus \text{curl}\mathbb{S}_2^{\mathcal{DS}}.
\end{equation}
The reduced space $\mathbb{V}_1^0$ can be given a global basis with the normal
components of the shape functions merged continuously on the shared edge of
two elements.
The dimension of the derived space is 
\begin{equation}\label{eq:hdivdim}
		  \text{dim}(\mathbb{V}_1^0) = 2\times 3 + 2 = 8.
\end{equation}
We need two independent basis functions on each edge of a quadrilateral $E$,
one of which has a vanishing average flux on the edge and the other one has a non-vanishing 
average flux on the edge.
On each edge, we define two independent basis functions.
$\pmb{\varphi}_{l,i}$ and $\pmb{\varphi}_{c,i}$, such that
\begin{equation}\label{eq:constbasishdiv_show}
\pmb{\varphi}_{c,i}\cdot \pmb{\nu}_j \rvert_{e_j} = \left \{
		 \begin{aligned}
					&0, \hspace{0.8cm} j \neq i, \\
					&1, \hspace{0.8cm} j=i.
		 \end{aligned}
		 \right.
\end{equation}
\begin{equation}\label{eq:linearbasishdiv_show}
		  \pmb{\varphi}_{l,i}\cdot \pmb{\nu}_j \rvert_{e_j} = \left \{
					\begin{aligned}
							  &0, \hspace{0.8cm} j \neq i, \\
							  &l_i(\mathbf{x}),\hspace{0.2cm} j=i,
					\end{aligned}
					\right.
\end{equation}
where $l_i$ is a non vanishing linear polynomial defined on edge $e_i$ 
orthogonal to one, i.e., $\int_{e_i}l_i ds = 0$.

To begin with, we write curl of the auxiliary rational function $R_i$~\eqref{eq:defR} as
\begin{equation}\label{eq:curlauxiliary}
		 \text{curl}R_i(\mathbf{x}) = 
					 \frac{\pmb{\tau}_{i+2}(\mathbf{x}) \lambda_i(\mathbf{x}) - \pmb{\tau}_{i}(\mathbf{x}) \lambda_{i+2}(\mathbf{x})}{(\lambda_i(\mathbf{x}) + \lambda_{i+2}(\mathbf{x}))^2},
\end{equation}
where the index $i$ is understood with modulo of 4. 
We first present a construction of basis function $\pmb{\varphi}_{l,i}$ as
\begin{equation}\label{eq:linearbasis_raw}
		  \tilde{\pmb{\varphi}}_{l,i} = \text{curl}(\lambda_{i+1}\lambda_{i+3}R_i)
		  =(\pmb{\tau}_{i+1}\lambda_{i+3} + \lambda_{i+1}\pmb{\tau}_{i+3})R_i + 
		  \lambda_{i+1} \lambda_{i+3}\text{curl}R_i.
\end{equation}
We then normalize it such that $l_i$ varying from $-1$ to $1$ as
\begin{equation}\label{eq:linearbasis}
		  \pmb{\varphi}_{l,i} = \frac{\tilde{\pmb{\varphi}}_{l,i}(\mathbf{x})
		  \lvert e_i \rvert}{4\lambda_{i+1}(\mathbf{x}_{e,i}) \lambda_{i+3}(\mathbf{x}_{e,i})R_i(\mathbf{x}_{e,i})},
\end{equation}
where the index $i$ is understood in the sense of modulo 4, and 
edge length $\lvert e_i \rvert = \lvert \mathbf{x}_{v,i} - \mathbf{x}_{v,i+3} \rvert$. In Figure~\ref{fig:varphil},
we draw $\pmb{\varphi}_{l,2}$ on the left element and $\pmb{\varphi}_{l,0}$ on
the right element sharing a edge,
and show normal flux
joining continuously as a linear function on the shared edge.

The construction of a the constant basis function is a bit more complicated.
We follow the process by~\cite{directSerendipity}.
We start by defining a linear nodal function $\phi^*_{v,i} \in \mathcal{DS}_{2}(E)$
such that $\phi^*_{v,i}(\mathbf{x}_{v,k}) = \delta_{ik}$. Its restriction to 
each edge of $E$ is a linear function as
\begin{equation}\label{eq:phi*}
		  \phi^*_{v,i}(\mathbf{x}) = \varphi^2_{v,i}(\mathbf{x}) + \big[\frac{1}{2}\varphi^2_{e,i} + \frac{1}{2}\varphi^2_{e,i+1} \big],
\end{equation}
where the index $i$ is understood in modulo of 4. Then define $\pmb{\varphi}^*_{v,i}(\mathbf{x}) = \text{curl} \phi^*_{v,i}$ so that
\begin{equation}
		  \pmb{\varphi}^*_{v,i}\cdot \pmb{\nu}_j \rvert_{e,j} = \left \{
					 \begin{aligned}
								&1/\lvert e_i \rvert, \hspace{0.8cm} j=i,\\
								&-1/\lvert e_i \rvert, \hspace{0.3cm} j=i+1,\\
								&0,                 \hspace{1.6cm}   j=i+2,i+3.
					 \end{aligned}
					 \right.
\end{equation}
We then define the "constant" basis function defined on the edge as
\begin{equation}\label{eq:constbasis}
		  \pmb{\varphi}_{e,i}= \frac{(\mathbf{x}_{v,i} - \mathbf{x}_{v,i+2}) \cdot \pmb{\nu}_{i+1} \lvert e_{i+1} \rvert\pmb{\varphi}^*_{v,i} + \mathbf{x} - \mathbf{x}_{v,i+2}}
		  {(\mathbf{x}_{v,i} - \mathbf{x}_{v,i+2}) \cdot \pmb{\nu}_{i+1} \lvert e_{i+1} \rvert / \lvert e_i \rvert + (\mathbf{x}_{v,i} - \mathbf{x}_{v,i+2})\cdot \pmb{\nu}_i},
\end{equation}
where the index $i$ is again understood in the sense of modulo of 4. The defined basis function $\pmb{\varphi}_{e,i}$ has flux of constant value $1$ on $e_i$ and $0$ 
on all the other edges. In Figure~\ref{fig:varphic},
we draw $\pmb{\varphi}_{c,2}$ on the left element and $\pmb{\varphi}_{c,0}$ on
the right element sharing a edge,
and show normal flux
joining continuously as a constant $1$ on the shared edge.

We consider the following space for each element $E \in \mathcal{T}_h$ as
\begin{equation}\label{eq:br}
		  V_1^{\mathcal{BR}}(E) = (\mathcal{DS}_1(E))^2 \oplus \text{span}\{\mathbf{b}_0, \mathbf{b}_1, \mathbf{b}_2, \mathbf{b}_3\},
\end{equation}
where $\mathbf{b}_i$ is a quadratic bubble function defined on each edge $e_i$.
We pick previously defined function $\varphi_{e,i} \in \mathcal{DS}_2$ as
\begin{equation}\label{eq:bubble}
		  \mathbf{b}_i = \varphi_{e,i}(\mathbf{x})\mathbf{n}_i
\end{equation}
which is quadratic when restricted to edge $e_i$. 
\begin{lem}\label{lem:unisolvence}
		  For an element $E\in \mathcal{T}_h$, $\mathbf{u}_h$ is uniquely defined
		  by the following degrees of freedom:

		  \text{(i)} for corner point $v_i$ and Cartesian direction unitnormal vectors $\{\mathbf{i},\mathbf{j}\}$,
		  \begin{equation*}
					 DOF^{\text{(i)}}_{(v_i,\mathbf{i})}(\mathbf{u}_h) = \mathbf{u}_h(v_i)\cdot \mathbf{i}, 
					 \quad DOF^{\text{(i)}}_{(v_i,\mathbf{j})}(\mathbf{u}_h) = \mathbf{u}_h(v_i)\cdot \mathbf{j};
		  \end{equation*}

		 \text{(ii)} for edge $e_i$,
		  \begin{equation*}
					 DOF^{(ii)}_{e_i} = \langle \mathbf{u}_h \cdot \pmb{\nu}_i, 1 \rangle,
		  \end{equation*}
\end{lem}
\begin{proof}
We prove the lemma by construction. 
We pick nodal badis on vertex $v_i$ in two Cartesian direction as
		  \begin{equation}
					 N_{v_i,x} = \begin{bmatrix}
								\psi_{v,i}(\mathbf{x})\\
								0
					 \end{bmatrix}, \quad  N_{v_i,y} = \begin{bmatrix}
								0\\
								\psi_{v,i}(\mathbf{x})
					 \end{bmatrix}
		  \end{equation}
We pick edge basis on edge $e_i$ as
\begin{equation}
		  N_{e} = \varphi_{e,i}\pmb{\nu}_i
\end{equation}
Note that bubble functions are quadratic when restricted to edge $e_i$, while
nodal basis functions are linear when restricted to edge $e_i$, therefore bubble
functions cannot be in the same space as nodal basis function.
We conclude
that this element is well-defined, i.e., the degrees of freedom uniquely 
determine the function.
\end{proof}

We need a $\pi$ operator, which mappes $(H^1(E))^2$ onto $V_1^{\mathcal{BR}}$ 
for our spaces. For $\mathbf{v} \in (H^1(\Omega))^2$, 
let $\pi \mathbf{v} \in \mathbb{V}_1^{\mathcal{BR}}$ be defined as the 
interpolant of the degrees of freedom from~\ref{lem:unisolvence}. For each $F$,
we require the following as

(i) For each corner point $v \in \partial F$
\begin{equation}
		  \pi \mathbf{v}(\mathbf{x}_v) \cdot \pmb{i} = \mathcal{I}_{\mathcal{DS}_1}\mathbf{v}(\mathbf{x}_v) \cdot \pmb{j}.
\end{equation}

(ii) On each edge $e \subset \partial E$,
\begin{equation}
		  \langle \pi \mathbf{u} \cdot \pmb{\nu}, \lambda_0 \rangle = \langle \mathbf{u} \cdot \pmb{\nu}, \lambda_0 \rangle_e \hspace{0.2cm} \forall \lambda_0 \in P_0(e).
\end{equation}

The interpolation operator $I_{\mathcal{DS}_1}$ has been defined in~\ref{eq:interpolation}.
We claim that we can represent for $\mathbf{u} \in (H^1(\Omega))^2$,
\begin{equation}
		  \pi \mathbf{u} = \mathbf{I}_{\mathcal{DS}_1} \mathbf{u} + 
		  \sum_{i=0}^3 \alpha_i \mathbf{b}_i,
\end{equation}
where the coefficient $\alpha_i = \Big( \int_{e_i} (\mathbf{u} - \mathcal{I}_{\mathcal{DS}_1} \mathbf{u})\cdot \pmb{\nu}_i d\sigma \Big) / \int_{e_i} \mathbf{b}_i \cdot 
\pmb{\nu}_i d\sigma$. It is obvious that $\pi \mathbf{u} \in \mathbb{V}_1^{\mathcal{BR}}$.
We now prove the approximability and stability of our projection operator and eventually derive the $\inf-\sup$ condition.

\begin{lem}\label{lem:inf-sup-br}
		  Assuming that $u\in (H1(\Omega))^2$.

		  (i) there exists a bounded linear operator $\pi$

		  (ii) $\forall \mathbf{v} \in (H^1(\Omega))^2$
\end{lem}

\section{Coupled Degenerate System}
Consider a degenerate scaled darcy-stokes coupled boundary-value problem:

Find velocity-pressure potential pair $(\check{\mathbf{v}}_r, \check{q}_l)$ and 
$(\check{\mathbf{v}}_s, \check{q})$ such that 
\begin{alignat}{2}
\label{eq:dimensionlessdarcy_flow_sc_bv}
		  &\check{\mathbf{v}}_r
		  + \phi^{1+\Theta}\nabla(\phi^{-1/2}\check{q}_l)
        &&= 0,\\
\label{eq:dimensionlessconsMass_flow_sc_bv}
		  &\phi^{-1/2}\nabla \cdot(\phi^{1+\Theta}\check{\mathbf{v}}_r)
        + \frac{1}{1-\phi}(\check{q}_l - \phi^{1/2}\check{q}) &&=0,\\
\label{eq:dimensionlessstokes_flow_sc_bv}
		  &\nabla \check{q} - \nabla \cdot \check{\tau}(\check{\mathbf{v}}_s) &&= -(1-\phi)\mathbf{n},\\
\label{eq:dimensionlesscompaction_flow_sc_bv}
		  &\nabla \cdot \check{\mathbf{v}}_s
        - \frac{\phi^{1/2}}{1-\phi}(\check{q}_l - \phi^{1/2}\check{q}) &&=0,
\end{alignat}
and
\begin{equation}\label{eq:bv}
		  \begin{aligned}
					 \check{\mathbf{v}}_r &= \mathbf{g}_{r} \hspace{0.5cm} \text{on} \hspace{0.2cm} \Gamma_{u,D}, \\
					 \check{\mathbf{v}}_s &= \mathbf{g}_{s} \hspace{0.5cm} \text{on} \hspace{0.2cm} \Gamma_{u,S}, \\
					 p &= p_0 \hspace{0.5cm} \text{on} \hspace{0.2cm} \Gamma_{i,D}, \\
					 \pmb{\tau}\cdot \pmb{\nu} - p &= \pmb{t}_0 \hspace{0.56cm} \text{on} \hspace{0.2cm} \Gamma_{i,S},
		  \end{aligned}
\end{equation}
where $\Gamma_{u,D}$ and $\Gamma_{u,S}$ denote the part of boundary we assign essential boundary conditions for 
Darcy and Stokes problem respectively, 
and $\Gamma_{i,D}$ and $\Gamma_{i,S}$ represent the part of boundary we assign natural boundary conditions 
respectively. We have $\Gamma_{u,D} \cup \Gamma_{i,D} = \partial \Omega$ and $\Gamma_{u,S} \cup \Gamma_{i,S} = \partial \Omega$.

We follow the procedure by~\cite{DegenerateDarcy} to discretize this boundary value problem.
To begin weith, a new Hilbert space is defined for this 
specific scaled formulation, where
the scaled velocity $\check{\mathbf{v}}_r$ should lie in, so
\begin{equation}\label{eq:scaledHspace}
		  \mathbb{V}^{\phi}_d = H_{\phi}(\text{div}; \Omega) = \Big\{ \mathbf{v} 
		  \in (L^2(\Omega))^2: \phi^{-1/2}\nabla \cdot (\phi^{1+\Phi} \mathbf{v})
		  \in L^2(\Omega) \Big\},
\end{equation}
which equipped with the inner product as
\begin{equation}\label{eq:scaledinnerp}
		  (\mathbf{u}_{\phi}, \mathbf{v}_{\phi})_{\mathbb{V}^{\phi}_r} = 
		  (\mathbf{u}_{\phi}, \mathbf{v}_{\phi}) + 
		  (\phi^{-1/2} \nabla \cdot (\phi^{1+\Theta}\mathbf{u}_\phi), \phi^{-1/2} \nabla \cdot (\phi^{1+\Theta}\mathbf{v}_\phi)).
\end{equation}
The normal trace on $\partial \Omega$ is well-defined as well as
\begin{equation}\label{eq:tracescaled}
		  \norm{\phi^{1/2+\Theta}\mathbf{v}_\phi\cdot \nu}_{-1/2, \partial \Omega} < C_{\Omega} \Big\{ \norm{\mathbf{v}_\phi} + \norm{\phi^{-1/2}\nabla \cdot (\phi^{1+\Theta}\mathbf{v}_\phi)} \Big\}.
\end{equation}

\subsection{Local Mass Conservation}
Before we continue the discussion, we need to make sure the formulation is 
well-defined by not dividing by zero.
The term $\phi^{-1/2}$ in the symmetric equations~\eqref{eq:weak_darcy} 
and~\eqref{eq:weak_continuous} causes trouble 
when there is no melting, i.e., $\phi = 0$.
In order to deal with it, we first define a element-averaged $\overline{\phi}_E$ 
as
\begin{equation}\label{eq:averaged_phi}
		  \overline{\phi}_E = \frac{1}{\lvert E \rvert} \int_E \phi(\mathbf{x}) d\mathbf{x},
\end{equation}
where $\lvert E \rvert$ is the area of element $E$.
Let
\begin{equation}\label{eq:newvariable_phi}
		  \phi_E = \left\{
					 \begin{aligned}
								&\overline{\phi}_E  \hspace{0.2cm} \text{if}\hspace{0.2cm} \overline{\phi}_E \neq 0,\\
								&0 \hspace{0.5cm} \text{if} \hspace{0.2cm} \overline{\phi}_E = 0.
					 \end{aligned}
					 \right.
\end{equation}
We now replace the $\phi^{-1/2}$ with $\phi_E^{-1/2}$. Then we expand the integral
with divergence theorem as
\begin{equation}\label{eq:divtheorem}
		  \begin{aligned}
					 \int_E \phi_E^{-1/2} \nabla \cdot (\phi^{1+\Theta} \pmb{\psi}_{r,h})\omega_l &=\int_{\partial E} \phi_E^{-1/2} \phi^{1+\Theta} \pmb{\psi}_{r,h}\cdot \pmb{\nu} \omega_f ds \\
					 &-\int_{E}\phi_E^{-1/2}\phi^{1+\Theta}\pmb{\psi}_{r,h} \cdot \nabla \omega_l d\mathbf{x}.
		  \end{aligned}
\end{equation}
The second term vanishes when $\omega_l$ is constant on element $E$.
We define an auxiliary auxiliary space $\omega_E$ begin 1 on $E$ and 0 elsewhere.
We sum equations~\eqref{eq:weak_compaction} and~\eqref{eq:weak_continuous} 
together 
assuming that $\omega = \omega_E \in \mathbb{W}_{S}$ and $\omega_l = \phi_E^{-1/2} \omega_E \in \mathbb{W}_{D}$. We can achieve local mass conservation of the
phase averaged mixture if we
replace the term $\phi^{1/2}$ in equation~\eqref{eq:weak_compaction} with
$\phi_E^{1/2}$ as
\begin{equation}
		  \int_{E}\nabla \cdot (\phi^{1+\Theta}\check{\mathbf{v}}_r + \check{\mathbf{v}}_s) d\mathbf{x} = 0.
\end{equation}
We now propose our local mass conserved version as
\begin{alignat}{2}
		  \label{eq:weak_stokes_mass}
		  &(\check{\pmb{\tau}}(\bold{v}_{s,h}),\nabla \pmb{\psi}_s) - (\check{q}_h, \nabla \cdot \pmb{\psi}_s) = -((1-\phi)\bold{n}, \pmb{\psi}_s) &&\forall\pmb{\psi}_s \in \mathbb{V}_1^{\mathcal{BR}},\\
		  \label{eq:weak_compaction_mass}
		  &(\nabla \cdot \check{\bold{v}}_{s,h}, \omega)  - \Big(\frac{\phi_E^{1/2}}{1-\phi} (\check{q}_{l,h} - \phi^{1/2}\check{q}_h), \omega\Big) = 0 &&\forall \omega \in \mathbb{W}_0^{\mathcal{BR}},\\
					\label{eq:weak_darcy_mass}
					 &(\bold{\check{v}}_{r,h},\pmb{\psi}_r)  - (\check{q}_{l,h},\phi_E^{-1/2} \nabla \cdot (\phi^{1+\Theta}\pmb{\psi}_{r,h})) = 0, &&\forall \pmb{\psi}_r \in \mathbb{V}_1^0,\\
					 \nonumber
					 &(\phi_E^{-1/2} \nabla \cdot (\phi^{1+\Theta}\check{\bold{v}}_{r,h}),\omega_f) + \Big(\frac{1}{1-\phi}(\check{q}_{l,h} - \phi^{1/2}\check{q}_h),\omega_f \Big) &&= 0 \\
					 \label{eq:weak_continuous_mass}
		  &					 &&\forall \omega_f \in 
		  \mathbb{W}_{0}.
\end{alignat}
