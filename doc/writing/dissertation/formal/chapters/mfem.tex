\section{Coupled Degenerate Darcy-Stokes System}
Now we consider a scaled nondimensionalized degenerate Darcy-Stokes coupled boundary-value problem:

Find two velocity-pressure potential pairs $(\check{\mathbf{v}}_r, \check{q}_l)$ and 
$(\check{\mathbf{v}}_s, \check{q})$ such that 
\begin{alignat}{2}
\label{eq:dimensionlessdarcy_flow_sc_bv}
		  &\check{\mathbf{v}}_r
		  + \phi^{1+\Theta}\nabla(\phi^{-1/2}\check{q}_l)
        &&= 0,\\
\label{eq:dimensionlessconsMass_flow_sc_bv}
		  &\phi^{-1/2}\nabla \cdot(\phi^{1+\Theta}\check{\mathbf{v}}_r)
        + \frac{1}{1-\phi}(\check{q}_l - \phi^{1/2}\check{q}) &&=0,\\
\label{eq:dimensionlessstokes_flow_sc_bv}
		  &\nabla \check{q} - \nabla \cdot \check{\tau}(\check{\mathbf{v}}_s) &&= -(1-\phi)\mathbf{n},\\
\label{eq:dimensionlesscompaction_flow_sc_bv}
		  &\nabla \cdot \check{\mathbf{v}}_s
        - \frac{\phi^{1/2}}{1-\phi}(\check{q}_l - \phi^{1/2}\check{q}) &&=0,
\end{alignat}
and
\begin{equation}\label{eq:bv}
		  \begin{aligned}
					 \check{\mathbf{v}}_r\cdot \pmb{\nu} &= g_{r} \hspace{0.5cm} \text{on} \hspace{0.2cm} \Gamma_{u,D}, \\
					 \check{\mathbf{v}}_s &= \mathbf{g}_{s} \hspace{0.5cm} \text{on} \hspace{0.2cm} \Gamma_{u,S}, \\
					 p &= p_0 \hspace{0.5cm} \text{on} \hspace{0.2cm} \Gamma_{i,D}, \\
					 \pmb{\tau}\cdot \pmb{\nu} - p &= \pmb{t}_0 \hspace{0.56cm} \text{on} \hspace{0.2cm} \Gamma_{i,S},
		  \end{aligned}
\end{equation}
where $\Gamma_{u,D}$ and $\Gamma_{u,S}$ denote the part of boundary we assign essential boundary conditions for 
Darcy and Stokes problem respectively, 
and $\Gamma_{i,D}$ and $\Gamma_{i,S}$ represent the part of boundary we assign natural boundary conditions 
respectively. We have $\Gamma_{u,D} \cup \Gamma_{i,D} = \partial \Omega$ and $\Gamma_{u,S} \cup \Gamma_{i,S} = \partial \Omega$.

We expand the gradient term in equation~\eqref{eq:dimensionlessdarcy_flow_sc_bv}
and the divergence term in equation~\eqref{eq:dimensionlessconsMass_flow_sc_bv} as
\begin{equation}
		  \begin{aligned}
					 \phi^{1+\Theta} \nabla (\phi^{-1/2} \check{q}_l) &= \phi^{1/2+\Theta} \nabla \check{q}_l +  \phi^{\Theta-1/2} \check{q}_l \nabla \phi, \\
					 \phi^{-1/2} \nabla \cdot (\phi^{1+\Theta} \check{\mathbf{v}}_r) &=
					 \phi^{1/2+\Theta} \nabla \cdot \check{\mathbf{v}}_r + 
					 \phi^{\Theta - 1/2} \nabla \phi \cdot \check{\mathbf{v}}_r,
		  \end{aligned}
\end{equation}
where they are well-defined provided that
\begin{equation}\label{eq:phiboundness}
		  \phi^{\Theta - 1/2} \nabla \phi \in (L^{\infty}(\Omega))^2.
\end{equation}

A scaled Hilbert space is defined as follows for this 
specific scaled formulation, where
the scaled velocity $\check{\mathbf{v}}_r$ should lie in.
\begin{defn}
Consider a scaled Hilbert space:
\begin{equation}\label{eq:scaledHspace}
		  \mathbb{V}^{\phi}_D = H_{\phi}(\text{div}; \Omega) = \Big\{ \mathbf{v} 
		  \in (L^2(\Omega))^2: \phi^{-1/2}\nabla \cdot (\phi^{1+\Phi} \mathbf{v})
		  \in L^2(\Omega) \Big\},
\end{equation}
which is a Hilbert space equipped with the inner product
\begin{equation}\label{eq:scaledinnerp}
		  (\mathbf{u}, \mathbf{v})_{\mathbb{V}^{\phi}_D} = 
		  (\mathbf{u}, \mathbf{v}) + 
		  (\phi^{-1/2} \nabla \cdot (\phi^{1+\Theta}\mathbf{u}), \phi^{-1/2} \nabla \cdot (\phi^{1+\Theta}\mathbf{v})).
\end{equation}
The normal trace on $\partial \Omega$ is well-defined as
\begin{equation}\label{eq:tracescaled}
		  \norm{\phi^{1/2+\Theta}\mathbf{v}\cdot \pmb{\nu}}_{H^{-1/2}(\partial \Omega)} < C_{\Omega} \Big\{ \norm{\mathbf{v}}_{L^2(\Omega)} + \norm{\phi^{-1/2}\nabla \cdot (\phi^{1+\Theta}\mathbf{v})}_{L^2(\Omega)} \Big\}
\end{equation}
		  subsequently, we have the space $H_\phi^{-1/2}(\text{div};\partial \Omega)$ understood as the image of this normal trace operator.
\end{defn}

\subsection{The scaled mixed finite element method}
We discretize the scaled mixed formulation with
two space pairs $\mathbb{V}_{D,h} \times \mathbb{W}_{D,h} \subset H(\text{div}; \Omega) \times L^2(\Omega)$
and $\mathbb{V}_{S,h} \times \mathbb{W}_{S,h} \subset (H^1(\Omega))^2 \times L^2(\Omega)$.
Recall the aforementioned inf-sup stable mixed finite element space pairs on $E \in \mathcal{T}_h$:
\begin{itemize}
\item For the Darcy part of the system.
\begin{equation}
\begin{aligned}
		  \mathbb{V}_{D,h}(E) = \mathbb{V}_1^0(E) &= (\mathbb{P}_1(E))^2 \oplus 
		  \text{curl}\,\mathbb{S}_2^{\mathcal{DS}}(E) \\
					 \mathbb{W}_{D,h}(E) = \mathbb{W}(E) &= \mathbb{P}_0(E)
\end{aligned}
\end{equation}

\item For the Stokes part of the system.
\end{itemize}
\begin{equation}
\begin{aligned}
		  \mathbb{V}_{S,h}(E) = \mathbb{V}^{\mathcal{BR}}_1(E) &= (\mathcal{DS}_1(E))^2 \oplus \text{span}\{\mathbf{b}_0,\mathbf{b}_1,\mathbf{b}_2,\mathbf{b}_3 \} \\
		  \mathbb{W}_{S,h}(E) = \mathbb{W}(E)   &= \mathbb{P}_0(E).
		  \end{aligned}
\end{equation}
We can impose essential boundary conditions by projecting extensions
$\tilde{\mathbf{g}}_r$ and $\tilde{\mathbf{g}}_s$ into
the finite element spaces as $\hat{\mathbf{g}}_r \in \mathbb{V}_{D,h}$ and $\hat{\mathbf{g}}_s \in \mathbb{V}_{S,h}$.

We also define 
\begin{equation}
		  \begin{aligned}
					 \mathbb{V}_{D,h,0} &= \{\mathbf{v} \in \mathbb{V}_{D,h} : \mathbf{v}\cdot \pmb{\nu} = 0 \hspace{0.2cm} \text{on} \hspace{0.2cm} \Gamma_{u,D}\} \\
					 \mathbb{V}_{S,h,\mathbf{0}} &= \{\mathbf{v} \in \mathbb{V}_{S,h} : \mathbf{v} = \mathbf{0} \hspace{0.2cm} \text{on} \hspace{0.2cm} \Gamma_{u,S}\}

$\check{\mathbf{v}}_{s,h} \in \mathbf{V}_{S,h,\mathbf{0}} + \hat{\mathbf{g}}_s$ and $\check{q}_h \in \mathbb{W}_{S,h}$ such that

		  &(\pmb{\tau}(\check{\bold{v}}_{s,h}),\nabla \pmb{\psi}_{s,h}) - (\check{q}_h, \nabla \cdot \pmb{\psi}_{s,h}) = -((1-\phi)\bold{n}, \pmb{\psi}_{s,h}), &&\forall\pmb{\psi}_{s,h} \in \mathbb{V}_{S,h,\mathbf{0}},\\

					 \label{eq:weak_continuous}
		  &					 &&\forall \omega_{l,h} \in 
		  \mathbb{W}_{D,h}.
\end{alignat}

\subsection{Local Mass Conservation}
Before we continue the discussion, we need to make sure that the formulation is 
well-defined, i.e., not dividing by zero.
The term $\phi^{-1/2}$ in the symmetric equations~\eqref{eq:weak_darcy} 
and~\eqref{eq:weak_continuous} causes trouble 
when there is no melting, i.e., $\phi = 0$.
In order to deal with it, we define an element-averaged variable $\phi_E$ 
as
\begin{equation}\label{eq:averaged_phi}
		  \phi_E = \frac{1}{\lvert E \rvert} \int_E \phi(\mathbf{x})\, d\mathbf{x},
\end{equation}
where $\lvert E \rvert$ is the area of element $E$.
Then the two terms involving divergence is modified as
\begin{equation}\label{eq:newvariable_phi}
		  \phi^{-1/2}\nabla \cdot (\phi^{1+\Theta} \mathbf{v}) \rvert_E = \left\{
					 \begin{alignedat}{2}
								&\phi_E^{-1/2}\nabla \cdot (\phi^{1+\Theta} \mathbf{v})  \hspace{0.2cm} &&\text{if}\hspace{0.2cm} \overline{\phi}_E \neq 0,\\
								&0 \hspace{0.5cm} &&\text{if} \hspace{0.2cm} \overline{\phi}_E = 0,
					 \end{alignedat}
					 \right.
\end{equation}
where the point-wise term $\phi^{-1/2}$ is modified to be a cell-averaged term $\phi_E^{-1/2}$. 
We expand the integral term involving divergence according to 
the divergence theorem as
\begin{equation}\label{eq:divtheorem}
		  \begin{aligned}
					 \int_E \phi_E^{-1/2} \nabla \cdot (\phi^{1+\Theta} \pmb{\psi}_{r,h})\omega_{l,h}\, d\mathbf{x} &=\int_{\partial E} \phi_E^{-1/2} \phi^{1+\Theta} \pmb{\psi}_{r,h}\cdot \pmb{\nu} \omega_{l,h}\, ds \\
					 &+\int_{E}\phi_E^{-1/2}\phi^{1+\Theta}\pmb{\psi}_{r,h} \cdot \nabla \omega_{l,h}\, d\mathbf{x}.
		  \end{aligned}
\end{equation}
The second term vanishes when $\omega_{l,h}$ is constant on element $E$.
We now propose our local mass conserved version by
replacing the term $\phi^{1/2}$ in equation~\eqref{eq:weak_compaction} with
$\phi_E^{1/2}$ as

\subsection{Recovery of the unscaled Darcy variables}
The unscaled Darcy potential is $\check{q}_f = \phi^{-1/2}\tilde{\check{q}}_l$, see~\eqref{eq:scaledVariables}. 
We can reterive an approximate Darcy potential. We assume that $\check{q}_{f,h} \in \mathbb{W}_{D,h}$, so on
an element $E$ we define
\begin{equation}
		  \check{q}_{l,h}\rvert_{E} = \left\{
					 \begin{alignedat}{2}
								&\phi_E^{-1/2} \tilde{\check{q}}_{l,h}\lvert_E \quad &&\text{if} \phi_E \neq 0,\\
								&0                                                   &&\text{if} \phi_E = 0,
					 \end{alignedat}
					 \right.
\end{equation}
where the liquid pressure potential $\check{q}_l$ does not hold any physical 
meanings when there is no melting presents ($\phi = 0$).

The relative Darcy velocity is given by $\check{\mathbf{v}}_r = \phi^{\Theta} \tilde{\check{\mathbf{v}}}$, 
see~\eqref{eq:scaledVariables} as well. To maintain local mass balance, we 
enforce that on each edge $e_i \subset \partial E$
\begin{equation}
		  \int_{e_i} \phi^{1+\Theta} \tilde{\check{\mathbf{v}}}_r \cdot \pmb{\nu}_i\, ds = \int_{e_i} \phi \check{\mathbf{v}}_{r,h} \cdot \pmb{\nu} \, ds.
\end{equation}

-\frac{\phi_f^{-1/2}}{1-\phi_f}\begin{pmatrix}x^2 \\ y^2 \end{pmatrix} + \phi_f^{1/2}\begin{pmatrix} 2 \\ 2 \end{pmatrix}\\
\phi_f^{1/2}\nabla \cdot (\check{\bold{v}}_r) + \frac{1}{1-\phi_f}\check{q}_f  &= \frac{-2}{1-\phi_f}(x+y) + \frac{1}{1-\phi_f}2(x+y)=0\\
\nabla \cdot \check{\bold{v}}_s - \frac{\phi_f^{1/2}}{1-\phi_f}\check{q}_f &= \frac{2\phi_f^{1/2}}{1-\phi_f}(x+y) - \frac{2\phi_f^{1/2}}{1-\phi_f}(x+y) = 0\\
-2(1-\phi_f)\nabla \cdot \sigma(\check{\bold{v}}_s) &= -2(1-\phi_f)\nabla \cdot (\mathcal{D}\check{\bold{v}}_s - \frac{1}{3}(\nabla \cdot \check{\bold{v}}_s) \bold{\text{I}}) = 
-\frac{8\phi_f^{1/2}}{3} \begin{pmatrix} 1 \\ 1 \end{pmatrix}
\end{aligned}
\end{equation}

Problem Setup 1: Column Compaction.

In this problemsetup, we perform a convergence test on a 2D grid with a pseudo 1D problem.
We study a compacting column in one direction. The column extends over $y \in [-L,L]$
and has no flow through top and bottom boundaries. We prescribe the following boundary conditions
\begin{itemize}
		  \item Bottom Edge;
					 \begin{itemize}
								\item Essential boundary condition:
										  \begin{equation*}
										  \left\{
													 \begin{aligned}
																 \check{\mathbf{v}}_s &= \mathbf{0};\\
																\mathbf{b} &= \mathbf{0};
													 \end{aligned}\right.
										  \end{equation*}
								\item Essential boundary condition: Normal Darcy flux $\check{\mathbf{v}}_r\cdot \pmb{\nu}= 0$;
					 \end{itemize}
		  \item Left and Right side Edges;
					 \begin{itemize}
								\item y component of Stokes velocity is free, x component is kept as zero;
								\item Darcy velocity, which is understood as flux is also kept zero;
							   \item normal flux dof fixed with a zero flux.
					 \end{itemize}
		  \item Top Edge;
					 \begin{itemize}
								\item Stokes velocity $\check{\mathbf{v}}_s = \mathbf{0}$;
								\item Darcy velocity $\check{\mathbf{v}}_l = \mathbf{0}$;
							   \item Normal flux is also determined to be zero.
					 \end{itemize}
\end{itemize}
We assume no velocity in x direction, $\check{\mathbf{v}}_s \cdot \mathbf{i} = 0$, 
and $\check{\mathbf{v}}_r \cdot \mathbf{i} = 0$, where $\mathbf{i}$ is 
the unit vector in x direction pointing from left to right.
Let $u = \phi \check{\mathbf{v}}_r \cdot \mathbf{j}$
and $v = \check{\mathbf{v}}_s \cdot \mathbf{j}$, where $\mathbf{j}$
is the unit vector in y direction pointing upwards.
\begin{equation}\label{eq:qdcolumncompaction}
		  \begin{aligned}
					 u + \phi^{2} q'_l &= 0, \\
					 u' + \phi(q_l - q_s)&= 0, \\
					 [q_s - \frac{1}{3}(1-4\phi)v']' &= 1-\phi,\\
					 v' - \phi(q_f - q_s) &=0,
		  \end{aligned}
\end{equation}
When no melting presents $\phi = 0$, the equations reduce to 
\begin{equation}
		  u = -v = 0,\quad\text{and}\quad q'_s = 1, \quad\text{and}\quad q_l = 0.
\end{equation}
When melting presents $\phi > 0$, the equations can be reduced to
\begin{equation}\label{eq:ode}
		  \phi^{2+2\Theta}\bigg(\frac{3+\phi - 4\phi^2}{3\phi}u' \bigg)' - u = \phi^{2+2\Theta}(1-\phi),
\end{equation}
where all other quantities can be determined according to $u$ as
\begin{equation}
		  v=-u, \quad \text{and}\quad  q'_l = -\phi^{-2-2\Theta}u, \quad \text{and}\quad q_s = u'/\phi + q_l.
\end{equation}
When $\phi$ is constant, we define an auxiliary function $R$ as
\begin{equation}
		  R = \bigg( \phi^{2+2\Theta}\frac{3+\phi - 4\phi^2}{3\phi} \bigg)^{-1/2}.
\end{equation}

In the previous literature, we made the assumption that $\phi = \phi_0 z^2 \ll 1$.
Here, we present an alternative way of constructing approximation solution of the
second order ordinary differential equation~\eqref{eq:ode}. 
Let $v = u/\phi^2 = u/(\phi_0^2 z^4)$,
\begin{equation}
		  v' = \frac{-4v}{z} + \frac{u'}{\phi_0^2 z^4}.
\end{equation}
Substitute $u$ with $v$, we have 
\begin{equation}\label{eq:nonhomoode}
		  f(u'',u',u,z) = \bigg[ \frac{3+\phi_0z^2 - 4\phi_+z^4}{3\phi_0z^2}\big(\phi_0^2 z^4 v' + 4\phi_0^2 z^3 v\big) \bigg]' - v = 1 - \phi_0z^2.
\end{equation}
We assume that there exists a solution that can be represented as a sum of 
an infinity series.
The solution can be understood as the sum of a particular
solution and a homogenous power-series solution,
which solves the homogenous second order ODE $f(u'',u',u',z) = 0$.
We apply Frobenius method here, assuming the homogenous solution
has the form as
\begin{equation}\label{eq:homogenous}
v_H = (z-z_0)^r \sum_{n=0}^\infty a_n (z-z_0)^n,
\end{equation}
where we pick $z_0 = 0$ for simplicity. We write the homogenous
recurrence as
\begin{equation}\label{eq:recursive}
\begin{aligned}
&\bigg\{ \phi_0 \Big[(n+r-1)(n+r) + 6(n+r) + 4 \Big] - 1 \bigg\} a_n + \\ 
&\bigg\{ \frac{\phi_0^2}{3} \Big[(n+r-2)(n+r+3) + 4 + 2(n+r+2) \Big] \bigg\} a_{n-2} +\\
&\bigg\{ \frac{4\phi_0^3}{3} \Big[ (n+r-4)(n+r+1) +4 +4(n+r) \Big] \bigg\} a_{n-4}  = 0,
\end{aligned}
\end{equation}
where we obtain the indicial equation be equating the coefficient associated to $a_0$ term to zero, i.e.,
\begin{equation}
\phi_0\big[(0+r-1)(0+r) +6(0+r) + 4\big] - 1 = 0,
\end{equation}
where we solve for the positive root as
\begin{equation}
r = \frac{-5+\sqrt{9+4/\phi_0}}{2}.
\end{equation}
Notice that odd and even coefficients are independent against each other, we can decompose the 
homogenous solution into two independent power-series according to even and odd coefficients as
\begin{equation}
v_1 = z^r \sum_{n=0}^{\infty} a_{2n} z^{2n}, \quad v_2 = z^r \sum_{n=0}^{\infty} a_{2n+1} z^{2n+1}.
\end{equation}
Then we proceed to determine the particular solution that solves the
nonhomogenous function $f(u'',u',u,z) = 1-\phi_0z^2$.
\begin{equation}
v_p = \sum_{n=0}^\infty c_n z^n.
\end{equation}
Substituting the particular solution into the non-homogenous ODE~\eqref{eq:nonhomoode}, we obtain the coefficient 
by matching the coefficients on the right hand side according to different orders.
\begin{equation}
\begin{aligned}
c_0 & = \frac{1}{4\phi_0 - 1} \\
c_2 & = \frac{-\phi_0 - 4\phi_0^2 c_0}{18\phi_0 - 1} \\
c_4 & = \frac{80/3 \phi_0^3 c_0  - 10 \phi_0^2 c_2}{40\phi_0 -1}.
\end{aligned}
\end{equation}
The rest of the coefficients can be determined with the recurrence form~\eqref{eq:recursive}.
Omitting the power-series $v_2$ containing only odd coefficient, we have the final form of
the series solution as
\begin{equation}
v = v_p + C v_1,
\end{equation}
and we determine the costant $C$ with boundary condition $v(-2) = 0$, i.e., $C = -v_p(-2)/v_1(-2)$.
Now we redo the convergence test $\phi = \phi_0 z^2$

We solve the full system of equations on a dimensionless 
rectangular domain $x \in [-0.5,0.5]$ and $z \in[-0.5, 0.0]$.
The boundary conditions are imposed by
assigning classical corner flow solution of
Stokes velocity in~\cite{SPIEGELMAN1987137}
as essential boundary condition all around
the computational domain 
\begin{equation}\label{eq:test2}
		  \mathbf{v}_s = \frac{2U_0}{\pi(x^2+y^2)}\begin{pmatrix}
					 \tan^{-1}(\frac{x}{-y})(x^2+y^2) + xy\\
z^2
		  \end{pmatrix},
\end{equation}
where $U_0 = 10^{-9}\text{m/s} \approx 3.2\text{cm/yr}$ is the
