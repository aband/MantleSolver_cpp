\Maroon{Fluid Mechanics} : High-viscosity Stokes flow governs the mantle (3), 
while magma follows Darcy’s law (1), alongside continuity (2) 
and compaction equations (4).
\begin{alignat}{2}
    \label{eq:darcy}
		  &\phi_l (\mathbf{v}_l - \mathbf{v}_s)
		  + \frac{k_{\phi_l}}{\mu_l}\nabla(p_l - \rho_f\mathbf{g})
	    &&= 0,\\
	\label{eq:continuous}
		  &\nabla \cdot (\phi_l\mathbf{v}_l + \phi_s\mathbf{v}_s) &&=0,\\
	\label{eq:stokes}
		  &\nabla\bar p - \nabla \cdot \tau(\mathbf{v}_s) &&= (\rho_s-\rho_l)\mathbf{g},\\
	\label{eq:compaction}
		  &\mu_s\nabla \cdot \mathbf{v}_s - \phi_l(p_l - p_s) &&=0, 
\end{alignat}
\Maroon{Transport} : Trace the bulk composition of one specie (5) and transport
bulk energy with enthalpy method (6).
\begin{equation}\label{eq:specietransportsimplified}
\frac{\partial \overline{c}}{\partial t} + \nabla \cdot \big( \overline{ c \mathbf{v}} - \phi_l \mathcal{D}_{l} \nabla c_l \big) = 0.
\end{equation}
\begin{equation}\label{eq:normalizedenthalpy}
		  \frac{\partial H}{\partial t} + \nabla \cdot \overline{\mathbf{v}} T = 
		  \frac{L}{c_p} \nabla \cdot \phi_s \mathbf{v}_s + \kappa \nabla^2 T + 
		  \frac{\alpha_\rho}{c_p} T \overline{\mathbf{v}} \cdot \mathbf{g},
\end{equation}

\Maroon{Multi-level WENO method} : 
New nonlinear weighting scheme,
\OliveGreen{(Arbogast, Huang and Tian, 2024)}
\begin{equation*}         
\hat{\omega}_l = \frac{\omega_l}{(\sigma_l + \epsilon h^2_0)^{sr_l + \eta_l}} 
\end{equation*}                             

The biasing term $\eta_l$ is defined as following to bias away from small 
stencils when solution is smooth:
\begin{equation*}
\eta_l = \left \{ \begin{aligned}
1, & \quad \text{if} \quad r_l = 1,\\
3, & \quad \text{if} \quad r_l = 2,\\
4, & \quad \text{if} \quad r_l \geq 3,\\
\end{aligned} 
\right .
\end{equation*}

The normalized nonlinear weights are then:
\begin{equation*}
\tilde{\omega}_l = \frac{\hat{\omega}_l}{\sum_k \hat{\omega}_l}
\end{equation*}
